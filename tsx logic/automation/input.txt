import React from 'react';
import {
  useCurrentFrame,
  useVideoConfig,
  interpolate,
  spring,
  AbsoluteFill,
  Easing,
} from 'remotion';

export const compositionConfig = {
  id: 'Scene1_RetiredOfficerIntro',
  durationInSeconds: 10,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene1_RetiredOfficerIntro: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({
      frame: Math.max(0, frame - delay),
      fps,
      config: { damping: 12, mass: 0.4, stiffness: 120, ...config },
    });

  const exitOp = interpolate(
    frame,
    [durationInFrames - 15, durationInFrames],
    [1, 0],
    { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' }
  );

  // Camera slow zoom in
  const cameraZoom = interpolate(frame, [0, durationInFrames], [1, 1.06], {
    extrapolateRight: 'clamp',
  });

  // Gradient background shift
  const gradAngle = interpolate(frame, [0, durationInFrames], [135, 155]);
  const bgGrad = `linear-gradient(${gradAngle}deg, #0a0a0a 0%, #1C1C1C 40%, #2D2D2D 100%)`;

  // Shape layers
  const shapes = Array.from({ length: 6 }, (_, i) => {
    const seed = seededRandom(i + 1);
    const size = 200 + seed * 400;
    const startX = seed * 1920;
    const startY = seededRandom(i + 10) * 1080;
    const drift = interpolate(frame, [0, durationInFrames], [0, 60 + seed * 40]);
    const rot = interpolate(frame, [0, durationInFrames], [0, 15 + seed * 20]);
    const breathe = 1 + Math.sin(frame * 0.02 + i) * 0.05;
    const opacity = 0.03 + seed * 0.05;
    const shapeType = i % 3;
    return { size, x: startX + drift, y: startY, rot, breathe, opacity, shapeType };
  });

  // Floating particles
  const particles = Array.from({ length: 25 }, (_, i) => {
    const seed = seededRandom(i + 50);
    const x = seed * 1920;
    const y = seededRandom(i + 60) * 1080;
    const floatY = Math.sin(frame * 0.03 + i * 0.5) * 15;
    const floatX = Math.cos(frame * 0.02 + i * 0.7) * 10;
    const size = 2 + seed * 4;
    const opacity = 0.1 + seed * 0.3;
    return { x: x + floatX, y: y + floatY, size, opacity };
  });

  // Motion texture — subtle grid lines
  const gridOpacity = interpolate(frame, [0, 30], [0, 0.06], {
    extrapolateRight: 'clamp',
  });

  // Content animations
  const titleSpring = sp(10);
  const subtitleSpring = sp(30);
  const icon1Spring = sp(45);
  const icon2Spring = sp(55);
  const icon3Spring = sp(65);
  const icon4Spring = sp(75);
  const lineSpring = sp(20);
  const characterSpring = sp(40, { damping: 14, stiffness: 100 });

  // Staggered icon data
  const icons = [
    { label: 'पेंशन', delay: icon1Spring, emoji: 'pension' },
    { label: 'मकान', delay: icon2Spring, emoji: 'house' },
    { label: 'बैंक बैलेंस', delay: icon3Spring, emoji: 'bank' },
    { label: 'इज़्ज़त', delay: icon4Spring, emoji: 'respect' },
  ];

  return (
    <AbsoluteFill
      style={{
        backgroundColor: '#1C1C1C',
        overflow: 'hidden',
        opacity: exitOp,
      }}
    >
      {/* Dynamic Gradient Background */}
      <AbsoluteFill
        style={{
          background: bgGrad,
          transform: `scale(${cameraZoom})`,
        }}
      />

      {/* Animated Shape Layers */}
      {shapes.map((s, i) => (
        <div
          key={`shape-${i}`}
          style={{
            position: 'absolute',
            left: s.x,
            top: s.y,
            width: s.size,
            height: s.size,
            transform: `rotate(${s.rot}deg) scale(${s.breathe})`,
            opacity: s.opacity,
            border:
              s.shapeType === 0
                ? '2px solid #FFC107'
                : s.shapeType === 1
                ? '2px solid #e9d700'
                : '2px solid #F5F5F5',
            borderRadius: s.shapeType === 0 ? '50%' : s.shapeType === 1 ? '0%' : '0%',
            clipPath:
              s.shapeType === 2
                ? 'polygon(50% 0%, 0% 100%, 100% 100%)'
                : undefined,
          }}
        />
      ))}

      {/* Motion Texture — Grid */}
      <AbsoluteFill style={{ opacity: gridOpacity }}>
        {Array.from({ length: 12 }, (_, i) => (
          <div
            key={`vline-${i}`}
            style={{
              position: 'absolute',
              left: i * 170,
              top: 0,
              width: 1,
              height: '100%',
              backgroundColor: '#F5F5F5',
              opacity: 0.3,
            }}
          />
        ))}
        {Array.from({ length: 7 }, (_, i) => (
          <div
            key={`hline-${i}`}
            style={{
              position: 'absolute',
              top: i * 165,
              left: 0,
              height: 1,
              width: '100%',
              backgroundColor: '#F5F5F5',
              opacity: 0.3,
            }}
          />
        ))}
      </AbsoluteFill>

      {/* Floating Particles */}
      {particles.map((p, i) => (
        <div
          key={`particle-${i}`}
          style={{
            position: 'absolute',
            left: p.x,
            top: p.y,
            width: p.size,
            height: p.size,
            borderRadius: '50%',
            backgroundColor: i % 3 === 0 ? '#FFC107' : i % 3 === 1 ? '#e9d700' : '#F5F5F5',
            opacity: p.opacity,
          }}
        />
      ))}

      {/* Main Content */}
      <AbsoluteFill
        style={{
          transform: `scale(${cameraZoom})`,
          display: 'flex',
          flexDirection: 'column',
          justifyContent: 'center',
          alignItems: 'center',
          padding: '0 120px',

      >
        {/* Accent Line */}
        <div
          style={{
            width: interpolate(lineSpring, [0, 1], [0, 120]),

            backgroundColor: '#e9d700',
            borderRadius: 3,
            marginBottom: 30,

          style={{
            fontFamily: "'Yantra One', sans-serif",
            fontSize: 92,
            color: '#F5F5F5',
            textAlign: 'center',
            lineHeight: 1.2,
            transform: `translateY(${interpolate(titleSpring, [0, 1], [60, 0])}px)`,
            opacity: titleSpring,
            textShadow: '0 4px 30px rgba(0,0,0,0.5)',
          }}
        >
          एक <span style={{ color: '#e9d700' }}>सेवा निवृत्त</span> अधिकारी
        </div>

        {/* Subtitle */}
        <div
          style={{
            fontFamily: "'Noto Sans Devanagari', sans-serif",
            fontSize: 48,

        {Array.from({ length: 6 }, (_, i) => {
          const ringSize = 200 + i * 180;
          const ringRot = interpolate(frame, [0, durationInFrames], [0, i % 2 === 0 ? 5 : -5]);
          return (
            <div
              key={`ring-${i}`}
              style={{
                position: 'absolute',
                left: '50%',
                top: '45%',
                width: ringSize,
                height: ringSize,
                borderRadius: '50%',
                border: '1px solid rgba(255, 193, 7, 0.12)',
                transform: `translate(-50%, -50%) rotate(${ringRot}deg)`,
              }}
            />
          );
        })}
      </AbsoluteFill>

      {/* Particles */}
      {particles.map((p, i) => (
        <div
          key={`p-${i}`}
          style={{
            position: 'absolute',
            left: p.x,
            top: p.y,
            width: p.size,
            height: p.size,
            borderRadius: '50%',
            backgroundColor: i % 2 === 0 ? '#FFC107' : '#e9d700',
            opacity: p.opacity,
          }}
        />
      ))}

      {/* Main Content */}
      <AbsoluteFill
        style={{
          display: 'flex',
          flexDirection: 'column',
          justifyContent: 'center',
          alignItems: 'center',
          padding: '0 150px',
        }}
      >
        {/* Accent line */}
        <div
          style={{
            width: interpolate(lineSpring, [0, 1], [0, 80]),
            height: 4,
            backgroundColor: '#FFC107',
            borderRadius: 2,
            marginBottom: 30,
          }}
        />

        {/* Chapter label */}
        <div
          style={{
            fontFamily: "'Inter', sans-serif",
            fontSize: 28,
            fontWeight: 700,
            color: '#FFC107',
            letterSpacing: 6,
            textTransform: 'uppercase',
            marginBottom: 15,
            transform: `translateY(${interpolate(chapterSpring, [0, 1], [30, 0])}px)`,
            opacity: chapterSpring,
          }}
        >
          CHAPTER 1 — RECAP
        </div>

        {/* Title */}
        <div
          style={{
            fontFamily: "'Yantra One', sans-serif",
            fontSize: 85,
            color: '#F5F5F5',
            textAlign: 'center',
            lineHeight: 1.2,
            marginBottom: 50,
            transform: `translateY(${interpolate(summarySpring, [0, 1], [50, 0])}px)`,
            opacity: summarySpring,
            textShadow: '0 0 60px rgba(255,193,7,0.1)',
          }}
        >
          आज़ादी का <span style={{ color: '#e9d700' }}>भ्रम</span>
        </div>

        {/* Summary bullets */}
        <div
          style={{
            display: 'flex',
            flexDirection: 'column',
            gap: 22,
            maxWidth: 1000,
          }}
        >
          {summaryPoints.map((point, idx) => {
            const bSpring = bulletSpring(idx);
            return (
              <div
                key={idx}
                style={{
                  display: 'flex',
                  alignItems: 'center',
                  gap: 20,
                  transform: `translateX(${interpolate(bSpring, [0, 1], [-60, 0])}px)`,
                  opacity: bSpring,
                }}
              >
                {/* Bullet diamond */}
                <div
                  style={{
                    width: 14,
                    height: 14,
                    backgroundColor: idx === 2 ? '#FFC107' : '#e9d700',
                    transform: 'rotate(45deg)',
                    flexShrink: 0,
                  }}
                />
                <div
                  style={{
                    fontFamily: "'Noto Sans Devanagari', sans-serif",
                    fontSize: 38,
                    color: idx === 2 ? '#FFC107' : '#F5F5F5',
                    fontWeight: idx === 2 ? 700 : 500,
                  }}
                >
                  {point}
                </div>
              </div>
            );
          })}
        </div>

        {/* Next chapter tease */}
        <div
          style={{
            marginTop: 60,
            display: 'flex',
            alignItems: 'center',
            gap: 15,
            transform: `translateY(${interpolate(nextSpring, [0, 1], [30, 0])}px)`,
            opacity: nextSpring,
          }}
        >
          <div
            style={{
              fontFamily: "'Noto Sans Devanagari', sans-serif",
              fontSize: 34,
              color: '#9E9E9E',
            }}
          >
            आगे — पहली चीज़ जो छोड़नी चाहिए
          </div>
          <svg
            width="28"
            height="28"
            viewBox="0 0 24 24"
            style={{
              transform: `translateX(${Math.sin(frame * 0.08) * 5}px)`,
            }}
          >
            <path d="M5 12h14M12 5l7 7-7 7" stroke="#FFC107" strokeWidth="2.5" fill="none" />
          </svg>
        </div>
      </AbsoluteFill>

      {/* Stick Figure — contemplative, center bottom */}
      <div
        style={{
          position: 'absolute',
          right: 120,
          bottom: 80,
          transform: `scale(${charSpring}) translateY(${Math.sin(frame * 0.05) * 2}px)`,
          opacity: charSpring * 0.5,
        }}
      >
        <svg width="90" height="150" viewBox="0 0 90 150">
          {/* Glow */}
          <circle cx="45" cy="35" r="35" fill="#FFC107" opacity="0.05" />
          {/* Head */}
          <circle
            cx="45"
            cy="35"
            r="18"
            stroke="#F5F5F5"
            strokeWidth="2"
            fill="none"
            transform={`rotate(${Math.sin(frame * 0.04) * 2}, 45, 35)`}
          />
          {/* Glasses */}
          <rect x="32" y="29" width="11" height="9" rx="2" stroke="#FFC107" strokeWidth="1.5" fill="none" />
          <rect x="47" y="29" width="11" height="9" rx="2" stroke="#FFC107" strokeWidth="1.5" fill="none" />
          <line x1="43" y1="33" x2="47" y2="33" stroke="#FFC107" strokeWidth="1.5" />
          {/* Neutral mouth */}
          <line x1="39" y1="44" x2="51" y2="44" stroke="#F5F5F5" strokeWidth="1.5" />
          {/* Body */}
          <line x1="45" y1="53" x2="45" y2="100" stroke="#F5F5F5" strokeWidth="2" />
          {/* Arms — one hand on chin (thinking) */}
          <line x1="45" y1="70" x2="25" y2="85" stroke="#F5F5F5" strokeWidth="2" />
          <line x1="45" y1="70" x2="65" y2="55" stroke="#F5F5F5" strokeWidth="2" />
          <line x1="65" y1="55" x2="55" y2="45" stroke="#F5F5F5" strokeWidth="2" />
          {/* Legs */}
          <line x1="45" y1="100" x2="30" y2="140" stroke="#F5F5F5" strokeWidth="2" />
          <line x1="45" y1="100" x2="60" y2="140" stroke="#F5F5F5" strokeWidth="2" />
        </svg>
      </div>

      {/* Vignette */}
      <AbsoluteFill
        style={{
          background:
            'radial-gradient(ellipse at center, transparent 50%, rgba(0,0,0,0.65) 100%)',
          pointerEvents: 'none',
        }}
      />

      {/* Watermark */}
      <div
        style={{
          position: 'absolute',
          bottom: 30,
          right: 45,
          fontFamily: "'Inter', sans-serif",
          fontSize: 22,
          fontWeight: 600,
          color: '#9E9E9E',
          opacity: interpolate(frame, [0, 30], [0, 0.3], {
            extrapolateRight: 'clamp',
          }),
        }}
      >
        ProBuffet HQ
      </div>
    </AbsoluteFill>
  );
};
