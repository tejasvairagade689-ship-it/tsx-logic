import React from 'react';
import {
  useCurrentFrame,
  useVideoConfig,
  interpolate,
  spring,
  AbsoluteFill,
} from 'remotion';

export const compositionConfig = {
  id: 'Scene1Ch1',
  durationInSeconds: 12,
  fps: 30,
  width: 1920,
  height: 1080,
};

const Scene1Ch1: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, cfg?: object) =>
    spring({
      frame: Math.max(0, frame - delay),
      fps,
      config: { damping: 14, mass: 0.5, stiffness: 100, ...cfg },
    });

  const exitOp = interpolate(
    frame,
    [durationInFrames - 15, durationInFrames],
    [1, 0],
    { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' }
  );

  // Slow background circle drift
  const circleDrift = interpolate(frame, [0, durationInFrames], [0, 30], {
    extrapolateRight: 'clamp',
  });

  // Text micro animation - subtle float
  const textFloat = Math.sin(frame * 0.06) * 4;

  // Word by word reveal
  const word1Op = interpolate(frame, [5, 15], [0, 1], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' });
  const word2Op = interpolate(frame, [15, 25], [0, 1], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' });
  const word3Op = interpolate(frame, [25, 35], [0, 1], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' });

  const lineScale = sp(0);
  const iconScale = sp(20);
  const subScale = sp(35);

  return (
    <AbsoluteFill
      style={{
        background: 'linear-gradient(135deg, #1C1C1C 0%, #2D2D2D 50%, #1C1C1C 100%)',
        overflow: 'hidden',
        opacity: exitOp,
        fontFamily: 'Inter, Poppins',
      }}
    >
      {/* === SHAPE LAYER: Large drifting circle === */}
      <svg
        style={{ position: 'absolute', top: -100, right: -100, opacity: 0.05 }}
        width="700"
        height="700"
      >
        <circle
          cx={350 + circleDrift}
          cy={350 - circleDrift * 0.5}
          r="300"
          fill="none"
          stroke="#e9d700"
          strokeWidth="2"
        />
        <circle
          cx={350 + circleDrift}
          cy={350 - circleDrift * 0.5}
          r="240"
          fill="none"
          stroke="#e9d700"
          strokeWidth="1"
        />
      </svg>

      {/* === SHAPE LAYER: Bottom left triangle === */}
      <svg
        style={{ position: 'absolute', bottom: -50, left: -50, opacity: 0.04 }}
        width="400"
        height="400"
      >
        <polygon
          points={`${200 + circleDrift * 0.3},20 20,380 380,380`}
          fill="#F5F5F5"
          stroke="#F5F5F5"
          strokeWidth="1"
        />
      </svg>

      {/* === TEXTURE: Grid lines === */}
      <svg
        style={{ position: 'absolute', top: 0, left: 0, opacity: 0.03 }}
        width="1920"
        height="1080"
      >
        {Array.from({ length: 20 }).map((_, i) => (
          <line
            key={`v${i}`}
            x1={i * 100}
            y1="0"
            x2={i * 100}
            y2="1080"
            stroke="#F5F5F5"
            strokeWidth="1"
          />
        ))}
        {Array.from({ length: 12 }).map((_, i) => (
          <line
            key={`h${i}`}
            x1="0"
            y1={i * 90}
            x2="1920"
            y2={i * 90}
            stroke="#F5F5F5"
            strokeWidth="1"
          />
        ))}
      </svg>

      {/* === FLOATING PARTICLES === */}
      {[...Array(8)].map((_, i) => {
        const px = (i * 237 + 120) % 1800;
        const py = (i * 133 + 80) % 900;
        const pFloat = Math.sin(frame * 0.04 + i * 1.2) * 8;
        const pOp = interpolate(frame, [i * 5, i * 5 + 20], [0, 0.4], {
          extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
        });
        return (
          <div
            key={i}
            style={{
              position: 'absolute',
              left: px,
              top: py + pFloat,
              width: 4,
              height: 4,
              borderRadius: '50%',
              backgroundColor: '#e9d700',
              opacity: pOp,
            }}
          />
        );
      })}

      {/* === MAIN CONTENT === */}
      <div
        style={{
          position: 'absolute',
          top: '50%',
          left: '50%',
          transform: `translate(-50%, -50%) translateY(${textFloat}px)`,
          textAlign: 'center',
          width: 1400,
        }}
      >
        {/* Eye / Vision SVG Icon */}
        <div
          style={{
            opacity: iconScale,
            transform: `scale(${iconScale})`,
            marginBottom: 32,
            display: 'flex',
            justifyContent: 'center',
          }}
        >
          <svg width="72" height="72" viewBox="0 0 72 72" fill="none">
            <ellipse cx="36" cy="36" rx="32" ry="20" stroke="#e9d700" strokeWidth="3" fill="none" />
            <circle cx="36" cy="36" r="10" fill="#e9d700" opacity="0.9" />

        </div>

        {/* Line reveal bar */}
        <div
          style={{
            height: 3,
            backgroundColor: '#e9d700',
            width: `${lineScale * 200}px`,
            margin: '0 auto 36px',
            borderRadius: 2,
          }}
        />

        {/* Main headline - word by word */}
        <div
          style={{
            fontFamily: 'Noto Sans Devanagari',
            fontSize: 72,
            fontWeight: 700,
            color: '#F5F5F5',
            lineHeight: 1.4,
            letterSpacing: '-0.5px',

        >
          <span style={{ opacity: word1Op, display: 'inline-block', marginRight: 12 }}>
            अक्सर जब हम
          </span>
          <span style={{ opacity: word2Op, display: 'inline-block', color: '#e9d700', marginRight: 12 }}>
            किसी इन्सान को
          </span>
          <span style={{ opacity: word3Op, display: 'inline-block' }}>
            जिन्दगी में...
          </span>
        </div>

        {/* Sub text */}
        <div
          style={{
            opacity: subScale,
            transform: `translateY(${(1 - subScale) * 20}px)`,
            fontFamily: 'Noto Sans Devanagari',
            fontSize: 44,
            fontWeight: 400,
            color: '#9E9E9E',
            marginTop: 24,
          }}
        >
          आर्थिक रूप से पूरी तरह आजाद और कामयाब
        </div>
      </div>

      {/* === STICK FIGURE — Buffett === */}
      <div
        style={{
          position: 'absolute',
          bottom: 80,
          left: 100,
          opacity: sp(40),
          transform: `scale(${sp(40)}) translateY(${Math.sin(frame * 0.05) * 3}px)`,
        }}
      >
        <svg width="80" height="140" viewBox="0 0 80 140" fill="none">
          {/* Golden glow */}
          <circle cx="40" cy="28" r="26" fill="#e9d700" opacity="0.12" />
          {/* Head */}
          <circle cx="40" cy="28" r="18" stroke="#F5F5F5" strokeWidth="2.5" fill="none" />
