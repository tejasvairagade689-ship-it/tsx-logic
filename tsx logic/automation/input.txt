  return x - Math.floor(x);
};

const Scene03Entrepot: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // Screen shake on box impacts
  const shakeAt = (impact: number) => {
    const d = frame - impact;
    if (d < 0 || d > 12) return 0;
    return Math.sin(d * 2.2) * (12 - d) * 1.4;
  };
  const shake = shakeAt(60) + shakeAt(85) + shakeAt(110);

        <polygon points="46,12 46,24 34,12" fill="#504099" />
      </svg>
    );
  };

  return (
    <AbsoluteFill style={{ backgroundColor: '#504099', overflow: 'hidden', opacity: exitOp }}>
      {/* Gradient background */}
      <AbsoluteFill style={{
        background: 'radial-gradient(circle at 50% 60%, #6B5BC7 0%, #504099 45%, #2E2470 100%)',
      }} />

      {/* Animated shape layers — orbit rings */}
      {[orbitR, orbitR + 130].map((r, i) => (
        <div key={i} style={{
          position: 'absolute', left: '50%', top: '56%',
          width: r * 2, height: r * 1.35,
          borderRadius: '50%',
          border: '2px dashed rgba(245,245,245,0.12)',
          transform: `translate(-50%,-50%) rotate(${frame * 0.02 * (i ? -1 : 1)}deg)`,
        }} />
      ))}

      {/* Motion texture — halftone corner blocks (punk) */}
      <div style={{
        position: 'absolute', left: 0, top: 0, width: 480, height: 300,
        backgroundImage: 'radial-gradient(rgba(0,0,0,0.2) 3px, transparent 3px)',
        backgroundSize: '26px 26px',
        clipPath: 'polygon(0 0, 100% 0, 0 100%)',

        backgroundSize: '26px 26px',
        clipPath: 'polygon(100% 100%, 0 100%, 100% 0)',

      {/* Floating particles */}
      {Array.from({ length: 14 }).map((_, i) => (
        <div key={i} style={{
          position: 'absolute',
          left: seededRandom(i + 14) * 1920,
          top: (seededRandom(i + 34) * 1080 + frame * 0.3) % 1120 - 40,
          width: 4, height: 4, borderRadius: '50%',
          background: i % 3 === 0 ? '#FFC107' : '#F5F5F5', opacity: 0.16,
        }} />
      ))}

      {/* Main content */}
      <AbsoluteFill style={{ alignItems: 'center' }}>

        {/* Title */}
        <div style={{
          position: 'absolute', top: 70,
          transform: `scale(${titleS}) rotate(-1.5deg)`,
        }}>
          <span style={{
            fontFamily: 'Anton, sans-serif', fontSize: 88, color: '#F5F5F5',
            background: '#000000', padding: '10px 44px', letterSpacing: 3,
          }}>
            LE VRAI COÛT COMMENCE <span style={{ color: '#e9d700' }}>APRÈS</span>
          </span>
        </div>

        {/* Core: the purchase */}
        <div style={{
          position: 'absolute', left: '50%', top: '56%',
          transform: `translate(-50%,-50%) scale(${coreS * (1 + Math.sin(frame * 0.06) * 0.02)})`,
        }}>
          <div style={{
            width: 300, height: 300, borderRadius: '50%',
            background: '#FFC107', border: '7px solid #000000',
            display: 'flex', flexDirection: 'column', alignItems: 'center', justifyContent: 'center',
            boxShadow: '0 0 70px rgba(255,193,7,0.35)',
          }}>
            <div style={{ fontFamily: 'Anton, sans-serif', fontSize: 55, color: '#000000' }}>
              L'ACHAT
            </div>
            <div style={{ fontFamily: 'Space Mono, monospace', fontWeight: 700, fontSize: 34, color: '#000000' }}>
              (jour 1)
            </div>
          </div>
        </div>

        {/* Orbiting satellite cost cards */}
        {sats.map((s2, i) => {
          const s = sp(s2.d);
          const angle = (i / 4) * Math.PI * 2 + frame * orbitSpeed - Math.PI / 2;
          const x = 960 + Math.cos(angle) * orbitR * 1.35;
          const y = 605 + Math.sin(angle) * orbitR * 0.62;
          return (
            <div key={i} style={{
              position: 'absolute', left: x, top: y,
              transform: `translate(-50%,-50%) scale(${s}) rotate(${Math.sin(frame * 0.04 + i * 2) * 3}deg)`,

            }}>
              <div style={{
                background: '#F5F5F5', borderRadius: 14, padding: '18px 24px',
                border: '4px solid #000000', boxShadow: '7px 8px 0 rgba(0,0,0,0.5)',
                display: 'flex', alignItems: 'center', gap: 16, minWidth: 300,
              }}>
                <Icon name={s2.icon} />
                <span style={{
                  fontFamily: 'Bricolage Grotesque, sans-serif', fontWeight: 800,

                }}>
                  {s2.label}
                </span>
              </div>
            </div>
          );
        })}

        {/* Connector pulses from core */}
        <svg width="1920" height="1080" style={{ position: 'absolute', pointerEvents: 'none' }}>
          {sats.map((_, i) => {
            const angle = (i / 4) * Math.PI * 2 + frame * orbitSpeed - Math.PI / 2;
            const x = 960 + Math.cos(angle) * orbitR * 1.35;
            const y = 605 + Math.sin(angle) * orbitR * 0.62;
            const on = frame > 140 + i * 50 ? 1 : 0;

        {/* Final line */}
        <div style={{
          position: 'absolute', bottom: 60,
          transform: `scale(${finalS}) rotate(1deg)`,
        }}>
          <span style={{
            fontFamily: 'Bricolage Grotesque, sans-serif', fontWeight: 800, fontSize: 46,
            color: '#000000', background: '#e9d700', padding: '10px 36px',
            boxShadow: '7px 7px 0 #000000',
          }}>
            L'étiquette n'était que l'acompte.
          </span>
        </div>
      </AbsoluteFill>

      {/* Vignette overlay */}
      <AbsoluteFill style={{
        background: 'radial-gradient(circle, transparent 55%, rgba(0,0,0,0.45) 100%)',
        pointerEvents: 'none',
      }} />

      {/* Watermark */}
      <div style={{
        position: 'absolute', bottom: 30, right: 45,
        fontFamily: 'Montserrat, sans-serif', fontSize: 22, fontWeight: 600,
        color: '#9E9E9E', opacity: 0.3,
      }}>
        Matures Finance
      </div>
    </AbsoluteFill>
  );
};
