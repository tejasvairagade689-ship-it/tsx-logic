import { useCurrentFrame, useVideoConfig, interpolate, Easing, spring, AbsoluteFill } from 'remotion';

export const compositionConfig = {
  id: 'Scene01_HookOpener',
  durationInSeconds: 9,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene01_HookOpener: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // Hero "65" reveal
  const sixtyFiveScale = sp(15, { damping: 9, stiffness: 140 });
  const sixtyFiveGlow = interpolate(frame, [30, 90], [0, 1], { extrapolateRight: 'clamp' });
  
  // Title reveals
  const titleY = interpolate(sp(45), [0, 1], [60, 0]);
  const titleOp = interpolate(sp(45), [0, 1], [0, 1]);
  
  const subY = interpolate(sp(75), [0, 1], [40, 0]);
  const subOp = interpolate(sp(75), [0, 1], [0, 1]);

  // Camera zoom
  const camZoom = interpolate(frame, [0, durationInFrames], [1, 1.08]);

  // Breathing for hero
  const breathe = 1 + Math.sin(frame * 0.04) * 0.015;

  // Pulse ring around 65
  const ringScale = 1 + (Math.sin(frame * 0.08) * 0.05);
  const ringOp = 0.4 + Math.sin(frame * 0.08) * 0.15;

  // Floating shapes
  const shapes = Array.from({ length: 8 }, (_, i) => i);
  
  // Particles
  const particles = Array.from({ length: 25 }, (_, i) => i);

  // Underline sweep
  const underlineW = interpolate(sp(60), [0, 1], [0, 600]);

  // Alert tag
  const tagScale = sp(20, { damping: 8 });

  return (
    <AbsoluteFill style={{ backgroundColor: '#2C3947', overflow: 'hidden', opacity: exitOp }}>
      {/* Dynamic gradient background */}
      <AbsoluteFill style={{
        background: 'radial-gradient(ellipse at 50% 40%, rgba(250,185,91,0.18) 0%, rgba(44,57,71,0) 55%), linear-gradient(135deg, #2C3947 0%, #1f2a35 100%)',
      }} />

      {/* Camera container */}
      <AbsoluteFill style={{ transform: `scale(${camZoom})`, transformOrigin: '50% 50%' }}>
        
        {/* Shape layer — large drifting geometric forms */}
        {shapes.map((i) => {
          const r = seededRandom(i + 1);
          const r2 = seededRandom(i + 50);
          const size = 200 + r * 280;
          const x = r * 1920;
          const y = r2 * 1080;
          const drift = Math.sin((frame + i * 30) * 0.01) * 20;
          const rot = (frame * 0.1) + i * 45;
          const isCircle = i % 3 === 0;
          const isTri = i % 3 === 1;
          return (
            <div key={`sh-${i}`} style={{
              position: 'absolute',
              left: x - size / 2,
              top: y - size / 2 + drift,
              width: size,
              height: size,
              transform: `rotate(${rot}deg)`,
              opacity: 0.05,
            }}>
              {isCircle && (
                <div style={{
                  width: '100%', height: '100%',
                  borderRadius: '50%',
                  border: '2px solid #E8EDF2',
                }} />
              )}
              {isTri && (
                <div style={{
                  width: 0, height: 0,
                  borderLeft: `${size/2}px solid transparent`,
                  borderRight: `${size/2}px solid transparent`,
                  borderBottom: `${size}px solid rgba(250,185,91,0.6)`,
                }} />
              )}
              {!isCircle && !isTri && (
                <div style={{
                  width: '100%', height: '100%',
                  border: '2px solid #FAB95B',
                }} />
              )}
            </div>
          );
        })}

        {/* Motion texture — grid lines */}
        <svg width="1920" height="1080" style={{ position: 'absolute', opacity: 0.06 }}>
          {Array.from({ length: 20 }).map((_, i) => (
            <line key={`v-${i}`} x1={i * 96} y1={0} x2={i * 96} y2={1080} stroke="#E8EDF2" strokeWidth="1" />
          ))}
          {Array.from({ length: 12 }).map((_, i) => (
            <line key={`h-${i}`} x1={0} y1={i * 90} x2={1920} y2={i * 90} stroke="#E8EDF2" strokeWidth="1" />

        </svg>

        {/* Floating particles */}
        {particles.map((i) => {
          const r = seededRandom(i + 100);
          const r2 = seededRandom(i + 200);
          const px = (r * 1920 + frame * (0.3 + r * 0.6)) % 1920;
          const py = r2 * 1080 + Math.sin((frame + i * 10) * 0.03) * 30;
          const size = 2 + r * 4;
          return (
            <div key={`p-${i}`} style={{
              position: 'absolute',

            <path d="M12 7v6" stroke="#FAB95B" strokeWidth="2.5" strokeLinecap="round" />

          </svg>
          <div style={{
            fontFamily: 'Inter, Poppins, sans-serif',
            fontSize: 30,
            fontWeight: 700,
            letterSpacing: 3,
            color: '#FAB95B',
            textTransform: 'uppercase',
          }}>
            URGENT FOR SENIORS
          </div>
        </div>

        {/* HERO "65" */}
        <div style={{
          position: 'absolute',
          top: '50%',
          left: '50%',
          transform: `translate(-50%, -50%) scale(${sixtyFiveScale * breathe})`,
        }}>
          {/* Pulse ring */}
          <div style={{
            position: 'absolute',
            top: '50%',
            left: '50%',
            width: 520,
            height: 520,
            transform: `translate(-50%, -50%) scale(${ringScale})`,
            borderRadius: '50%',
            border: '3px solid #FAB95B',
            opacity: ringOp,
          }} />

            position: 'absolute',
