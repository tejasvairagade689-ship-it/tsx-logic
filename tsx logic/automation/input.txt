  const camY = Math.cos(frame * 0.006) * 9;

  const capIn = sp(150);
  const stampIn = sp(300, { damping: 9, stiffness: 70 });

  const display =
    frame < 190

      ? '-5,000'
      : frame < 470
      ? '0.00'
      : 'OFF';

    easing: Easing.out(Easing.cubic),

  const crossB = interpolate(frame, [124, 154], [crossLen, 0], {
    extrapolateLeft: 'clamp',
    extrapolateRight: 'clamp',
    easing: Easing.out(Easing.cubic),
  });

            })`,
          }}
        />
        <div
          style={{
            position: 'absolute',
            left: -180,
            top: 520,
            width: 760,
            height: 760,
            backgroundColor: RED,
            opacity: 0.05,
            transform: `rotate(${-frame * 0.09}deg)`,

                  fill={INK}
                  opacity={0.09 + d * 0.05}
                />

            position: 'absolute',
            left: s.x,
            top: s.y,
            width: s.w,
            height: s.h,

            opacity: s.o,

              letterSpacing: 5,
              color: INK,
              fontWeight: 700,
