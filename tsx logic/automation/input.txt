import React from 'react';
// This is your composition file. It contains the React component
// that defines your video.
import React from 'react';
import { useCurrentFrame, useVideoConfig, interpolate, Easing, spring, AbsoluteFill } from 'remotion';

export const compositionConfig = {
  id: 'Scene01Hook',
  durationInSeconds: 10,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene01Hook: React.FC = () => {
  // useCurrentFrame() returns the current frame number.
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  const camZoom = interpolate(frame, [0, durationInFrames], [1, 1.06], { extrapolateRight: 'clamp' });
  const word1 = sp(10);
  const word2 = sp(28);
  const ringScale = sp(45, { damping: 8 });
  const breathe = Math.sin(frame * 0.05) * 6;

  return (
    <AbsoluteFill style={{ backgroundColor: '#14FFEC', overflow: 'hidden', opacity: exitOp }}>
      {/* Gradient overlay */}
      <AbsoluteFill style={{ background: 'radial-gradient(circle at 30% 20%, rgba(255,255,255,0.35), transparent 60%)' }} />

      {/* Animated shape layers */}
      {[0, 1, 2].map((i) => {
        const s = seededRandom(i + 1);
        return (
          <div key={i} style={{
            position: 'absolute',
            left: `${s * 70}%`,
            top: `${seededRandom(i + 5) * 70}%`,
            width: 360 + s * 200,
            height: 360 + s * 200,
            borderRadius: i === 1 ? '20%' : '50%',
            border: `3px solid rgba(0,0,0,0.07)`,
            transform: `rotate(${frame * (0.15 + s * 0.1)}deg) scale(${1 + Math.sin(frame * 0.02 + i) * 0.05})`,
          }} />
        );
      })}

      {/* Motion texture grid */}
      <AbsoluteFill style={{
        backgroundImage: 'linear-gradient(rgba(0,0,0,0.04) 1px, transparent 1px), linear-gradient(90deg, rgba(0,0,0,0.04) 1px, transparent 1px)',
        backgroundSize: '80px 80px',
        transform: `translateY(${(frame * 0.4) % 80}px)`,
      }} />

      {/* Floating particles */}
      {Array.from({ length: 16 }).map((_, i) => {
        const r = seededRandom(i + 20);
        return (
          <div key={i} style={{
            position: 'absolute',
            left: `${r * 100}%`,
            top: `${(seededRandom(i + 30) * 100 + frame * 0.3) % 100}%`,
            width: 6 + r * 8,
            height: 6 + r * 8,
            borderRadius: '50%',
            backgroundColor: 'rgba(0,0,0,0.15)',
          }} />
        );
      })}

      {/* Main content with spring animations */}
      <AbsoluteFill style={{
        justifyContent: 'center', alignItems: 'center', flexDirection: 'column',
        transform: `scale(${camZoom})`,
      }}>
        <div style={{
          fontFamily: 'Inter, Poppins',
          fontSize: 76,
          fontWeight: 800,
          color: '#212121',
          letterSpacing: 2,
          opacity: word1,
          transform: `translateY(${(1 - word1) * 40}px)`,
        }}>
          MARRIED &amp; APPROACHING
        </div>
        <div style={{
          fontFamily: 'Inter, Poppins',
          fontSize: 170,
          fontWeight: 900,
          color: '#000000',
          lineHeight: 0.95,
          marginTop: 10,
          opacity: word2,
          transform: `translateY(${(1 - word2) * 60}px) translateY(${breathe}px)`,
        }}>
          RETIREMENT
        </div>

        {/* Reveal ring around */}
        <div style={{
          marginTop: 50,
          width: 220 * ringScale,
          height: 220 * ringScale,
          borderRadius: '50%',
          border: '10px solid #000000',
          position: 'relative',
          opacity: ringScale,
        }}>
          <svg width="100%" height="100%" viewBox="0 0 100 100" style={{ transform: `rotate(${frame * 2}deg)` }}>
            <path d="M50 20 L60 45 L40 45 Z" fill="#212121" />
          </svg>
        </div>
        <div style={{
          fontFamily: 'JetBrains Mono',
          fontSize: 34,
          fontWeight: 700,
          color: '#212121',
          marginTop: 30,
          opacity: sp(60),
        }}>
          STAY TILL THE VERY END
        </div>
      </AbsoluteFill>

      {/* Vignette overlay */}
      <AbsoluteFill style={{ boxShadow: 'inset 0 0 350px rgba(0,0,0,0.35)', pointerEvents: 'none' }} />

      {/* Watermark */}
      <div style={{
        position: 'absolute', bottom: 30, right: 45,
        fontFamily: 'Inter', fontSize: 22, fontWeight: 600,
        color: '#212121', opacity: interpolate(frame, [0, 40], [0, 0.5], { extrapolateRight: 'clamp' }),
      }}>
        Senior Fund USA
      </div>
    </AbsoluteFill>
  );
};

export default Scene01Hook;
