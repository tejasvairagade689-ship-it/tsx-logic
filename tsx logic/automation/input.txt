	const sp = (delay: number, config?: object) =>
		spring({
			frame: Math.max(0, frame - delay),
			fps,
			config: { damping: 12, mass: 0.4, stiffness: 120, ...config },
		});

	const exitOp = interpolate(
		frame,
		[durationInFrames - 15, durationInFrames],
		[1, 0],
		{ extrapolateLeft: 'clamp', extrapolateRight: 'clamp' }
	);

	// Camera slow zoom out
	const cameraZoom = interpolate(frame, [0, durationInFrames], [1.05, 1], {
		extrapolateLeft: 'clamp',
		extrapolateRight: 'clamp',
		easing: Easing.out(Easing.cubic),
	});

	return (
		<AbsoluteFill
			style={{
				backgroundColor: '#1C1C1C',
				overflow: 'hidden',
				opacity: exitOp,
			}}
		>
			{/* Dynamic Gradient - Conclusive dark red */}
			<div
				style={{
					position: 'absolute',
					width: '100%',
					height: '100%',
					background: `
						radial-gradient(circle at 50% 50%, rgba(255,0,0,0.1) 0%, transparent 35%),
						radial-gradient(circle at ${55 + Math.sin(frame * 0.02) * 6}% ${48 + Math.cos(frame * 0.015) * 5}%, rgba(233,215,0,0.04) 0%, transparent 30%),
						linear-gradient(180deg, #2D2D2D 0%, #1C1C1C 40%, #0d0d0d 100%)
					`,
				}}
			/>

			{/* Shape Layers - Converging lines to center */}
			<svg
				style={{
					position: 'absolute',
					width: '100%',
					height: '100%',
					opacity: 0.04,
				}}
			>
				{Array.from({ length: 12 }).map((_, i) => {
					const angle = (i / 12) * Math.PI * 2;
					return (
						<line
							key={`cv-${i}`}
							x1={960 + Math.cos(angle) * 900}
							y1={540 + Math.sin(angle) * 900}
							x2={960}
							y2={540}
							stroke="#ff0000"
							strokeWidth="0.8"
						/>
					);
				})}
				{/* Center target circle */}
				<circle cx="960" cy="540" r={80 + Math.sin(frame * 0.08) * 10} fill="none" stroke="#ff0000" strokeWidth="1.5" opacity="0.3" />
				<circle cx="960" cy="540" r={40 + Math.cos(frame * 0.1) * 5} fill="none" stroke="#e9d700" strokeWidth="1" opacity="0.4" />
			</svg>

			{/* Motion Texture - Pulse rings from center */}
			{Array.from({ length: 3 }).map((_, i) => {
				const pulsePhase = (frame * 0.8 + i * 30) % 90;
				const pulseScale = interpolate(pulsePhase, [0, 90], [0, 1.5]);
				const pulseOpacity = interpolate(pulsePhase, [0, 90], [0.08, 0]);
				return (
					<div
						key={`pulse-${i}`}
						style={{
							position: 'absolute',
							width: 200,
							height: 200,
							borderRadius: '50%',
							border: '2px solid #ff0000',
							left: '50%',
							top: '50%',
							transform: `translate(-50%, -50%) scale(${pulseScale})`,
							opacity: pulseOpacity,
						}}
					/>
				);
			})}

			{/* Floating Particles */}
			{Array.from({ length: 20 }).map((_, i) => {
				const s1 = seededRandom(i + 6000);
				const s2 = seededRandom(i + 6100);
				const size = 2 + s1 * 3;
				return (
					<div
						key={`p-${i}`}
						style={{
							position: 'absolute',
							width: size,
							height: size,
							borderRadius: '50%',
							backgroundColor: i % 2 === 0 ? '#ff0000' : '#e9d700',
							opacity: 0.1 + s1 * 0.15,
							left: s1 * 1920 + Math.sin(frame * 0.03 + i) * 20,
							top: s2 * 1080 + Math.cos(frame * 0.025 + i) * 18,
						}}
					/>
				);
			})}

			{/* Main Content */}
			<AbsoluteFill
				style={{
					display: 'flex',
					flexDirection: 'column',
					alignItems: 'center',
					justifyContent: 'center',
					transform: `scale(${cameraZoom})`,
				}}
			>
				{/* Counter visual - 5 slots filling up */}
				<div
					style={{
						display: 'flex',
						gap: 18,
						marginBottom: 40,
						transform: `translateY(${interpolate(sp(0), [0, 1], [-50, 0])}px)`,
						opacity: sp(0),
					}}
				>
					{Array.from({ length: 5 }).map((_, i) => {
						const fillDelay = 5 + i * 6;
						const filled = sp(fillDelay);
						return (
							<div
								key={`slot-${i}`}
								style={{
									width: 80,
									height: 100,
									borderRadius: 10,
									border: `2px solid ${filled > 0.5 ? '#ff0000' : 'rgba(245,245,245,0.2)'}`,
									background: `rgba(255,0,0,${interpolate(filled, [0, 1], [0, 0.2])})`,
									display: 'flex',
									flexDirection: 'column',
									alignItems: 'center',
									justifyContent: 'center',
									transition: 'all 0.3s',
								}}
							>
								<div
									style={{
										fontFamily: 'JetBrains Mono',
										fontSize: 32,
										fontWeight: 900,
										color: filled > 0.5 ? '#ff0000' : '#9E9E9E',
									}}
								>
									{i + 1}
								</div>
								<div
									style={{
										fontFamily: 'Inter',
										fontSize: 12,
										color: filled > 0.5 ? '#ff0000' : '#9E9E9E',
										fontWeight: 600,
									}}
								>
									{i < 3 ? 'CARD' : 'UPI'}
								</div>
								{filled > 0.5 && (
									<div
										style={{
											fontFamily: 'Inter',
											fontSize: 16,
											color: '#ff0000',
											fontWeight: 900,
											marginTop: 4,
										}}
									>
										✓
									</div>
								)}
							</div>
						);
					})}
				</div>

				{/* Hero Statement */}
				<div style={{ textAlign: 'center' }}>
					<div
						style={{
							fontFamily: 'Yantra One',
							fontSize: 100,
							fontWeight: 900,
							color: '#F5F5F5',
							lineHeight: 1.1,
							transform: `translateY(${interpolate(sp(15), [0, 1], [40, 0])}px)`,
							opacity: sp(15),
						}}
					>
						मुफ्त लिमिट
					</div>
					<div
						style={{
							fontFamily: 'Yantra One',
							fontSize: 130,
							fontWeight: 900,
							color: '#ff0000',
							lineHeight: 1,
							transform: `scale(${interpolate(sp(25, { damping: 8 }), [0, 1], [1.3, 1])})`,
							opacity: sp(25),
							textShadow: '0 0 60px rgba(255,0,0,0.4)',
						}}
					>
						में गिनेगा!
					</div>
				</div>

				{/* Equation bar */}
				<div
					style={{
						marginTop: 35,
						display: 'flex',
						alignItems: 'center',
						gap: 15,
						transform: `scaleX(${sp(35)})`,
						opacity: sp(35),
					}}
				>
					<div
						style={{
							background: '#2D2D2D',
							border: '1.5px solid #e9d700',
							borderRadius: 8,
							padding: '10px 25px',
							fontFamily: 'JetBrains Mono',
							fontSize: 26,
							color: '#e9d700',
							fontWeight: 700,
						}}
					>
						3 CARD + 2 UPI
					</div>
					<div
						style={{
							fontFamily: 'JetBrains Mono',
							fontSize: 36,
							color: '#e9d700',
							fontWeight: 900,
						}}
					>
						=
					</div>
					<div
						style={{
							background: 'rgba(255,0,0,0.15)',
							border: '1.5px solid #ff0000',
							borderRadius: 8,
							padding: '10px 25px',
							fontFamily: 'JetBrains Mono',
							fontSize: 26,
							color: '#ff0000',
							fontWeight: 700,
						}}
					>
						5/5 USED ⚠️
					</div>
				</div>
			</AbsoluteFill>

			{/* Stick Figure - pointing at counter */}
			<div
				style={{
					position: 'absolute',
					bottom: 40,
					left: 100,
					transform: `scale(${sp(20)}) translateY(${Math.sin(frame * 0.05) * 2}px)`,
					opacity: sp(20),
				}}
			>
				<svg width="90" height="160" viewBox="0 0 90 160">
					<circle cx="45" cy="32" r="18" fill="none" stroke="#F5F5F5" strokeWidth="2.5" />
					<circle cx="38" cy="30" r="6.5" fill="none" stroke="#FFC107" strokeWidth="2" />
					<circle cx="54" cy="30" r="6.5" fill="none" stroke="#FFC107" strokeWidth="2" />
					<line x1="44" y1="30" x2="48" y2="30" stroke="#FFC107" strokeWidth="1.5" />
					{/* Alert expression */}
					<circle cx="38" cy="26" r="2" fill="#F5F5F5" />
					<circle cx="54" cy="26" r="2" fill="#F5F5F5" />
					<ellipse cx="46" cy="42" rx="6" ry="4" fill="none" stroke="#F5F5F5" strokeWidth="1.5" />
					<line x1="45" y1="50" x2="45" y2="100" stroke="#F5F5F5" strokeWidth="2.5" />
					{/* Pointing up */}
					<line x1="45" y1="70" x2="20" y2="88" stroke="#F5F5F5" strokeWidth="2.5" strokeLinecap="round" />
					<line x1="45" y1="70" x2={75 + Math.cos(frame * 0.06) * 3} y2={50 + Math.sin(frame * 0.05) * 2} stroke="#F5F5F5" strokeWidth="2.5" strokeLinecap="round" />
					<line x1="45" y1="100" x2="30" y2="145" stroke="#F5F5F5" strokeWidth="2.5" strokeLinecap="round" />
					<line x1="45" y1="100" x2="60" y2="145" stroke="#F5F5F5" strokeWidth="2.5" strokeLinecap="round" />
				</svg>
			</div>

			{/* Vignette */}
			<div
				style={{
					position: 'absolute',
					width: '100%',
					height: '100%',
					background: 'radial-gradient(ellipse at center, transparent 45%, rgba(0,0,0,0.65) 100%)',
					pointerEvents: 'none',
				}}
			/>

			{/* Watermark */}
			<div
				style={{
					position: 'absolute',
					bottom: 30,
					right: 45,
					fontFamily: 'Inter',
					fontSize: 22,
					fontWeight: 600,
					color: '#9E9E9E',
					opacity: interpolate(frame, [0, 30], [0, 0.3], {
						extrapolateLeft: 'clamp',
						extrapolateRight: 'clamp',
					}),
				}}
			>
				ProBuffet HQ
			</div>
		</AbsoluteFill>
	);
};

export default Scene10_LimitMeinGina;
