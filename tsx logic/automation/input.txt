  fps: 30,
  width: 1920,

  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);

    { n: 'PPF', d: 130, ic: 'shield' },

    return (<svg width="66" height="66" viewBox="0 0 100 100"><path d="M22 44a28 28 0 0 1 48-14" {...s} /><path d="M78 56a28 28 0 0 1-48 14" {...s} /><path d="M62 16l12 14-16 8M38 84L26 70l16-8" {...s} /></svg>);

      {/* Dynamic gradient background */}
      <AbsoluteFill style={{ background: 'linear-gradient(0deg, #6B5029 0%, #C2A56D 55%, #EDD6A6 100%)' }} />

      {/* Animated shape layers */}
      <AbsoluteFill>
        {[0, 1, 2, 3, 4].map((i) => {
          const r = seededRandom(i + 2500);
          return (
            <div key={i} style={{
              position: 'absolute',
              left: `${r * 100 - 12}%`, top: `${seededRandom(i + 41) * 100 - 12}%`,

              background: i % 2 ? '#F4F2F2' : '#000000', opacity: 0.05,

      </AbsoluteFill>

      {/* Motion texture — rising ticks */}
