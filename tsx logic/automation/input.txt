import React from 'react';
import {
  useCurrentFrame,
  useVideoConfig,
  interpolate,
  spring,
  AbsoluteFill,
  Easing,
} from 'remotion';

export const compositionConfig = {
  id: 'Scene1_NeejiSawaal',
  durationInSeconds: 6,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene1_NeejiSawaal: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({
      frame: Math.max(0, frame - delay),
      fps,
      config: { damping: 12, mass: 0.4, stiffness: 120, ...config },
    });

  const exitOp = interpolate(
    frame,
    [durationInFrames - 15, durationInFrames],
    [1, 0],
    { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' }
  );

  // Camera shake on "chubhe" moment
  const shakeX = frame > 60 && frame < 90 ? Math.sin(frame * 1.5) * 3 : 0;
  const shakeY = frame > 60 && frame < 90 ? Math.cos(frame * 2) * 2 : 0;

  // Zoom in slowly
  const zoom = interpolate(frame, [0, durationInFrames], [1, 1.05], {
    extrapolateRight: 'clamp',
  });

  return (
    <AbsoluteFill
      style={{
        backgroundColor: '#1C1C1C',
        overflow: 'hidden',
        opacity: exitOp,
        transform: `scale(${zoom}) translate(${shakeX}px, ${shakeY}px)`,
      }}
    >
      {/* Gradient Background */}
      <AbsoluteFill
        style={{
          background:
            'radial-gradient(ellipse at 30% 50%, #2D2D2D 0%, #1C1C1C 60%, #0a0a0a 100%)',
        }}
      />

      {/* Animated Shape Layers */}
      {[...Array(5)].map((_, i) => {
        const seed = seededRandom(i + 1);
        const size = 200 + seed * 400;
        const startX = seed * 1920;
        const startY = seededRandom(i + 10) * 1080;
        const drift = Math.sin(frame * 0.008 + i) * 40;
        const rotation = frame * 0.1 * (i % 2 === 0 ? 1 : -1);
        const shapeOpacity = 0.03 + seed * 0.05;
        const scaleBreath = 1 + Math.sin(frame * 0.02 + i) * 0.05;

        return (
          <div
            key={`shape-${i}`}
            style={{
              position: 'absolute',
              left: startX + drift,
              top: startY + Math.cos(frame * 0.006 + i) * 30,
              width: size,
              height: size,
              borderRadius: i % 3 === 0 ? '50%' : i % 3 === 1 ? '10%' : '0%',
              border: `2px solid rgba(233, 215, 0, ${shapeOpacity})`,
              transform: `rotate(${rotation}deg) scale(${scaleBreath})`,
              opacity: shapeOpacity,
            }}
          />
        );
      })}

      {/* Motion Texture - Vertical Lines */}
      {[...Array(12)].map((_, i) => (
        <div
          key={`line-${i}`}
          style={{
            position: 'absolute',
            left: 160 * i + Math.sin(frame * 0.01 + i) * 10,
            top: 0,
            width: 1,
            height: '100%',
            background: `linear-gradient(180deg, transparent 0%, rgba(245, 245, 245, 0.03) 50%, transparent 100%)`,
            opacity: interpolate(frame, [0, 30], [0, 0.5], {
              extrapolateRight: 'clamp',
            }),
          }}
        />
      ))}

      {/* Floating Particles */}
      {[...Array(25)].map((_, i) => {
        const pSeed = seededRandom(i + 50);
        const pSize = 2 + pSeed * 4;
        const pX = pSeed * 1920;
        const pY =
          (seededRandom(i + 60) * 1080 + frame * (0.5 + pSeed * 1.5)) % 1080;
        const pOpacity = 0.1 + pSeed * 0.3;

        return (
          <div
            key={`particle-${i}`}
            style={{
              position: 'absolute',
              left: pX + Math.sin(frame * 0.02 + i) * 15,
              top: pY,
              width: pSize,
              height: pSize,
              borderRadius: '50%',
              backgroundColor: i % 3 === 0 ? '#e9d700' : '#F5F5F5',
              opacity: pOpacity,
            }}
          />
        );
      })}

      {/* Main Content */}
      <AbsoluteFill
        style={{
          display: 'flex',
          flexDirection: 'column',
          alignItems: 'center',
          justifyContent: 'center',
        }}
      >
        {/* Question Mark - Big dramatic entrance */}
        <div
          style={{
            position: 'absolute',
            top: 80,
            right: 120,
            fontSize: 400,
            fontFamily: 'Yantra One',
            color: 'rgba(233, 215, 0, 0.06)',
            transform: `scale(${sp(0)}) rotate(${interpolate(frame, [0, 180], [15, -5], { extrapolateRight: 'clamp' })}deg)`,
            lineHeight: 1,
          }}
        >
          ?
        </div>

        {/* Main Text */}
        <div
          style={{
            transform: `translateY(${interpolate(sp(10), [0, 1], [80, 0])}px)`,
            opacity: sp(10),
            textAlign: 'center',
            padding: '0 120px',
          }}
        >
          <div
            style={{
              fontFamily: 'Yantra One',
              fontSize: 120,
              color: '#F5F5F5',
              lineHeight: 1.2,
              textShadow: '0 0 60px rgba(233, 215, 0, 0.2)',
            }}
          >
            एक सवाल
          </div>
        </div>

        <div
          style={{
            transform: `translateY(${interpolate(sp(25), [0, 1], [60, 0])}px)`,
            opacity: sp(25),
            textAlign: 'center',
            marginTop: 20,
          }}
        >
          <div
            style={{
              fontFamily: 'Noto Sans Devanagari',
              fontSize: 70,
              color: '#e9d700',
              fontWeight: 700,
              letterSpacing: 2,
            }}
          >
            बड़ा निजी है...
          </div>
        </div>

        {/* Underline animation */}
        <div
          style={{
            width: interpolate(sp(40), [0, 1], [0, 500]),
            height: 4,
            backgroundColor: '#e9d700',
            marginTop: 30,
            borderRadius: 2,
            boxShadow: '0 0 20px rgba(233, 215, 0, 0.5)',
          }}
        />

        {/* "Chubhe bhi" subtitle with delay */}
        <div
          style={{
            transform: `translateY(${interpolate(sp(55), [0, 1], [40, 0])}px)`,
            opacity: sp(55),
            marginTop: 35,
          }}
        >
          <div
            style={{
              fontFamily: 'Noto Sans Devanagari',
              fontSize: 48,
              color: '#9E9E9E',
              fontStyle: 'italic',
            }}
          >
            शायद चुभे भी...
          </div>
        </div>
      </AbsoluteFill>

      {/* Stick Figure - enters late, thinking pose */}
      <div
        style={{
          position: 'absolute',
          bottom: 80,
          left: 100,
          transform: `translateX(${interpolate(sp(50), [0, 1], [-150, 0])}px)`,
          opacity: sp(50),
        }}
      >
        <svg width="120" height="200" viewBox="0 0 120 200">
          {/* Glow */}
          <circle cx="60" cy="40" r="50" fill="rgba(233, 215, 0, 0.08)" />
          {/* Head */}
          <circle
            cx="60"
            cy="40"
            r="22"
            fill="none"
