```tsx
import React from 'react';
import { useCurrentFrame, useVideoConfig, interpolate, spring, AbsoluteFill } from 'remotion';

export const compositionConfig = {
  id: 'Scene02',
  durationInSeconds: 5,
  fps: 30,
  width: 1920,
  height: 1080,
};

const sr = (s: number) => Math.sin(s * 9999) * 10000 - Math.floor(Math.sin(s * 9999) * 10000);

const Scene02: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();
  const sp = (d: number, c?: any) => spring({ frame: Math.max(0, frame - d), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...c } });
  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' });

  return (
    <AbsoluteFill style={{ backgroundColor: '#1C1C1C', overflow: 'hidden', opacity: exitOp, fontFamily: "'Yantra One', 'Noto Sans Devanagari', sans-serif" }}>
      {/* Dynamic gradient background */}
      <AbsoluteFill style={{ background: 'linear-gradient(45deg, #2D2D2D 0%, #1C1C1C 100%)' }} />
      
      {/* Animated shape layers */}
      <div style={{ position: 'absolute', top: '50%', right: '10%', width: 500, height: 500, borderRadius: '30%', background: 'rgba(255, 193, 7, 0.04)', transform: `rotate(${frame * 0.05}deg) scale(${1 + Math.sin(frame * 0.01) * 0.03})` }} />

      {/* Motion texture */}
      <AbsoluteFill style={{ backgroundImage: 'linear-gradient(0deg, transparent 24%, rgba(158, 158, 158, .1) 25%, rgba(158, 158, 158, .1) 26%, transparent 27%, transparent 74%, rgba(158, 158, 158, .1) 75%, rgba(158, 158, 158, .1) 76%, transparent 77%, transparent)', backgroundSize: '50px 50px' }} />

      {/* Floating particles */}
      {[...Array(6)].map((_, i) => (
        <div key={i} style={{ position: 'absolute', left: `${10 + i * 12}%`, top: `${20 + sr(i+1) * 50}%`, width: 6, height: 6, borderRadius: '50%', backgroundColor: '#F5F5F5', opacity: 0.4, transform: `translateY(${Math.sin(frame * 0.04 + i) * 15}px)` }} />
      ))}

      {/* Main content with spring animations */}
      <AbsoluteFill style={{ display: 'flex', flexDirection: 'column', alignItems: 'flex-start', justifyContent: 'center', paddingLeft: '150px' }}>
        <h1 style={{ fontSize: '140px', fontWeight: 'bold', color: '#e9d700', transform: `translateX(${sp(15, { stiffness: 80 })}px)`, opacity: sp(15) }}>
          Senior Fund USA
        </h1>
        <h2 style={{ fontSize: '90px', fontWeight: '600', color: '#F5F5F5', marginTop: '20px', transform: `translateX(${sp(30, { stiffness: 80 })}px)`, opacity: sp(30) }}>
          mein aapka swagat hai.
        </h2>
      </AbsoluteFill>

      {/* Stick figure character with micro-movements */}
      <div style={{ position: 'absolute', bottom: 120, left: 150, transform: `translateY(${Math.sin(frame * 0.05) * 2}px)`, opacity: sp(50) }}>
        <svg width="120" height="200" viewBox="0 0 120 200">
          <circle cx="60" cy="60" r="50" fill="rgba(255, 193, 7, 0.2)" />
          <circle cx="60" cy="60" r="30" fill="none" stroke="#F5F5F5" strokeWidth="4" />
          <circle cx="50" cy="55" r="8" fill="none" stroke="#FFC107" strokeWidth="2" />
          <circle cx="70" cy="55" r="8" fill="none" stroke="#FFC107" strokeWidth="2" />
          <line x1="58" y1="55" x2="62" y2="55" stroke="#FFC107" strokeWidth="2" />
          <line x1="60" y1="90" x2="60" y2="150" stroke="#F5F5F5" strokeWidth="4" />
          <line x1="60" y1="100" x2="30" y2="80" stroke="#F5F5F5" strokeWidth="4" />
          <line x1="60" y1="100" x2="90" y2="120" stroke="#F5F5F5" strokeWidth="4" />
          <line x1="60" y1="150" x2="30" y2="190" stroke="#F5F5F5" strokeWidth="4" />
          <line x1="60" y1="150" x2="90" y2="190" stroke="#F5F5F5" strokeWidth="4" />
        </svg>
      </div>

      {/* Vignette overlay */}
      <AbsoluteFill style={{ background: 'radial-gradient(circle, transparent 40%, rgba(0,0,0,0.7) 100%)', pointerEvents: 'none' }} />

      {/* Watermark "Senior Fund USA" bottom-right */}
      <div style={{ position: 'absolute', bottom: 30, right: 45, fontFamily: "'Inter', sans-serif", fontSize: '22px', fontWeight: 600, color: '#9E9E9E', opacity: interpolate(frame, [0, 30, durationInFrames - 30, durationInFrames], [0, 0.3, 0.3, 0], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' }) }}>
        Senior Fund USA
      </div>
    </AbsoluteFill>
  );
};

export default Scene02;
```
