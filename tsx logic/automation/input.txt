import React from 'react';

import { useCurrentFrame, useVideoConfig, interpolate, Easing, spring, AbsoluteFill } from 'remotion';

export const compositionConfig = {
  id: 'Scene01',
  durationInSeconds: 15,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene01: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  const q1 = sp(20);
  const q2 = sp(45);
  const line = interpolate(frame, [50, 90], [0, 1], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' });
  const camZoom = interpolate(frame, [0, durationInFrames], [1.05, 1.12], { extrapolateRight: 'clamp' });

  return (
    <AbsoluteFill style={{ backgroundColor: '#D2DAFF', overflow: 'hidden', opacity: exitOp, fontFamily: 'Inter, Poppins, sans-serif' }}>
      {/* Gradient layer */}
      <AbsoluteFill style={{ background: 'radial-gradient(circle at 30% 25%, rgba(255,255,255,0.55), rgba(210,218,255,0) 60%)' }} />

      {/* Shape layers — slow drifting geometric forms */}
      {[0, 1, 2, 3].map((i) => {
        const r = seededRandom(i + 1);
        const drift = Math.sin(frame * 0.012 + i) * 30;
        const rot = (frame * 0.1 + i * 40) % 360;
        const size = 360 + r * 260;
        return (
          <div
            key={i}
            style={{
              position: 'absolute',
              left: `${10 + r * 70}%`,
              top: `${5 + seededRandom(i + 9) * 70}%`,
              width: size,
              height: size,
              borderRadius: i % 2 === 0 ? '50%' : '24px',
              border: '2px solid #1E2022',
              opacity: 0.05,
              transform: `translateY(${drift}px) rotate(${rot}deg) scale(${1 + Math.sin(frame * 0.02 + i) * 0.04})`,
            }}
          />
        );
      })}

      {/* Motion texture — vertical grid lines */}
      {Array.from({ length: 16 }).map((_, i) => (
        <div
          key={`g${i}`}
          style={{
            position: 'absolute',
            left: `${(i / 16) * 100}%`,
            top: 0,
            width: 1,
            height: '100%',
            backgroundColor: '#1E2022',
            opacity: 0.04,
            transform: `translateX(${Math.sin(frame * 0.02 + i) * 6}px)`,
          }}
        />
      ))}

      {/* Floating particles */}
      {Array.from({ length: 18 }).map((_, i) => {
        const r = seededRandom(i + 50);
        const y = (seededRandom(i + 70) * 100 - ((frame * (0.2 + r * 0.3)) % 110) + 110) % 110;
        return (
          <div
            key={`p${i}`}
            style={{
              position: 'absolute',
              left: `${r * 100}%`,
              top: `${y}%`,
              width: 6 + r * 6,
              height: 6 + r * 6,
              borderRadius: '50%',
              backgroundColor: '#1E2022',
              opacity: 0.12,
            }}
          />
        );
      })}

      {/* Main content */}
      <AbsoluteFill style={{ alignItems: 'center', justifyContent: 'center', transform: `scale(${camZoom})` }}>
        <div style={{ width: '78%', textAlign: 'center' }}>
          <div
            style={{
              fontFamily: "'Yantra One', sans-serif",
              fontSize: 102,
              fontWeight: 700,
              color: '#000000',
              lineHeight: 1.15,
              opacity: q1,
              transform: `translateY(${(1 - q1) * 60}px)`,
            }}
          >
            भाई, एक बात बताओ
          </div>

          <div
            style={{
              height: 8,
              width: `${line * 240}px`,
              backgroundColor: '#1E2022',
              margin: '34px auto',
              borderRadius: 4,
            }}
          />

          <div
            style={{
              fontFamily: "'Noto Sans Devanagari', sans-serif",
              fontSize: 66,
              fontWeight: 600,
              color: '#1E2022',
              lineHeight: 1.3,
              opacity: q2,
              transform: `translateY(${(1 - q2) * 40}px)`,
            }}
          >
            पापा या दादा ने जब घर खरीदा था —<br />उनकी सैलरी कितनी थी, घर कितने का?
          </div>
        </div>
      </AbsoluteFill>

      {/* Vignette overlay */}
      <AbsoluteFill style={{ boxShadow: 'inset 0 0 320px rgba(30,32,34,0.35)', pointerEvents: 'none' }} />

      {/* Watermark */}
      <div style={{ position: 'absolute', bottom: 30, right: 45, fontFamily: 'Inter, sans-serif', fontSize: 22, fontWeight: 600, color: '#1E2022', opacity: 0.3 }}>
        ProBuffet HQ
      </div>
    </AbsoluteFill>
  );
};

export default Scene01;
