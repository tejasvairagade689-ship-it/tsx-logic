import React from 'react';
import {
  useCurrentFrame,
  useVideoConfig,
  interpolate,
  spring,
  AbsoluteFill,
  Easing,
} from 'remotion';

export const compositionConfig = {
  id: 'Scene1_TakleefIntro',
  durationInSeconds: 6,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene1_TakleefIntro: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({
      frame: Math.max(0, frame - delay),
      fps,
      config: { damping: 12, mass: 0.4, stiffness: 120, ...config },
    });

  const exitOp = interpolate(
    frame,
    [durationInFrames - 15, durationInFrames],
    [1, 0],
    { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' }
  );

  // Camera zoom
  const cameraZoom = interpolate(frame, [0, durationInFrames], [1.05, 1], {
    extrapolateLeft: 'clamp',
    extrapolateRight: 'clamp',
  });

  // Screen shake on impact
  const shakeX = frame > 10 && frame < 25 ? Math.sin(frame * 1.5) * 3 : 0;
  const shakeY = frame > 10 && frame < 25 ? Math.cos(frame * 2) * 2 : 0;

  return (
    <AbsoluteFill
      style={{
        backgroundColor: '#1C1C1C',
        overflow: 'hidden',
        opacity: exitOp,
        transform: `scale(${cameraZoom}) translate(${shakeX}px, ${shakeY}px)`,
      }}
    >
      {/* Dynamic Gradient Background */}
      <AbsoluteFill
        style={{
          background: `radial-gradient(ellipse at ${30 + Math.sin(frame * 0.02) * 10}% ${40 + Math.cos(frame * 0.015) * 10}%, #2D0A0A 0%, #1C1C1C 50%, #0D0D0D 100%)`,
        }}
      />

      {/* Animated Shape Layers */}
      {[...Array(6)].map((_, i) => {
        const seed = seededRandom(i + 1);
        const size = 100 + seed * 300;
        const startX = seed * 1920;
        const startY = seededRandom(i + 10) * 1080;
        const rotation = frame * (0.1 + seed * 0.2);
        const breathe = 1 + Math.sin(frame * 0.03 + i) * 0.05;
        const shapes = ['circle', 'rect', 'polygon'];
        const shapeType = shapes[i % 3];

        return (
          <div
            key={`shape-${i}`}
            style={{
              position: 'absolute',
              left: startX + Math.sin(frame * 0.008 + i) * 40,
              top: startY + Math.cos(frame * 0.006 + i) * 30,
              transform: `rotate(${rotation}deg) scale(${breathe})`,
              opacity: 0.04 + seed * 0.04,
            }}
          >
            <svg width={size} height={size} viewBox={`0 0 ${size} ${size}`}>
              {shapeType === 'circle' && (
                <circle
                  cx={size / 2}
                  cy={size / 2}
                  r={size / 2.2}
                  fill="none"
                  stroke="#ff0000"
                  strokeWidth="1.5"
                />
              )}
              {shapeType === 'rect' && (
                <rect
                  x={size * 0.1}
                  y={size * 0.1}
                  width={size * 0.8}
                  height={size * 0.8}
                  fill="none"
                  stroke="#e9d700"
                  strokeWidth="1.5"
                />
              )}
              {shapeType === 'polygon' && (
                <polygon
                  points={`${size / 2},${size * 0.1} ${size * 0.9},${size * 0.9} ${size * 0.1},${size * 0.9}`}
                  fill="none"
                  stroke="#FFC107"
                  strokeWidth="1.5"
                />
              )}
            </svg>
          </div>
        );
      })}

      {/* Motion Texture - Diagonal Lines */}
      <AbsoluteFill style={{ opacity: 0.03 }}>
        <svg width="1920" height="1080">
          {[...Array(30)].map((_, i) => (
            <line
              key={`line-${i}`}
              x1={i * 70 + Math.sin(frame * 0.02) * 20}
              y1={0}
              x2={i * 70 + 200 + Math.sin(frame * 0.02) * 20}
              y2={1080}
              stroke="#F5F5F5"
              strokeWidth="0.5"
            />
          ))}
        </svg>
      </AbsoluteFill>

      {/* Floating Particles */}
      {[...Array(25)].map((_, i) => {
        const px = seededRandom(i + 50) * 1920;
        const py = seededRandom(i + 100) * 1080;
        const pSize = 2 + seededRandom(i + 150) * 4;
        const speed = 0.5 + seededRandom(i + 200) * 1.5;

        return (
          <div
            key={`particle-${i}`}
            style={{
              position: 'absolute',
              left: px + Math.sin(frame * 0.02 * speed + i) * 30,
              top: py + Math.cos(frame * 0.015 * speed + i) * 25 - frame * 0.3,
              width: pSize,
              height: pSize,
              borderRadius: '50%',
              backgroundColor: i % 3 === 0 ? '#ff0000' : i % 3 === 1 ? '#e9d700' : '#FFC107',
              opacity: 0.15 + Math.sin(frame * 0.05 + i) * 0.1,
            }}
          />
        );
      })}

      {/* Main Content */}
      <AbsoluteFill
        style={{
          display: 'flex',
          flexDirection: 'column',
          alignItems: 'center',
          justifyContent: 'center',
          padding: '0 120px',
        }}
      >
        {/* Big Impact Word */}
        <div
          style={{
            fontFamily: 'Yantra One',
            fontSize: 220,
            color: '#ff0000',
            fontWeight: 900,
            transform: `scale(${sp(5)}) translateY(${interpolate(sp(5), [0, 1], [80, 0])}px)`,
            opacity: sp(5),
            textShadow: '0 0 80px rgba(255,0,0,0.4), 0 4px 20px rgba(0,0,0,0.8)',
            letterSpacing: '-4px',
            lineHeight: 1,
          }}
        >
          तकलिफ
        </div>

        {/* Subtitle */}
        <div
          style={{
            fontFamily: 'Noto Sans Devanagari',
            fontSize: 52,
            color: '#F5F5F5',
            fontWeight: 700,
            marginTop: 30,
            transform: `translateY(${interpolate(sp(20), [0, 1], [40, 0])}px)`,
            opacity: sp(20),
            textAlign: 'center',
            lineHeight: 1.4,
          }}
        >
          पैसा बनाने की सबसे बड़ी तकलिफ
        </div>

        {/* Crossed out words */}
        <div
          style={{
            display: 'flex',
            gap: 60,
            marginTop: 40,
            opacity: sp(40),
            transform: `translateY(${interpolate(sp(40), [0, 1], [30, 0])}px)`,
          }}
        >
          {['सही स्टॉक', 'म्यूचुअल फंड'].map((text, idx) => (
            <div
              key={idx}
              style={{
                position: 'relative',
                fontFamily: 'Noto Sans Devanagari',
                fontSize: 44,
                color: '#9E9E9E',
                fontWeight: 600,
                opacity: sp(45 + idx * 15),
              }}
            >
              {text}
              <div
                style={{
                  position: 'absolute',
                  top: '50%',
                  left: -10,
                  width: `${interpolate(sp(55 + idx * 15), [0, 1], [0, 110])}%`,
                  height: 4,
                  backgroundColor: '#ff0000',
                  transform: 'rotate(-3deg)',
                }}
              />
            </div>
          ))}
        </div>
      </AbsoluteFill>

      {/* Stick Figure - Thinking Pose */}
      <div
        style={{
          position: 'absolute',
          right: 120,
          bottom: 100,
          opacity: sp(50),
          transform: `translateY(${interpolate(sp(50), [0, 1], [60, 0]) + Math.sin(frame * 0.05) * 2}px)`,
        }}
      >
        <svg width="140" height="200" viewBox="0 0 140 200">
          {/* Golden Glow */}
          <circle cx="70" cy="45" r="40" fill="#FFC107" opacity="0.15" />
          {/* Head */}
          <circle
            cx="70"
            cy="45"
            r="22"
            fill="none"
            stroke="#F5F5F5"
            strokeWidth="3"
            transform={`rotate(${Math.sin(frame * 0.04) * 3}, 70, 45)`}

          {/* Glasses */}
          <rect x="55" y="38" width="12" height="10" rx="2" fill="none" stroke="#FFC107" strokeWidth="2" />
          <rect x="73" y="38" width="12" height="10" rx="2" fill="none" stroke="#FFC107" strokeWidth="2" />
