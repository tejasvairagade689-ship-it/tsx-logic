import React from 'react';

import { useCurrentFrame, useVideoConfig, interpolate, Easing, spring, AbsoluteFill } from 'remotion';

export const compositionConfig = {
  id: 'Scene01PictureSomething',
  durationInSeconds: 16,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene01PictureSomething: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // Slow cinematic zoom on whole composition
  const camScale = interpolate(frame, [0, durationInFrames], [1.04, 1.12], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // Word 1 entrance
  const w1 = sp(8);
  const w2 = sp(28);
  const w3 = sp(48);

  // Underline sweep
  const underline = interpolate(frame, [60, 95], [0, 1], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp', easing: Easing.out(Easing.cubic),
  });

  // Sub caption fade
  const subOp = interpolate(frame, [110, 140], [0, 1], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });
  const subY = interpolate(frame, [110, 140], [30, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp', easing: Easing.out(Easing.cubic),
  });

  // Eye icon micro-pulse (picture / look)
  const eyePulse = 1 + Math.sin(frame * 0.06) * 0.04;
  const eyeOp = interpolate(frame, [30, 55], [0, 1], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // Breathing shapes
  const breath = 1 + Math.sin(frame * 0.025) * 0.06;
  const drift = Math.sin(frame * 0.02) * 40;

  return (
    <AbsoluteFill style={{ backgroundColor: '#1E1F22', overflow: 'hidden', opacity: exitOp }}>
      {/* Dynamic gradient background */}
      <AbsoluteFill
        style={{
          background: 'radial-gradient(circle at 30% 25%, #2A2B2F 0%, #1E1F22 55%, #161719 100%)',
        }}
      />

      {/* Animated transparent shape layers (depth) */}
      <AbsoluteFill style={{ transform: `scale(${camScale})` }}>
        <div
          style={{
            position: 'absolute', top: -120 + drift, left: -80,
            width: 720, height: 720, borderRadius: '50%',
            border: '2px solid #D4A373', opacity: 0.06,
            transform: `scale(${breath}) rotate(${frame * 0.15}deg)`,
          }}
        />
        <div
          style={{
            position: 'absolute', bottom: -200, right: -120 - drift,
            width: 640, height: 640,
            border: '2px solid #A3B19B', opacity: 0.05,
            transform: `rotate(${frame * -0.1}deg) scale(${1 / breath})`,
          }}
        />
        <div
          style={{
            position: 'absolute', top: '40%', left: '60%',
            width: 0, height: 0,
            borderLeft: '180px solid transparent',
            borderRight: '180px solid transparent',
            borderBottom: '320px solid #D4A373', opacity: 0.04,
            transform: `rotate(${frame * 0.08}deg)`,
          }}
        />
      </AbsoluteFill>

      {/* Motion texture — thin grid lines */}
      <AbsoluteFill style={{ opacity: 0.05 }}>
        {[...Array(8)].map((_, i) => (
          <div
            key={`h-${i}`}
            style={{
              position: 'absolute', left: 0, right: 0,
              top: `${(i + 1) * 11}%`, height: 1,
              backgroundColor: '#C5BAAF',
              transform: `translateX(${Math.sin((frame + i * 20) * 0.02) * 25}px)`,
            }}
          />
        ))}
      </AbsoluteFill>

      {/* Floating particles */}
      <AbsoluteFill>
        {[...Array(22)].map((_, i) => {
          const px = seededRandom(i + 1) * 1920;
          const py = seededRandom(i + 50) * 1080;
          const size = 3 + seededRandom(i + 9) * 5;
          const floatY = Math.sin((frame + i * 30) * 0.03) * 22;
          const pOp = 0.2 + seededRandom(i + 3) * 0.4;
          return (
            <div
              key={`p-${i}`}
              style={{
                position: 'absolute',
                left: px, top: py + floatY,
                width: size, height: size, borderRadius: '50%',
                backgroundColor: i % 3 === 0 ? '#D4A373' : '#C5BAAF',
                opacity: pOp,
              }}
            />
          );
        })}
      </AbsoluteFill>

      {/* MAIN CONTENT */}
      <AbsoluteFill
        style={{
          transform: `scale(${camScale})`,
          display: 'flex', flexDirection: 'column',
          justifyContent: 'center', alignItems: 'center',
        }}
      >
        {/* Eye icon (picture something / look) */}
        <svg
          width="150" height="150" viewBox="0 0 120 120"
          style={{
            opacity: eyeOp,
            transform: `scale(${eyePulse * (0.6 + w1 * 0.4)})`,
            marginBottom: 30,
            filter: 'drop-shadow(0 0 18px rgba(212,163,115,0.35))',
          }}
        >
          <path
            d="M10 60 Q60 18 110 60 Q60 102 10 60 Z"
            fill="none" stroke="#D4A373" strokeWidth="5" strokeLinejoin="round"
          />
          <circle cx="60" cy="60" r="20" fill="none" stroke="#F7EBE1" strokeWidth="5" />
          <circle cx="60" cy="60" r="9" fill="#D4A373" />
        </svg>

        {/* Word-by-word headline */}
        <div style={{ textAlign: 'center', lineHeight: 1.05 }}>
          <div
            style={{
              fontFamily: 'Fraunces, serif', fontWeight: 600,
              fontSize: 200, color: '#F7EBE1',
              opacity: w1,
              transform: `translateY(${(1 - w1) * 60}px)`,
            }}
          >
            PICTURE
          </div>
          <div
            style={{
              fontFamily: 'Fraunces, serif', fontWeight: 600,
              fontSize: 200, color: '#D4A373',
              opacity: w2,
              transform: `translateY(${(1 - w2) * 60}px)`,
              position: 'relative',
            }}
          >
            SOMETHING
            {/* Underline sweep */}
            <div
              style={{
                position: 'absolute', bottom: -10, left: '50%',
                transform: 'translateX(-50%)',
                width: `${underline * 70}%`, height: 8,
                backgroundColor: '#A3B19B', borderRadius: 4,
              }}
            />
          </div>
        </div>

        {/* Sub caption */}
        <div
          style={{
            marginTop: 60, maxWidth: 1100, textAlign: 'center',
            fontFamily: 'DM Sans, sans-serif', fontWeight: 500,
            fontSize: 54, color: '#C5BAAF',
            opacity: subOp,
            transform: `translateY(${subY}px)`,
          }}
        >
          The number isn't just a statistic.
        </div>
      </AbsoluteFill>

      {/* Vignette overlay */}
      <AbsoluteFill
        style={{
          background: 'radial-gradient(circle at 50% 50%, transparent 55%, rgba(0,0,0,0.55) 100%)',
          pointerEvents: 'none',
        }}
      />

      {/* Watermark */}
      <div
