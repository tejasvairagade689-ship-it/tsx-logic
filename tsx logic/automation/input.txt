import React from 'react';

import { useCurrentFrame, useVideoConfig, interpolate, Easing, spring, AbsoluteFill } from 'remotion';

export const compositionConfig = {
  id: 'Scene01CrossedLine',
  durationInSeconds: 14,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene01CrossedLine: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // Camera: slow push-in
  const camScale = interpolate(frame, [0, durationInFrames], [1, 1.08], { easing: Easing.inOut(Easing.ease) });

  // The line sweeps across, then the word DEFAULT stamps down
  const lineW = interpolate(frame, [20, 60], [0, 1400], { extrapolateRight: 'clamp', easing: Easing.out(Easing.cubic) });
  const stamp = sp(150, { damping: 9, stiffness: 160 });
  const shake = frame >= 150 && frame <= 162 ? Math.sin(frame * 3) * (162 - frame) * 0.8 : 0;

  const strikeDelinquency = interpolate(frame, [235, 260], [0, 100], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' });
  const strikeBehind = interpolate(frame, [265, 290], [0, 100], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' });

  return (
    <AbsoluteFill style={{ backgroundColor: '#76ABAE', overflow: 'hidden', opacity: exitOp }}>
      {/* Gradient */}
      <AbsoluteFill style={{ background: 'linear-gradient(135deg, #76ABAE 0%, #5E9497 55%, #4E8285 100%)' }} />

      {/* Shape Layers */}
      <div style={{ position: 'absolute', width: 900, height: 900, borderRadius: '50%', border: '3px solid #EEEEEE', opacity: 0.06, left: -250, top: -300, transform: `rotate(${frame * 0.05}deg) scale(${1 + Math.sin(frame / 60) * 0.03})` }} />
      <div style={{ position: 'absolute', width: 600, height: 600, background: '#3F4F44', opacity: 0.05, right: -180, bottom: -220, transform: `rotate(${25 + frame * 0.03}deg)` }} />

      {/* Texture: diagonal lines */}
      <AbsoluteFill style={{ backgroundImage: 'repeating-linear-gradient(45deg, #EEEEEE 0px, #EEEEEE 1px, transparent 1px, transparent 90px)', opacity: 0.05 }} />

      {/* Particles */}
      {Array.from({ length: 16 }).map((_, i) => (
        <div key={i} style={{
          position: 'absolute',
          left: seededRandom(i + 1) * 1920,
          top: ((seededRandom(i + 50) * 1080) + frame * (0.2 + seededRandom(i) * 0.5)) % 1080,
          width: 3 + seededRandom(i + 9) * 4, height: 3 + seededRandom(i + 9) * 4,
          borderRadius: '50%', background: '#EEEEEE', opacity: 0.12 + seededRandom(i + 3) * 0.1,
        }} />
      ))}

      {/* Content */}
      <AbsoluteFill style={{ justifyContent: 'center', alignItems: 'center', transform: `scale(${camScale}) translateX(${shake}px)` }}>
        <div style={{ opacity: sp(5), transform: `translateY(${(1 - sp(5)) * 40}px)`, fontFamily: 'Inter, Poppins', fontSize: 44, fontWeight: 600, color: '#EEEEEE', letterSpacing: 10, textTransform: 'uppercase' }}>
          Somewhere in America
        </div>

        {/* The line being crossed */}
        <div style={{ width: lineW, height: 8, background: '#3F4F44', margin: '46px 0', borderRadius: 4 }} />

        <div style={{ transform: `scale(${stamp})`, fontFamily: 'Mont Heavy, Inter', fontSize: 190, fontWeight: 900, color: '#3F4F44', letterSpacing: 6, lineHeight: 1 }}>
          DEFAULT

          FULL FEDERAL DEFAULT
        </div>

import React from 'react';

  id: 'Scene10NoticeWave',
  durationInSeconds: 14,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene10NoticeWave: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });
