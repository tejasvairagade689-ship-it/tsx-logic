import React from 'react';
import { useCurrentFrame, useVideoConfig, interpolate, Easing, spring, AbsoluteFill } from 'remotion';

export const compositionConfig = {
  id: 'PaycheckVsAffordability',
  durationInSeconds: 21,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const PaycheckVsAffordability: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // Paycheck line rises, affordability line falls
  const payGrow = interpolate(frame, [10, 120], [0, 1], { extrapolateRight: 'clamp', easing: Easing.out(Easing.cubic) });
  const affDrop = interpolate(frame, [40, 150], [0, 1], { extrapolateRight: 'clamp', easing: Easing.out(Easing.cubic) });
  const camZoom = interpolate(frame, [0, durationInFrames], [1.05, 1.12]);

  return (
    <AbsoluteFill style={{ backgroundColor: '#C46A2E', overflow: 'hidden', opacity: exitOp }}>
      {/* Dynamic gradient background */}
      <AbsoluteFill style={{ background: 'radial-gradient(circle at 30% 20%, #FF9B51 0%, #C46A2E 55%, #8F4A1E 100%)' }} />

      {/* Animated shape layers (slow drifting geometry) */}
      <AbsoluteFill style={{ transform: `scale(${camZoom})` }}>
        <div style={{
          position: 'absolute', width: 700, height: 700, borderRadius: '50%',
          border: '2px solid #FFF449', opacity: 0.05, top: -180, left: -120,
          transform: `rotate(${frame * 0.15}deg) scale(${1 + Math.sin(frame / 40) * 0.05})`,
        }} />
        <div style={{
          position: 'absolute', width: 520, height: 520, background: '#25343F', opacity: 0.05,
          bottom: -160, right: -100, transform: `rotate(${-frame * 0.1}deg)`,
        }} />
      </AbsoluteFill>

      {/* Motion texture — faint vertical grid */}
      <AbsoluteFill style={{ opacity: 0.06 }}>
        {Array.from({ length: 20 }).map((_, i) => (
          <div key={i} style={{
            position: 'absolute', left: `${i * 5}%`, top: 0, width: 1, height: '100%',
            background: '#25343F', transform: `translateY(${Math.sin((frame + i * 20) / 30) * 12}px)`,
          }} />
        ))}
      </AbsoluteFill>

      {/* Floating particles */}
      {Array.from({ length: 26 }).map((_, i) => {
        const s = seededRandom(i + 1);
        const drift = (frame * (0.3 + s)) % 1120;
        return (
          <div key={i} style={{
            position: 'absolute', left: seededRandom(i + 5) * 1920, top: (seededRandom(i + 9) * 1080 + drift) % 1080,
            width: 4 + s * 5, height: 4 + s * 5, borderRadius: '50%',
            background: i % 3 === 0 ? '#FFF449' : '#25343F', opacity: 0.25,
          }} />
        );
      })}

      {/* Main content — headline + diverging graph */}
      <AbsoluteFill style={{ justifyContent: 'center', alignItems: 'center', flexDirection: 'column' }}>
        <div style={{
          fontFamily: 'Mont Heavy, Inter', fontWeight: 900, fontSize: 100, lineHeight: 1.02,
          color: '#25343F', textAlign: 'center', opacity: sp(6), transform: `translateY(${(1 - sp(6)) * 40}px)`, letterSpacing: -2,
        }}>
          PAYCHECK <span style={{ color: '#FFF449', textShadow: '0 2px 0 #25343F' }}>UP</span>.
          <br />LIFE <span style={{ color: '#FFF8F0' }}>LESS</span> AFFORDABLE.
        </div>

        {/* Diverging line graph SVG */}
        <svg width={980} height={340} style={{ marginTop: 30, opacity: sp(30) }}>
          <line x1={40} y1={300} x2={940} y2={300} stroke="#25343F" strokeWidth={3} opacity={0.4} />
          {/* Paycheck rising line */}
          <polyline
            points={`40,240 ${40 + 900 * payGrow},${240 - 150 * payGrow}`}
            fill="none" stroke="#FFF449" strokeWidth={9} strokeLinecap="round"
          />
          <circle cx={40 + 900 * payGrow} cy={240 - 150 * payGrow} r={12} fill="#FFF449" stroke="#25343F" strokeWidth={3} />
          {/* Affordability falling line */}
          <polyline
            points={`40,140 ${40 + 900 * affDrop},${140 + 150 * affDrop}`}
            fill="none" stroke="#25343F" strokeWidth={9} strokeLinecap="round" strokeDasharray="4 14"
          />
          <circle cx={40 + 900 * affDrop} cy={140 + 150 * affDrop} r={12} fill="#25343F" />
        </svg>
      </AbsoluteFill>

      {/* Vignette overlay */}
      <AbsoluteFill style={{ boxShadow: 'inset 0 0 320px rgba(40,25,10,0.6)', pointerEvents: 'none' }} />

      {/* Watermark */}
      <div style={{ position: 'absolute', bottom: 30, right: 45, fontFamily: 'Inter', fontSize: 22, fontWeight: 600, color: '#25343F', opacity: 0.3 }}>
        KuxAI.7x
      </div>
    </AbsoluteFill>
  );
};

export default PaycheckVsAffordability;
