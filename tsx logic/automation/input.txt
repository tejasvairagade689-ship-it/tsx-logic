```tsx
import React from "react";
import {
  AbsoluteFill,
  Composition,
  Sequence,
  interpolate,
  useCurrentFrame,
  useVideoConfig,
} from "remotion";

/* ---------------- SCENE 1 ---------------- */

const IntroScene = () => {
  return (
    <AbsoluteFill
      style={{
        background: "#000",
        justifyContent: "center",
        alignItems: "center",
        color: "white",
      }}
    >
      <div
        style={{
          fontSize: 100,
          fontWeight: "bold",
          textAlign: "center",
        }}
      >
        WHAT IS A
        <br />
        BLACK HOLE?
      </div>
    </AbsoluteFill>
  );
};

/* ---------------- SCENE 2 ---------------- */

const GravityScene = () => {
  const frame = useCurrentFrame();

  const x = interpolate(
    frame,
    [0, 150],
    [900, 540],
    {
      extrapolateRight: "clamp",
    }
  );

  return (
    <AbsoluteFill
      style={{
        background: "#020617",
      }}
    >
      <h1
        style={{
          color: "white",
          textAlign: "center",
          marginTop: 200,
          fontSize: 70,
        }}
      >
        Gravity Pulls
        <br />
        Everything In
      </h1>

      {/* Light Particle */}
      <div
        style={{
          position: "absolute",
          left: x,
          top: 900,
          width: 50,
          height: 50,
          borderRadius: "50%",
          background: "yellow",
          boxShadow: "0 0 20px yellow",
        }}
      />

      {/* Black Hole */}
      <div
        style={{
          position: "absolute",
          left: 440,
          top: 800,
          width: 200,
          height: 200,
          borderRadius: "50%",
          background: "black",
          boxShadow: "0 0 60px purple",
        }}
      />
    </AbsoluteFill>
  );
};

/* ---------------- SCENE 3 ---------------- */

        style={{

      <p
        style={{

      </p>
    </AbsoluteFill>
  );
};

/* ---------------- SCENE 4 ---------------- */

const SpaceTimeScene = () => {
  const frame = useCurrentFrame();

  return (
    <AbsoluteFill
      style={{
        background: "#020617",
        justifyContent: "center",
        alignItems: "center",
      }}
    >
      <h1
        style={{
          position: "absolute",
          top: 180,
          color: "white",
          fontSize: 70,
        }}
      >
        Space & Time Bend
      </h1>

      {Array.from({ length: 15 }).map((_, i) => (
        <div
          key={i}
          style={{
            width: 900,
            height: 2,
            background: "white",
            marginBottom: 25,
            transform: `translateX(${
              Math.sin(frame / 10 + i) * 80
            }px)`,
          }}
        />
      ))}
    </AbsoluteFill>
  );
};

/* ---------------- SCENE 5 ---------------- */
```
