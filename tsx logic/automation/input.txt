  id: 'Scene1_Intro',
  durationInSeconds: 13,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene1_Intro: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();
  
  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // Camera zoom
  const camScale = interpolate(frame, [0, 60, durationInFrames - 15], [0.95, 1.05, 1], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // Particle positions
  const particles = Array.from({ length: 25 }, (_, i) => {
    const seed = i + 1;
    return {
      x: seededRandom(seed * 1) * 1920,
      y: seededRandom(seed * 2) * 1080,
      size: seededRandom(seed * 3) * 60 + 20,
      speed: seededRandom(seed * 4) * 0.5 + 0.2,
      opacity: seededRandom(seed * 5) * 0.3 + 0.1,
    };
  });

  return (
    <AbsoluteFill style={{ backgroundColor: '#ffffff', overflow: 'hidden', opacity: exitOp, transform: `scale(${camScale})` }}>

            strokeWidth="1"
            opacity="0.08"

        ))}
