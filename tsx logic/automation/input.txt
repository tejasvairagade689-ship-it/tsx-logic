          <span style={{ fontFamily: "'Yantra One', 'Noto Sans Devanagari', sans-serif", fontSize: 96, color: '#000000' }}>
            परिवार के बिज़नेस का <span style={{ background: '#000000', color: '#C9A227', padding: '0 24px', display: 'inline-block', transform: 'rotate(-1deg)' }}>सही ढांचा?</span>
          </span>
        </div>

        {/* 4 entity door-cards */}
        <div style={{ position: 'absolute', top: 270, width: '100%', display: 'flex', justifyContent: 'center', gap: 40 }}>
          {doors.map((d, i) => {
            const v = sp(d.delay);
            const hover = Math.sin(frame * 0.06 + i * 1.8) * 5;
            return (
              <div key={i} style={{
                width: 340, height: 430, position: 'relative',
                transform: `rotate(${(i % 2 ? 1.8 : -1.8)}deg) translateY(${(1 - v) * 560 + hover}px)`,
                opacity: Math.min(1, v * 1.3),
              }}>
                {/* Door frame */}
                <div style={{
                  position: 'absolute', inset: 0, background: i === 3 ? '#000000' : '#F4F2F2',
                  border: '4px solid #000000', borderTopLeftRadius: 160, borderTopRightRadius: 160,
                  boxShadow: '10px 12px 0 rgba(0,0,0,0.3)',
                }} />
                {/* Door knob micro-animation */}
                <div style={{
                  position: 'absolute', right: 44, top: 250, width: 20, height: 20, borderRadius: '50%',
                  background: '#C9A227', transform: `scale(${1 + Math.sin(frame * 0.12 + i * 2) * 0.15})`,
                }} />
                <div style={{ position: 'absolute', top: 120, width: '100%', textAlign: 'center' }}>
                  <div style={{
                    fontFamily: "'JetBrains Mono', monospace", fontSize: 40, fontWeight: 800,
                    color: i === 3 ? '#C9A227' : '#000000',
                  }}>
                    {d.name}
                  </div>
                  <div style={{
                    fontFamily: "'Noto Sans Devanagari', sans-serif", fontSize: 30, fontWeight: 700,
                    color: i === 3 ? '#F4F2F2' : '#000000', opacity: 0.6, marginTop: 12, padding: '0 24px',
                  }}>
                    {d.hint}
                  </div>
                </div>
                {/* Number plate */}
                <div style={{
                  position: 'absolute', top: 34, left: '50%', transform: 'translateX(-50%)',
                  background: i === 3 ? '#C9A227' : '#000000', color: i === 3 ? '#000000' : '#F4F2F2',
                  fontFamily: "'JetBrains Mono', monospace", fontSize: 30, fontWeight: 800, padding: '2px 18px',
                }}>
                  0{i + 1}
                </div>
              </div>
            );
          })}
        </div>

        {/* Criteria ribbon */}
        <div style={{
          position: 'absolute', bottom: 90, width: '100%', display: 'flex', justifyContent: 'center', gap: 26,
        }}>
          {criteria.map((c, i) => {
            const v = sp(420 + i * 40);
            return (
              <div key={i} style={{
                background: '#000000', color: '#F4F2F2', padding: '10px 34px',
                fontFamily: 'Inter, sans-serif', fontSize: 36, fontWeight: 800, letterSpacing: 2,
                transform: `rotate(${(seededRandom(i + 4) - 0.5) * 4}deg) scale(${Math.min(1, v * 1.1)})`,
                opacity: Math.min(1, v * 1.3),
                borderBottom: '6px solid #C9A227',
                boxShadow: '6px 7px 0 rgba(0,0,0,0.3)',
              }}>
                {c}
              </div>
            );
          })}
        </div>
      </AbsoluteFill>

      <AbsoluteFill style={{ background: 'radial-gradient(ellipse at center, transparent 55%, rgba(0,0,0,0.28) 100%)', pointerEvents: 'none' }} />

      <div style={{ position: 'absolute', bottom: 30, right: 45, fontFamily: 'Inter, sans-serif', fontSize: 22, fontWeight: 600, color: '#000000', opacity: 0.3 }}>
        ProBuffet HQ
      </div>
    </AbsoluteFill>
  );
};
