import React from 'react';
import { useCurrentFrame, useVideoConfig, interpolate, spring, AbsoluteFill } from 'remotion';

export const compositionConfig = {
  id: 'Scene1_EnergySignal',
  durationInSeconds: 6,
  fps: 30,
  width: 1920,
  height: 1080,
};

// Welcome to the Remotion snippet!
// This component renders a glowing energy sector signal chart.

const Scene1_EnergySignal: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps } = useVideoConfig();
  
  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [165, 180], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  const points = Array.from({ length: 40 }, (_, i) => {
    const x = 80 + i * 44;
    const base = 420 + Math.sin(i * 0.5) * 60;
    const noise = Math.sin(i * 2.3) * 35 + Math.cos(i * 1.1) * 25;
    const y = base + noise + (i > 35 ? -(i - 35) * 10 : 0);
    return `${x},${y}`;
  }).join(' ');

  const endX = 80 + 39 * 44;
  const endY = 420 + Math.sin(39 * 0.5) * 60 + Math.sin(39 * 2.3) * 35 + Math.cos(39 * 1.1) * 25 - (39 - 35) * 10;

  const pulse = interpolate(frame % 30, [0, 15, 30], [1, 1.5, 1], { extrapolateRight: 'clamp' });
  const pulseOp = interpolate(frame % 30, [0, 15, 30], [0.9, 0.4, 0.9], { extrapolateRight: 'clamp' });

  return (
    <AbsoluteFill style={{ backgroundColor: '#D9D9D9', overflow: 'hidden', opacity: exitOp }}>
      <div style={{
        position: 'absolute', top: 70, left: 140,
        fontFamily: 'Inter, Poppins, sans-serif', fontSize: 52, fontWeight: 800, color: '#1C1C1C', letterSpacing: '-0.5px'
      }}>
        Energy Sector
        <div style={{ width: 200, height: 5, backgroundColor: '#C8A97E', marginTop: 10, borderRadius: 2 }} />
      </div>

      <div style={{
        position: 'absolute', top: 180, left: 140, right: 140, bottom: 160,
        backgroundColor: '#1C1C1C', borderRadius: 24, overflow: 'hidden', boxShadow: '0 20px 60px rgba(0,0,0,0.25)'
      }}>
        {[0,1,2,3].map(i => (
          <div key={i} style={{ position: 'absolute', top: 120 + i * 160, left: 60, right: 60, height: 1, backgroundColor: 'rgba(255,255,255,0.08)' }} />
        ))}

        <svg style={{ position: 'absolute', top: 0, left: 0, width: '100%', height: '100%' }}>
          <defs>
            <filter id="glow1">
              <feGaussianBlur stdDeviation="5" result="coloredBlur"/>
              <feMerge><feMergeNode in="coloredBlur"/><feMergeNode in="SourceGraphic"/></feMerge>
            </filter>
          </defs>
          <polyline
            points={points}
            fill="none"
            stroke="#00BCD4"
            strokeWidth={4}
            filter="url(#glow1)"
            strokeDasharray={2400}
            strokeDashoffset={interpolate(frame, [0, 100], [2400, 0], { extrapolateRight: 'clamp' })}
            strokeLinecap="round"
          />
          <circle cx={endX} cy={endY} r={14 * pulse} fill="#00BCD4" opacity={pulseOp} filter="url(#glow1)" />
        </svg>

        <div style={{
          position: 'absolute', right: 90, top: 200,
          backgroundColor: '#00BCD4', color: '#1C1C1C', padding: '10px 24px', borderRadius: 8,
          fontFamily: 'Inter', fontSize: 26, fontWeight: 800, transform: `scale(${sp(90)})`, transformOrigin: 'center'

        ProBuffet HQ
      </div>
    </AbsoluteFill>
  );
};

export default Scene1_EnergySignal;
