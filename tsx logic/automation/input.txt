            {[100, 95, 88, 92, 80].map((w, i) => (
              <div key={i} style={{
                height: 14,
                width: `${w}%`,
                background: '#215E61',
                opacity: 0.2 * textProgress,
              }} />
            ))}
          </div>

          {/* Hidden items (revealed by magnifying glass) */}
          {hiddenItems.map((item, i) => {
            const dist = Math.sqrt(Math.pow(magnifyX - item.x, 2) + Math.pow(magnifyY - item.y, 2));
            const revealedByMagnify = dist < 140 ? 1 : 0;
            return (
              <div
                key={i}
                style={{
                  position: 'absolute',
                  left: item.x - 220,
                  top: item.y - 200,
                  fontFamily: 'Anton',
                  fontSize: 40,
                  color: '#FBF5A7',
                  letterSpacing: 2,
                  opacity: revealedByMagnify,
                  transform: `scale(${0.85 + revealedByMagnify * 0.15})`,
                  background: '#215E61',
                  padding: '6px 16px',
                  borderRadius: 4,
                  whiteSpace: 'nowrap',
                  pointerEvents: 'none',
                }}
              >
                {item.label}
              </div>
            );
          })}
        </div>

        {/* Magnifying glass */}
        {frame >= 180 && (
          <div
            style={{
              position: 'absolute',
              left: magnifyX - 110,
              top: magnifyY - 110,
              width: 220,
              height: 220,
              opacity: magnifyIn,
              transform: `scale(${0.7 + magnifyIn * 0.3})`,
              pointerEvents: 'none',
            }}
          >
            <svg width="220" height="220" viewBox="0 0 220 220">
              <defs>
                <radialGradient id="lensGrad">
                  <stop offset="0%" stopColor="#FBF5A7" stopOpacity="0.15" />
                  <stop offset="100%" stopColor="#FBF5A7" stopOpacity="0.05" />
                </radialGradient>
              </defs>
              <circle cx="90" cy="90" r="74" fill="url(#lensGrad)" stroke="#FBF5A7" strokeWidth="8" />
              <circle cx="90" cy="90" r="74" fill="none" stroke="#215E61" strokeWidth="3" />
              {/* Handle */}
              <line x1="142" y1="142" x2="200" y2="200" stroke="#FBF5A7" strokeWidth="14" strokeLinecap="round" />
              <line x1="142" y1="142" x2="200" y2="200" stroke="#215E61" strokeWidth="6" strokeLinecap="round" />
              {/* Highlight */}
              <ellipse cx="65" cy="68" rx="22" ry="14" fill="#F4F2F2" opacity="0.3" />
            </svg>
          </div>
        )}

        {/* Final reveal — bottom punch */}
        {frame >= 420 && (
          <div
            style={{
              position: 'absolute',
              bottom: 60,
              left: '50%',
              transform: `translateX(-50%) scale(${0.9 + finalIn * 0.1})`,
              opacity: finalIn,
              textAlign: 'center',
            }}
          >
            <div style={{
              fontFamily: 'Clash Display',
              fontSize: 72,
              fontWeight: 900,
              color: '#FBF5A7',
              letterSpacing: -1,
              lineHeight: 1,
            }}>
              Mais regardez ce qu'il cache.
            </div>
            <div style={{
              fontFamily: 'Bricolage Grotesque',
              fontSize: 28,
              color: '#F4F2F2',
              fontStyle: 'italic',
              marginTop: 8,
              opacity: 0.85,
            }}>
              — c'est là que tout se joue.
            </div>
          </div>
        )}
      </AbsoluteFill>

      {/* Vignette */}
      <AbsoluteFill
        style={{
          background: 'radial-gradient(ellipse at center, rgba(0,0,0,0) 45%, rgba(0,0,0,0.6) 100%)',
          pointerEvents: 'none',
        }}
      />

      {/* Watermark */}
      <div
        style={{
          position: 'absolute',
          bottom: 30,
          right: 45,
          fontFamily: 'Montserrat',
          fontSize: 22,
          fontWeight: 600,
          color: '#9E9E9E',
          opacity: 0.3,
          letterSpacing: 1,
        }}
      >
        Matures Finance
      </div>
    </AbsoluteFill>
  );
};

export default Scene10_LeChiffreCache;
