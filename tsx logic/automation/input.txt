  durationInSeconds: 20,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // Headline reveal
  const headlineY = interpolate(sp(20), [0, 1], [80, 0]);
  const headlineOp = interpolate(sp(20), [0, 1], [0, 1]);

  // Subline reveal
  const sublineOp = interpolate(sp(80), [0, 1], [0, 1]);
  const sublineY = interpolate(sp(80), [0, 1], [40, 0]);

  const timeOp = interpolate(sp(140), [0, 1], [0, 1]);
  const timeScale = sp(140);

  // Steam from coffee
  const steamPhase = frame * 0.04;

  // Camera slow zoom
  const cameraScale = interpolate(frame, [0, durationInFrames], [1, 1.06]);

        style={{
          background: 'radial-gradient(circle at 30% 40%, #B8DC95 0%, #A5CF83 45%, #8FBE70 100%)',
          transform: `scale(${cameraScale})`,
        }}
      />

      {/* Animated shape layers */}
      {[0, 1, 2, 3, 4].map((i) => {
        const seed = seededRandom(i + 1);
        const size = 280 + seed * 320;

          <div
            key={`shape-${i}`}

              transform: `rotate(${rot}deg)`,

            y1={0}

            strokeWidth="1"

                    <g stroke={i === 0 ? '#EEEEEE' : '#000000'} strokeWidth="5" fill="none">
                      <line x1="15" y1="25" x2="30" y2="25" />
                      <path d="M 30 25 L 40 75 L 95 75 L 100 45 L 35 45" />
                      <circle cx="50" cy="95" r="8" />
                      <circle cx="85" cy="95" r="8" />
                    </g>
                  )}
                </svg>

                {/* Label */}
                <div
                  style={{
                    fontFamily: 'Anton, sans-serif',
                    fontSize: 38,
                    fontWeight: 900,
                    color: i === 0 ? '#EEEEEE' : '#000000',
                    letterSpacing: 1,
                    textAlign: 'center',
                  }}
                >
                  {d.label}
                </div>

                {/* Arrow indicator */}
                <div
                  style={{
                    fontFamily: 'Anton, sans-serif',
                    fontSize: 56,
                    fontWeight: 900,
                    color: i === 0 ? '#EEEEEE' : '#000000',
                    lineHeight: 0.8,
                  }}
                >
                  ↑
                </div>
              </div>
            </div>
          );
        })}
      </div>

      {/* Final stamp - ALLES TEURER */}
      <div
        style={{
          position: 'absolute',
          bottom: 100,
          left: '50%',
          marginLeft: -380,
          width: 760,
          opacity: stampOp,
          transform: `scale(${stampScale}) rotate(${stampRot}deg)`,
        }}
      >
        <div
          style={{
            backgroundColor: '#000000',
            border: '8px solid #000000',
            padding: '28px 50px',
            textAlign: 'center',
            boxShadow: '14px 14px 0px rgba(0,0,0,0.3)',
          }}
        >
          <div
            style={{
              fontFamily: 'Anton, sans-serif',
              fontSize: 110,
              fontWeight: 900,
              color: '#EEEEEE',
              letterSpacing: 4,
              lineHeight: 1,
            }}
          >
            ALLES TEURER
          </div>
        </div>
      </div>

      {/* Vignette overlay */}
      <AbsoluteFill
        style={{
          background: 'radial-gradient(ellipse at center, transparent 40%, rgba(0,0,0,0.5) 100%)',
          pointerEvents: 'none',
        }}
      />

      {/* Watermark */}
      <div
        style={{
          position: 'absolute',
          bottom: 30,
          right: 45,
          fontFamily: 'Inter, sans-serif',
          fontSize: 22,
          fontWeight: 600,
          color: '#000000',
          opacity: 0.4,
        }}
      >
        Kux legt an
      </div>
    </AbsoluteFill>
  );
};
