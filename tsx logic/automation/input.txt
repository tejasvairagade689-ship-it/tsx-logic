  fps: 30,
  width: 1920,

};

  return x - Math.floor(x);
};

    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',

  // Beat reveals
  const helloIn = sp(5);
  const phoneRingIn = sp(30);
  const irsLogoIn = sp(60);
  const warningIn = sp(95);
  const ctaIn = sp(140);

  // Phone shake (ringing)

  const phoneRot = Math.sin(frame * 0.5) * (frame > 30 && frame < 110 ? 4 : 0);

  // Pulse for warning
  const warnPulse = 1 + Math.sin(frame * 0.18) * 0.04;

  // Camera zoom
  const camZoom = interpolate(frame, [0, durationInFrames], [1, 1.05]);

    <AbsoluteFill style={{ backgroundColor: '#EEE3CB', overflow: 'hidden', opacity: exitOp }}>
