import React from 'react';
import { useCurrentFrame, useVideoConfig, interpolate, Easing, spring, AbsoluteFill } from 'remotion';

export const compositionConfig = {
  id: 'Scene1_Intro',
  durationInSeconds: 10,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene1_Intro: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // Camera zoom in
  const camScale = interpolate(frame, [0, durationInFrames], [1.08, 1.0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // Spotlight sweep
  const spotX = interpolate(frame, [0, durationInFrames], [-200, 200], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // Word reveal
  const word1Op = interpolate(frame, [10, 22], [0, 1], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' });
  const word2Op = interpolate(frame, [22, 34], [0, 1], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' });
  const word3Op = interpolate(frame, [34, 46], [0, 1], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' });

  // Stick figure
  const figureY = sp(40) * 0;
  const breathe = Math.sin(frame * 0.05) * 2;
  const headNod = Math.sin(frame * 0.04) * 3;
  const figureEntrance = sp(40) * 80;

  // Particles
  const particles = Array.from({ length: 18 }, (_, i) => ({
    x: seededRandom(i * 3 + 1) * 1920,
    y: seededRandom(i * 3 + 2) * 1080,
    r: seededRandom(i * 3 + 3) * 3 + 1,
    speed: seededRandom(i * 5) * 0.4 + 0.1,
    op: seededRandom(i * 7) * 0.4 + 0.1,
  }));

  return (
    <AbsoluteFill style={{ backgroundColor: '#1C1C1C', overflow: 'hidden', opacity: exitOp }}>

      {/* Gradient Background */}
      <AbsoluteFill style={{ transform: `scale(${camScale})` }}>
        <svg width="1920" height="1080" style={{ position: 'absolute' }}>
          <defs>
            <radialGradient id="bg1" cx="30%" cy="50%" r="70%">
              <stop offset="0%" stopColor="#2D2D2D" />
              <stop offset="100%" stopColor="#1C1C1C" />
            </radialGradient>
            <radialGradient id="spot1" cx="50%" cy="50%" r="50%">
              <stop offset="0%" stopColor="#e9d700" stopOpacity="0.07" />
              <stop offset="100%" stopColor="#e9d700" stopOpacity="0" />
            </radialGradient>
          </defs>
          <rect width="1920" height="1080" fill="url(#bg1)" />
          <ellipse cx={960 + spotX} cy={540} rx={500} ry={400} fill="url(#spot1)" />
        </svg>

        {/* Shape Layer — drifting large circle + rectangle */}
        <svg width="1920" height="1080" style={{ position: 'absolute' }}>
          <circle
            cx={200 + Math.sin(frame * 0.01) * 20}
            cy={300 + Math.cos(frame * 0.008) * 15}
            r={320}
            fill="none"
            stroke="#e9d700"
            strokeWidth="1.5"
            opacity={0.05}
          />
          <rect
            x={1400 + Math.sin(frame * 0.012) * 12}
            y={100 + Math.cos(frame * 0.009) * 10}
            width={380}
            height={380}
            rx={12}
            fill="none"
            stroke="#F5F5F5"
            strokeWidth="1"
            opacity={0.04}
            transform={`rotate(${frame * 0.05}, 1590, 290)`}
          />
          <polygon
            points={`${900 + Math.sin(frame * 0.007) * 8},${820} ${960},${730} ${1020 + Math.cos(frame * 0.009) * 6},${820}`}
            fill="none"
            stroke="#FFC107"
            strokeWidth="1.2"
            opacity={0.06}
          />
        </svg>

        {/* Motion Texture — horizontal scan lines */}
        <svg width="1920" height="1080" style={{ position: 'absolute', opacity: 0.03 }}>
          {Array.from({ length: 30 }, (_, i) => (
            <line key={i} x1={0} y1={i * 36} x2={1920} y2={i * 36} stroke="#F5F5F5" strokeWidth="0.8" />
          ))}
        </svg>

        {/* Floating Particles */}
        <svg width="1920" height="1080" style={{ position: 'absolute' }}>
          {particles.map((p, i) => (
            <circle
              key={i}
              cx={(p.x + frame * p.speed * 30) % 1920}
              cy={p.y - frame * p.speed * 10}
              r={p.r}
              fill="#e9d700"
              opacity={p.op}
            />
          ))}
        </svg>

        {/* Main Content */}
        <div style={{
          position: 'absolute',
          top: 0, left: 0, width: '100%', height: '100%',

          }}>
            जो मैंने साल भर के
          </div>

          {/* Highlight */}
          <div style={{
            opacity: word3Op,
            fontFamily: 'Noto Sans Devanagari',
            fontSize: 72,
            fontWeight: 900,

        <div style={{
