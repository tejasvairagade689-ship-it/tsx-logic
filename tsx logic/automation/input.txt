  id: 'Scene01_GasPriceShock',
  durationInSeconds: 19,
  fps: 30,
  width: 1920,
  height: 1080,
};

  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

  const priceShake = frame > 90 && frame < 110 ? Math.sin(frame * 2) * 8 : 0;

  const subY = interpolate(sp(110), [0, 1], [40, 0]);

  const washOp = interpolate(frame, [340, 380], [0, 1], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' });

  const flashOp = interpolate(frame, [220, 235, 250], [0, 0.4, 0], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' });

  // Gas pump SVG bob
  const pumpY = Math.sin(frame * 0.06) * 6;

  // Floating particles
  const particles = new Array(28).fill(0).map((_, i) => ({
    x: seededRandom(i + 1) * 1920,
    y: seededRandom(i + 11) * 1080,
    size: 3 + seededRandom(i + 21) * 6,
    speed: 0.3 + seededRandom(i + 31) * 0.8,
  }));

    <AbsoluteFill style={{ backgroundColor: '#B5BAFF', overflow: 'hidden', opacity: exitOp }}>
      {/* Gradient background */}
      <AbsoluteFill
        style={{
          background: 'radial-gradient(ellipse at 30% 40%, rgba(255,255,255,0.35) 0%, rgba(181,186,255,0) 60%), linear-gradient(135deg, #B5BAFF 0%, #9FA5F0 100%)',
        }}
      />

      {/* Animated shape layers */}
      <AbsoluteFill style={{ transform: `scale(${cameraZoom})` }}>
        <div
          style={{
            position: 'absolute',
            top: 120,
            left: -100,
            width: 600,
            height: 600,
            borderRadius: '50%',
            background: '#EEEEEE',
            opacity: 0.06,
            transform: `rotate(${frame * 0.15}deg) scale(${1 + Math.sin(frame * 0.02) * 0.05})`,
          }}
        />
        <div
          style={{
            position: 'absolute',
            bottom: -200,
            right: -150,
            width: 700,
            height: 700,
            background: '#000000',
            opacity: 0.04,
            transform: `rotate(${frame * -0.1}deg)`,
            borderRadius: 40,
          }}
        />
        <div
          style={{
            position: 'absolute',
            top: 400,
            right: 200,
            width: 0,
            height: 0,
            borderLeft: '180px solid transparent',
            borderRight: '180px solid transparent',
            borderBottom: '300px solid #000000',
            opacity: 0.05,
            transform: `rotate(${frame * 0.2}deg)`,
          }}
        />
      </AbsoluteFill>

              left: 0,

              height: 1,
              background: '#000000',
              transform: `translateX(${Math.sin((frame + i * 10) * 0.02) * 20}px)`,
            }}
          />

          <div
            key={i}
            style={{
              position: 'absolute',
              left: p.x,
              top: (p.y + frame * p.speed) % 1080,
              width: p.size,
              height: p.size,
              borderRadius: '50%',
              background: '#EEEEEE',
              opacity: 0.5,
