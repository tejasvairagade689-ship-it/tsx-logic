import React from 'react';
import {
  useCurrentFrame,
  useVideoConfig,
  interpolate,
  Easing,
  AbsoluteFill,
} from 'remotion';

export const compositionConfig = {
  id: 'elSilencioDelMiedo',
  durationInSeconds: 10,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const ElSilencioDelMiedo: React.FC = () => {
  const frame = useCurrentFrame();
  const { durationInFrames } = useVideoConfig();

  const bg = '#0D0D0D';
  const particle = '#6B5B3E';
  const glow = '#1A1510';
  const titleColor = '#FCFAEE';

  const fadeOut = interpolate(frame, [durationInFrames - 10, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp',
    extrapolateRight: 'clamp',
  });

  const chapterOpacity = interpolate(frame, [20, 35], [0, 1], {
    extrapolateLeft: 'clamp',
    extrapolateRight: 'clamp',
    easing: Easing.out(Easing.cubic),
  });

  const chapterBlur = interpolate(frame, [20, 35], [15, 0], {
    extrapolateLeft: 'clamp',
    extrapolateRight: 'clamp',
  });

  const chapterY = interpolate(frame, [20, 35], [15, 0], {
    extrapolateLeft: 'clamp',
    extrapolateRight: 'clamp',
    easing: Easing.out(Easing.cubic),
  });

  const lineWidth = interpolate(frame, [35, 50], [0, 120], {
    extrapolateLeft: 'clamp',
    extrapolateRight: 'clamp',
    easing: Easing.out(Easing.cubic),
  });

  const words = ['El', 'Silencio', 'Del', 'Miedo'];

  return (
    <AbsoluteFill
      style={{
        backgroundColor: bg,
        overflow: 'hidden',
      }}
    >
      <div
        style={{
          position: 'absolute',
          inset: 0,
          opacity: fadeOut,
          background: `
            radial-gradient(circle at 20% 20%, ${glow} 0%, transparent 35%),
            radial-gradient(circle at 80% 30%, rgba(107,91,62,0.08) 0%, transparent 28%),
            radial-gradient(circle at 50% 70%, rgba(26,21,16,0.9) 0%, transparent 40%),
            radial-gradient(circle at 60% 40%, rgba(107,91,62,0.05) 0%, transparent 30%),
            linear-gradient(180deg, #111111 0%, #0D0D0D 100%)
          `,
        }}
      />

      {[...Array(40)].map((_, i) => {
        const size = 1 + seededRandom(i + 1) * 2;
        return (
          <div
            key={`grain-${i}`}
            style={{
              position: 'absolute',
              left: `${seededRandom(i + 10) * 100}%`,
              top: `${seededRandom(i + 50) * 100}%`,
              width: size,
              height: size,
              borderRadius: '50%',
              backgroundColor: particle,
              opacity: 0.02 + seededRandom(i + 100) * 0.03,
            }}
          />
        );
      })}

      {[...Array(15)].map((_, i) => {
        const baseX = seededRandom(i + 200) * 1920;
        const baseY = seededRandom(i + 240) * 1080;
        const size = 1 + seededRandom(i + 280) * 3;
        const tw = 0.35 + 0.25 * Math.sin(frame / 12 + i);
        return (
          <div
            key={`star-${i}`}
            style={{
              position: 'absolute',
              left: baseX + Math.sin(frame / 30 + i) * 6,
              top: baseY + Math.cos(frame / 28 + i) * 6,
              width: size,
              height: size,
              borderRadius: '50%',
              backgroundColor: particle,
              opacity: tw,
              filter: `blur(${size * 0.3}px)`,
            }}
          />
        );
      })}

      {[...Array(5)].map((_, i) => {
        const size = 40 + seededRandom(i + 400) * 60;
        const x = seededRandom(i + 430) * 1800;
        const y = seededRandom(i + 460) * 960;
        return (
          <div
            key={`orb-${i}`}
            style={{
              position: 'absolute',
              left: x + Math.sin(frame / 50 + i) * 40,
              top: y + Math.cos(frame / 55 + i) * 35,
              width: size,
              height: size,
              borderRadius: '50%',
              backgroundColor: particle,
              opacity: 0.03 + seededRandom(i + 500) * 0.03,
              filter: `blur(${size * 0.6}px)`,
            }}
          />
        );
      })}

      <div
        style={{
          position: 'absolute',
          inset: 0,
          opacity: 0.03 + ((Math.sin((frame / 120) * Math.PI * 2) + 1) / 2) * 0.07,
          background: `radial-gradient(circle at center, ${particle} 0%, transparent 45%)`,
        }}
      />

      <div
        style={{
          position: 'absolute',
          width: 340,
          height: 340,
          left: 120,
          top: 120,
          border: `1px solid ${particle}`,
          borderRadius: '50%',
          opacity: 0.06,
          transform: `rotate(${frame * 0.12}deg)`,
        }}
      />
      <div
        style={{
          position: 'absolute',
          width: 220,
          height: 220,
          right: 180,
          bottom: 140,
          border: `1px solid ${particle}`,
          opacity: 0.05,
          transform: `rotate(${frame * -0.18}deg)`,
        }}
      />
      <div
        style={{
          position: 'absolute',
          left: '50%',
          top: 140 + Math.sin(frame / 40) * 8,
          width: 0,
          height: 0,
          borderLeft: '70px solid transparent',
          borderRight: '70px solid transparent',
          borderBottom: `120px solid transparent`,
          outline: `1px solid ${particle}`,
          opacity: 0.05,
          transform: 'translateX(-50%)',
        }}
      />

      <div
        style={{
          position: 'absolute',
          inset: 0,
          background: 'radial-gradient(circle, transparent 40%, rgba(13,13,13,0.9) 100%)',
        }}
      />

      <div
        style={{
          position: 'absolute',
          inset: 0,
          display: 'flex',
          flexDirection: 'column',
          justifyContent: 'center',
          alignItems: 'center',
          zIndex: 20,
          opacity: fadeOut,
          padding: '0 140px',
        }}
      >
        <div
          style={{
            fontFamily: 'Playfair Display, Cormorant Garamond, EB Garamond, Georgia, serif',
            fontSize: 46,
            fontWeight: 500,
            fontStyle: 'italic',
            letterSpacing: 6,
            color: '#FF1E00',
            margin: 0,
            opacity: chapterOpacity,
            filter: `blur(${chapterBlur}px)`,
            transform: `translateY(${chapterY}px)`,
          }}
        >
          Capítulo 1
        </div>

        <div
          style={{
            marginTop: 18,
            width: lineWidth * 2,
            height: 1,
            backgroundColor: '#FF1E00',
            opacity: 0.6,
          }}
        />

        <div
          style={{
            marginTop: 34,
            display: 'flex',
            gap: 20,
            flexWrap: 'wrap',
            justifyContent: 'center',
          }}
        >
          {words.map((word, i) => {
            const start = 50 + i * 5;
            const op = interpolate(frame, [start, start + 10], [0, 1], {
              extrapolateLeft: 'clamp',
              extrapolateRight: 'clamp',
              easing: Easing.out(Easing.back(1.3)),
            });
            const sc = interpolate(frame, [start, start + 10], [1.1, 1], {
              extrapolateLeft: 'clamp',
              extrapolateRight: 'clamp',
              easing: Easing.out(Easing.back(1.3)),
            });
            const glowPulse = 8 + ((Math.sin(frame / 18) + 1) / 2) * 12;
            return (
              <span
                key={word}
                style={{
                  fontFamily: 'Playfair Display, Cormorant Garamond, EB Garamond, Georgia, serif',

                  fontWeight: 700,
                  letterSpacing: 2,
                  color: titleColor,
                  margin: 0,
                  opacity: op,
                  transform: `scale(${sc})`,
                  textShadow: `0 0 ${glowPulse}px ${particle}`,
                }}
              >
                {word}
              </span>
            );
          })}
        </div>

              fontSize: 20, fontWeight: 500,
              color: '#6B5B3E',
              opacity: 0.4,
              letterSpacing: 1.5,
              fontFamily: 'Playfair Display, Georgia, serif',
              fontStyle: 'italic',

              Mentor by Elisa
            </span>
          </div>
        );
      })()}
    </AbsoluteFill>
  );
};

export default ElSilencioDelMiedo;
