  fps: 30,
  width: 1920,
  height: 1080,
};

  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

  const { fps, durationInFrames } = useVideoConfig();

    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // Camera slow zoom
  const camScale = interpolate(frame, [0, durationInFrames], [1.0, 1.08]);

  // Question mark hero entrance
  const qmScale = sp(20, { damping: 8, stiffness: 100 });
  const qmRotate = Math.sin(frame * 0.03) * 4;

  // "दोस्तों" entrance
  const dostonOp = interpolate(frame, [10, 30], [0, 1], { extrapolateRight: 'clamp' });
  const dostonY = interpolate(sp(10), [0, 1], [40, 0]);

  // "एक सवाल" word reveal
  const ekSawalOp = interpolate(frame, [50, 75], [0, 1], { extrapolateRight: 'clamp' });
  const ekSawalScale = sp(50);

  // "?" mega highlight
  const highlightWidth = interpolate(frame, [90, 130], [0, 100], { extrapolateRight: 'clamp', easing: Easing.out(Easing.cubic) });

  // Chaiwala silhouette appear
  const chaiwalaOp = interpolate(frame, [180, 220], [0, 1], { extrapolateRight: 'clamp' });
  const chaiwalaY = interpolate(sp(180, { damping: 10 }), [0, 1], [60, 0]);

  // Steam animation
  const steamY = (frame * 1.2) % 80;
  const steamOp = interpolate(steamY, [0, 40, 80], [0, 0.6, 0]);

  // Clock 5 AM
  const clockScale = sp(280, { damping: 9 });
  const clockHand = interpolate(frame, [280, durationInFrames], [0, 30]);

  // Bottom subtext
  const subOp = interpolate(frame, [380, 420], [0, 1], { extrapolateRight: 'clamp' });

  return (
    <AbsoluteFill style={{ backgroundColor: '#4A4466', overflow: 'hidden', opacity: exitOp }}>
      {/* Gradient overlay */}

      {/* Animated shape layer - large circles drifting */}
      {[0, 1, 2, 3, 4].map((i) => {

        const size = 200 + seededRandom(i + 20) * 300;
        const drift = Math.sin(frame * 0.008 + i) * 30;
        return (
          <div key={`shape-${i}`} style={{
            position: 'absolute',
            left: x + drift,
            top: y - drift,
            width: size,
            height: size,
            borderRadius: '50%',
            border: '2px solid #EEEEEE',
            opacity: 0.05,
            transform: `scale(${1 + Math.sin(frame * 0.02 + i) * 0.1})`,
          }} />
        );
      })}

      {/* Grid texture motion */}
      <svg width="1920" height="1080" style={{ position: 'absolute', opacity: 0.06 }}>
