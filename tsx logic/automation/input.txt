    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',

    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
    easing: Easing.out(Easing.cubic),

  const textY = sp(10) * 0;
  const heroY = interpolate(sp(10), [0, 1], [60, 0]);

  // Pulse ring
  const pulseScale = interpolate(frame % 60, [0, 60], [0.8, 1.4], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
    easing: Easing.out(Easing.quad),
  });
  const pulseOp = interpolate(frame % 60, [0, 60], [0.4, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

    x: seededRandom(i * 3 + 1) * 1920,
    y: seededRandom(i * 3 + 2) * 1080,
    size: seededRandom(i * 3 + 3) * 4 + 2,
    speed: seededRandom(i * 7 + 4) * 0.4 + 0.1,
    seed: i,

    <AbsoluteFill style={{ backgroundColor: '#1C1C1C', overflow: 'hidden', opacity: exitOp }}>

        transform: `scale(${camZoom})`,
      }} />

      {/* Shape layer 1 — large drifting circle */}
      <AbsoluteFill style={{ pointerEvents: 'none' }}>
        <svg width="1920" height="1080" style={{ position: 'absolute' }}>
          <circle
            cx={960 + Math.sin(frame * 0.012) * 30}
            cy={540 + Math.cos(frame * 0.008) * 20}
            r={480}
            fill="none"
            stroke="#e9d700"
            strokeWidth="1"

            cx={200 + Math.sin(frame * 0.007) * 15}
            cy={200 + Math.cos(frame * 0.01) * 10}
            r={260}
            fill="#FFC107"
            opacity={0.03}
          />
          <rect
            x={1500 + Math.sin(frame * 0.009) * 10}
            y={600 + Math.cos(frame * 0.006) * 8}
            width={320}
            height={320}
            rx={20}
            fill="#e9d700"
            opacity={0.03}
            transform={`rotate(${frame * 0.08}, 1660, 760)`}
          />
        </svg>
      </AbsoluteFill>

      {/* Motion texture — horizontal scan lines */}
      <AbsoluteFill style={{ pointerEvents: 'none' }}>
        <svg width="1920" height="1080" style={{ position: 'absolute' }}>
          {Array.from({ length: 12 }, (_, i) => (
            <line

              x2={1920}
              y2={i * 95 + ((frame * 1.2) % 95)}

              strokeWidth="0.5"
