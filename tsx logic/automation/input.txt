import React from 'react';

import { useCurrentFrame, useVideoConfig, interpolate, Easing, spring, AbsoluteFill } from 'remotion';

export const compositionConfig = {
  id: 'Scene01_PhoneCall',
  durationInSeconds: 17,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene01_PhoneCall: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // Camera zoom
  const camScale = interpolate(frame, [0, durationInFrames], [1, 1.08], { extrapolateRight: 'clamp' });

  // Word-by-word reveal
  const word1 = sp(10);
  const word2 = sp(35);
  const word3 = sp(70);

  // Phone shake on ring
  const ringShake = Math.sin(frame * 0.8) * (frame > 60 && frame < 200 ? 4 : 0);

  // Pulse rings around phone
  const pulse1 = (frame * 2) % 100;
  const pulse2 = ((frame * 2) + 33) % 100;
  const pulse3 = ((frame * 2) + 66) % 100;

  return (
    <AbsoluteFill style={{ backgroundColor: '#FF61F8', overflow: 'hidden', opacity: exitOp }}>
      {/* Gradient overlay */}
      <AbsoluteFill style={{
        background: 'radial-gradient(circle at 30% 40%, rgba(238,238,238,0.15) 0%, transparent 60%), radial-gradient(circle at 70% 70%, rgba(8,6,22,0.25) 0%, transparent 55%)',
      }} />

      {/* Animated grid texture */}
      <svg width="1920" height="1080" style={{ position: 'absolute', opacity: 0.08 }}>
        <defs>
          <pattern id="grid01" width="80" height="80" patternUnits="userSpaceOnUse">
            <path d="M 80 0 L 0 0 0 80" fill="none" stroke="#080616" strokeWidth="1.5" />
          </pattern>
        </defs>
        <rect width="1920" height="1080" fill="url(#grid01)" transform={`translate(${(frame * 0.3) % 80}, ${(frame * 0.2) % 80})`} />
      </svg>

      {/* Shape Layer - drifting geometric forms */}
      <div style={{
        position: 'absolute', left: 200 + Math.sin(frame * 0.015) * 30, top: 150 + Math.cos(frame * 0.02) * 20,
        width: 380, height: 380, borderRadius: '50%', border: '3px solid #080616', opacity: 0.06,
        transform: `rotate(${frame * 0.1}deg)`,
      }} />
      <div style={{
        position: 'absolute', right: 100 + Math.cos(frame * 0.018) * 25, bottom: 100 + Math.sin(frame * 0.022) * 30,
        width: 450, height: 450, background: '#EEEEEE', opacity: 0.05,
        transform: `rotate(${45 + frame * 0.08}deg)`,
      }} />
      <div style={{
        position: 'absolute', left: '60%', top: '20%',
        width: 0, height: 0, borderLeft: '150px solid transparent', borderRight: '150px solid transparent',
        borderBottom: '260px solid #080616', opacity: 0.04,
        transform: `rotate(${-frame * 0.15}deg)`,
      }} />

      {/* Floating particles */}
      {Array.from({ length: 35 }).map((_, i) => {
        const sx = seededRandom(i + 1) * 1920;
        const sy = seededRandom(i + 100) * 1080;
        const size = 4 + seededRandom(i + 200) * 8;
        const drift = Math.sin(frame * 0.02 + i) * 30;
        return (
          <div key={i} style={{
            position: 'absolute', left: sx + drift, top: sy + Math.cos(frame * 0.018 + i) * 25,
            width: size, height: size, borderRadius: '50%',
            background: i % 2 === 0 ? '#EEEEEE' : '#080616',
            opacity: 0.25 + Math.sin(frame * 0.05 + i) * 0.15,
          }} />
        );
      })}

      {/* Main content area with camera zoom */}
      <div style={{ position: 'absolute', inset: 0, transform: `scale(${camScale})`, transformOrigin: 'center' }}>

        {/* Big Phone Icon - left side */}
        <div style={{
          position: 'absolute', left: '12%', top: '50%',
          transform: `translateY(-50%) translateX(${ringShake}px) scale(${sp(20)})`,
        }}>
          {/* Pulse rings */}
          <div style={{
            position: 'absolute', left: '50%', top: '50%',
            width: 280 + pulse1 * 3, height: 280 + pulse1 * 3,
            marginLeft: -(280 + pulse1 * 3) / 2, marginTop: -(280 + pulse1 * 3) / 2,
            borderRadius: '50%', border: '4px solid #080616', opacity: (100 - pulse1) / 250,
          }} />
          <div style={{
            position: 'absolute', left: '50%', top: '50%',
            width: 280 + pulse2 * 3, height: 280 + pulse2 * 3,
            marginLeft: -(280 + pulse2 * 3) / 2, marginTop: -(280 + pulse2 * 3) / 2,
            borderRadius: '50%', border: '4px solid #EEEEEE', opacity: (100 - pulse2) / 250,
          }} />
          <div style={{
            position: 'absolute', left: '50%', top: '50%',
            width: 280 + pulse3 * 3, height: 280 + pulse3 * 3,

          <svg width="280" height="280" viewBox="0 0 100 100" style={{ position: 'relative' }}>
            <rect x="20" y="10" width="60" height="80" rx="10" fill="#080616" stroke="#EEEEEE" strokeWidth="2" />

            <circle cx="50" cy="56" r="3" fill="#080616" />

        <div style={{
          position: 'absolute', right: '6%', top: '50%', transform: 'translateY(-50%)',
          textAlign: 'right', maxWidth: '60%',
        }}>
          {/* Label */}
          <div style={{
            fontFamily: 'JetBrains Mono, monospace', fontSize: 32, color: '#080616',
            letterSpacing: '6px', marginBottom: 30, opacity: sp(5),
            transform: `translateX(${interpolate(sp(5), [0, 1], [40, 0])}px)`,
          }}>
            // INCOMING CALL
          </div>

          {/* Word 1 */}
          <div style={{
            fontFamily: 'Yantra One, serif', fontSize: 187, color: '#080616',

            transform: `translateY(${interpolate(word1, [0, 1], [60, 0])}px)`,

            दोस्तो,

          {/* Word 2 - highlighted */}
          <div style={{
            fontFamily: 'Yantra One, serif', fontSize: 153, color: '#EEEEEE',

            opacity: word2,
