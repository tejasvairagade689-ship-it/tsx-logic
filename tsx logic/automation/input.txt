  id: 'Scene1_Confession',
  durationInSeconds: 11,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene1_Confession: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // Camera zoom in slow
  const camZoom = interpolate(frame, [0, durationInFrames], [1, 1.04], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // Red crack line reveal
  const crackWidth = interpolate(frame, [10, 50], [0, 900], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
    easing: Easing.out(Easing.expo),
  });

  // Text stagger
  const line1Op = interpolate(frame, [8, 22], [0, 1], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' });
  const line2Op = interpolate(frame, [22, 38], [0, 1], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' });
  const line1Y = sp(8) * 0 + interpolate(sp(8), [0, 1], [40, 0]);
  const line2Y = sp(22) * 0 + interpolate(sp(22), [0, 1], [40, 0]);

  // Stick figure breath
  const breathY = Math.sin(frame * 0.05) * 2;
  const headNod = Math.sin(frame * 0.04) * 3;
  const figureScale = sp(40);

  // Particles
  const particles = Array.from({ length: 18 }, (_, i) => ({
    x: seededRandom(i * 3) * 1920,
    y: seededRandom(i * 3 + 1) * 1080,
    size: seededRandom(i * 3 + 2) * 4 + 2,
    speed: seededRandom(i * 5) * 0.4 + 0.1,
    opacity: seededRandom(i * 7) * 0.3 + 0.05,

      {/* Gradient Background */}

        transform: `scale(${camZoom})`,
        background: 'radial-gradient(ellipse at 20% 50%, #2a0a0a 0%, #1C1C1C 60%, #0d0d0d 100%)',

      {/* Shape Layer — large drifting circle */}

          position: 'absolute',
          width: 700,
          height: 700,

          opacity: 0.04,
          top: -150,
          left: -100,
          transform: `rotate(${frame * 0.15}deg) scale(${1 + Math.sin(frame * 0.02) * 0.03})`,
        }} />
        <div style={{
          position: 'absolute',
          width: 400,
          height: 400,
          borderRadius: '50%',
          border: '1px solid #e9d700',
          opacity: 0.05,
          bottom: 50,
          right: 200,
          transform: `rotate(${-frame * 0.1}deg)`,
        }} />
        {/* Triangle shape */}
