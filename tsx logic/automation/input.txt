  id: 'Scene01_PropertyTaxNoticeShock',
  durationInSeconds: 10,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene01_PropertyTaxNoticeShock: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // Camera zoom
  const camZoom = interpolate(frame, [0, 300], [1, 1.08], { extrapolateRight: 'clamp' });

  // Envelope entrance
  const envIn = sp(8, { damping: 14, stiffness: 90 });
  const envTilt = Math.sin(frame * 0.04) * 2;

  // Notice flap open
  const flapOpen = interpolate(sp(40), [0, 1], [0, -160]);

  // Paper slide out
  const paperY = interpolate(sp(70, { damping: 18 }), [0, 1], [80, -40]);
  const paperOp = interpolate(frame, [70, 95], [0, 1], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' });

  // Red stamp impact
  const stampScale = sp(140, { damping: 6, stiffness: 200 });
  const stampRot = interpolate(sp(140), [0, 1], [-30, -12]);
  const shake = frame > 140 && frame < 160 ? Math.sin(frame * 1.8) * 4 : 0;

  // Headline reveal
  const headlineOp = interpolate(frame, [180, 210], [0, 1], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' });
  const headlineY = interpolate(sp(180), [0, 1], [30, 0]);

  // Dollar particles
  const particles = Array.from({ length: 18 }).map((_, i) => {
    const seed = i + 1;
    const startX = seededRandom(seed) * 1920;
    const speed = 0.4 + seededRandom(seed + 10) * 0.6;
    const y = ((frame * speed * 3) + seededRandom(seed + 5) * 1080) % 1200 - 100;
    const op = 0.15 + seededRandom(seed + 7) * 0.25;
    const size = 18 + seededRandom(seed + 3) * 14;
    return { x: startX, y, op, size, seed };
  });

  return (
    <AbsoluteFill style={{ backgroundColor: '#C0C9EE', overflow: 'hidden', opacity: exitOp }}>
      {/* Dynamic gradient background */}
      <AbsoluteFill
        style={{
          background: 'radial-gradient(ellipse at 30% 40%, #D4DBF0 0%, #C0C9EE 45%, #A8B3DC 100%)',
        }}
      />

      {/* Animated shape layers - drifting transparent geometry */}
      <div
        style={{
          position: 'absolute',
          width: 720,
          height: 720,
          borderRadius: '50%',
          border: '2px solid #4E220F',
          opacity: 0.06,
          top: -200,
          right: -180,
          transform: `rotate(${frame * 0.15}deg) scale(${1 + Math.sin(frame * 0.02) * 0.05})`,
        }}
      />
      <div
        style={{
          position: 'absolute',
          width: 540,
          height: 540,
          backgroundColor: '#4E220F',
          opacity: 0.05,
          bottom: -160,
          left: -120,
          transform: `rotate(${45 + frame * 0.08}deg)`,
          borderRadius: 40,
        }}
      />
      <div
        style={{
          position: 'absolute',
          width: 380,
          height: 380,
          opacity: 0.04,
          top: 280,
          right: 380,
          transform: `rotate(${frame * 0.1}deg)`,
        }}
      >
        <svg viewBox="0 0 100 100" width="100%" height="100%">
          <polygon points="50,5 95,80 5,80" fill="#000000" />
        </svg>
      </div>

            y1={0}

            y2={1080}
            stroke="#4E220F"

            x1={0}
            y1={i * 96 + (frame * 0.2) % 96}
            x2={1920}
            y2={i * 96 + (frame * 0.2) % 96}
