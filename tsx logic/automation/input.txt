  durationInSeconds: 10,
  fps: 30,
  width: 1920,
  height: 1080,
};

  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // Camera zoom
  const camZoom = interpolate(frame, [0, durationInFrames], [1.0, 1.08], { extrapolateRight: 'clamp' });

  // Word reveals
  const word1 = sp(10);
  const word2 = sp(35);
  const word3 = sp(60);
  const underline = interpolate(frame, [80, 110], [0, 1], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' });
  const ringPulse = 1 + Math.sin(frame * 0.08) * 0.04;

  // Particles
  const particles = Array.from({ length: 24 }).map((_, i) => {
    const sx = seededRandom(i + 1) * 1920;
    const sy = seededRandom(i + 7) * 1080;
    const drift = Math.sin(frame * 0.02 + i) * 20;
    const op = 0.15 + seededRandom(i + 3) * 0.25;
    const size = 3 + seededRandom(i + 5) * 5;
    return { sx, sy, drift, op, size, i };
  });

  return (
    <AbsoluteFill style={{ backgroundColor: '#412D15', overflow: 'hidden', opacity: exitOp }}>
      {/* Dynamic gradient background */}
      <AbsoluteFill
        style={{
          background:
            'radial-gradient(ellipse at 30% 40%, rgba(250,185,91,0.18) 0%, rgba(65,45,21,0) 55%), radial-gradient(ellipse at 75% 75%, rgba(232,237,242,0.08) 0%, rgba(65,45,21,0) 60%)',

          height: 900,

          border: '2px solid #FAB95B',
          opacity: 0.05,
          transform: `rotate(${frame * 0.15}deg) scale(${1 + Math.sin(frame * 0.02) * 0.05})`,
        }}
      />
      <div
        style={{
          position: 'absolute',
          right: -300,
          bottom: -250,
          width: 1100,
          height: 1100,
          border: '2px solid #E8EDF2',
          opacity: 0.04,
          transform: `rotate(${-frame * 0.1}deg)`,
        }}
      />
      <div
        style={{
          position: 'absolute',
          left: '60%',
          top: '15%',
          width: 400,
          height: 400,
          background: '#FAB95B',
          opacity: 0.03,
          transform: `rotate(45deg) translateY(${Math.sin(frame * 0.03) * 30}px)`,
        }}
      />

      {/* Motion texture - vertical lines grid */}
      <svg width="1920" height="1080" style={{ position: 'absolute', inset: 0, opacity: 0.08 }}>
        {Array.from({ length: 20 }).map((_, i) => (
          <line
            key={i}
            x1={i * 100 + (frame * 0.3) % 100}
            y1={0}
            x2={i * 100 + (frame * 0.3) % 100}
            y2={1080}
            stroke="#E8EDF2"
