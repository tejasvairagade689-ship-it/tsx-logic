  fps: 30,
  width: 1920,
  height: 1080,
};

  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

  const { fps, durationInFrames } = useVideoConfig();

    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',

  return (
    <AbsoluteFill style={{ backgroundColor: '#B5B9F0', overflow: 'hidden', opacity: exitOp }}>
      {/* Gradient layer */}
      <AbsoluteFill style={{ background: 'radial-gradient(circle at 30% 25%, #C4C8F5 0%, #B5B9F0 45%, #9DA2E8 100%)' }} />

      {/* Shape layers — large drifting geometric forms */}

  fps: 30,

  height: 1080,

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene10EffetLaffer: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  const headIn = sp(6);
  // Laffer curve draw
  const curveProg = interpolate(frame, [30, 130], [0, 1], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp', easing: Easing.inOut(Easing.cubic) });
  const quoteIn = sp(175, { stiffness: 80 });

  // build laffer path points (parabola)
  const buildPath = () => {
    const x0 = 300, y0 = 780, w = 760, h = 480;
    let d = `M ${x0} ${y0}`;
    const steps = 60;
    const upto = Math.floor(steps * curveProg);
    for (let i = 1; i <= upto; i++) {
      const t = i / steps;
      const px = x0 + t * w;
      const py = y0 - Math.sin(t * Math.PI) * h;
      d += ` L ${px} ${py}`;
    }
    return d;
  };

      </svg>

        <span style={{ fontFamily: "'Anton', sans-serif", fontSize: 80, color: '#1D2128' }}>L'EFFET LAFFER</span>

        {/* axes */}

        <text x="700" y="830" textAnchor="middle" fontFamily="Montserrat, sans-serif" fontSize="32" fill="#1D2128">TAUX DE TAXATION →</text>
