  id: 'Scene1Ch1',
  durationInSeconds: 12,
  fps: 30,
  width: 1920,
  height: 1080,
};

  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  const camScale = interpolate(frame, [0, durationInFrames], [1, 1.06], { extrapolateRight: 'clamp' });

  // Red slash reveal

  const word1Op = interpolate(frame, [5, 20], [0, 1], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' });
  const word2Op = interpolate(frame, [20, 35], [0, 1], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' });
  const word3Op = interpolate(frame, [35, 50], [0, 1], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' });

  const word1Y = interpolate(frame, [5, 25], [60, 0], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp', easing: Easing.out(Easing.back(1.5)) });

    x: seededRandom(i * 3) * 1920,
    y: seededRandom(i * 3 + 1) * 1080,

          position: 'absolute', inset: 0,

        }} />

          position: 'absolute',

          borderRadius: '50%',
          border: '1.5px solid rgba(233,215,0,0.06)',
          top: -180, left: -120,
          transform: `rotate(${frame * 0.08}deg)`,
        }} />
        <div style={{
          position: 'absolute',
          width: 500, height: 500,

          transform: `rotate(${frame * 0.05}deg)`,

          ))}
        </svg>
