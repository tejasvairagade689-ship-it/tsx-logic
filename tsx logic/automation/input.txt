```tsx
import React from 'react';
import { useCurrentFrame, useVideoConfig, interpolate, Easing, spring, AbsoluteFill } from 'remotion';

export const compositionConfig = {
  id: 'Scene09_WalkThrough',
  durationInSeconds: 9,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene09_WalkThrough: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // Headline
  const headOp = interpolate(frame, [10, 30], [0, 1], { extrapolateRight: 'clamp' });
  const headY = interpolate(sp(10), [0, 1], [-40, 0]);

  // Three checklist items stagger
  const items = ['WHO QUALIFIES', 'THE NUMBERS', 'WHAT TO DO'];
  const itemStarts = [50, 110, 170];

  // Big "EVERYTHING" reveal
  const everyOp = interpolate(frame, [225, 250], [0, 1], { extrapolateRight: 'clamp' });
  const everyScale = sp(225, { damping: 8, stiffness: 110 });

  // Figure
  const figOp = interpolate(frame, [60, 90], [0, 1], { extrapolateRight: 'clamp' });
  const figY = Math.sin(frame * 0.05) * 2;
  const figHeadRot = Math.sin(frame * 0.04) * 3;

  // Camera slow pan up
  const cameraY = interpolate(frame, [0, durationInFrames], [20, -20], { extrapolateRight: 'clamp' });

  return (
    <AbsoluteFill style={{ backgroundColor: '#1C1C1C', overflow: 'hidden', opacity: exitOp }}>
      {/* Gradient */}
      <AbsoluteFill
        style={{
          background: 'linear-gradient(180deg, #2D2D2D 0%, #1C1C1C 100%)',
        }}
      />

      {/* Shape Layer */}
      <div
        style={{
          position: 'absolute',
          top: '10%',
          right: '10%',
          width: 400,
          height: 400,
          borderRadius: '50%',
          border: '4px solid #FFC107',
          opacity: 0.06,
          transform: `scale(${1 + Math.sin(frame * 0.03) * 0.06})`,
        }}
      />
      <div
        style={{
          position: 'absolute',
          bottom: -100,
          right: -100,
          width: 450,
          height: 450,
          background: '#e9d700',
          opacity: 0.04,
          borderRadius: 40,
          transform: `rotate(${frame * 0.1}deg)`,
        }}
      />
      <div
        style={{
          position: 'absolute',
          top: '40%',
          left: '5%',
          width: 280,
          height: 280,
          background: '#FFC107',
          opacity: 0.05,
          clipPath: 'polygon(50% 0%, 100% 50%, 50% 100%, 0% 50%)',
          transform: `rotate(${-frame * 0.08}deg)`,
        }}
      />

      {/* Motion texture - subtle plus signs */}
      <svg style={{ position: 'absolute', inset: 0, width: '100%', height: '100%', opacity: 0.05 }}>
        {Array.from({ length: 8 }).map((_, row) =>
          Array.from({ length: 14 }).map((_, col) => (
            <g key={`${row}-${col}`}>
              <line x1={col * 140 + 60} y1={row * 140 + 50} x2={col * 140 + 60} y2={row * 140 + 70} stroke="#e9d700" strokeWidth={2} />
              <line x1={col * 140 + 50} y1={row * 140 + 60} x2={col * 140 + 70} y2={row * 140 + 60} stroke="#e9d700" strokeWidth={2} />
            </g>
          ))
        )}
      </svg>

      {/* Particles */}
      {Array.from({ length: 22 }).map((_, i) => {
        const seed = seededRandom(i + 11);
        const x = seed * 1920;
        const y = (seededRandom(i + 55) * 1080 + frame * 0.6) % 1080;
        return (
          <div
            key={i}
            style={{
              position: 'absolute',
              left: x,
              top: y,
              width: 3,
              height: 3,
              borderRadius: '50%',
              background: '#FFC107',
              opacity: 0.4,
            }}
          />
        );
      })}

      {/* Content */}
      <div
        style={{
          position: 'absolute',
          inset: 0,
          display: 'flex',
          flexDirection: 'column',
          alignItems: 'center',
          paddingTop: 80,
          transform: `translateY(${cameraY}px)`,
        }}
      >
        {/* Headline */}
        <div
          style={{
            fontFamily: 'Inter',
            fontSize: 88,
            fontWeight: 900,
            color: '#F5F5F5',
            opacity: headOp,
            transform: `translateY(${headY}px)`,
            letterSpacing: -2,
            textAlign: 'center',
            lineHeight: 1.1,
          }}
        >
          I'LL COVER <span style={{ color: '#e9d700' }}>EVERYTHING.</span>
        </div>

        {/* Checklist */}
        <div style={{ marginTop: 90, display: 'flex', flexDirection: 'column', gap: 35 }}>
          {items.map((label, i) => {
            const itemOp = interpolate(frame, [itemStarts[i], itemStarts[i] + 25], [0, 1], { extrapolateRight: 'clamp' });
            const itemX = interpolate(sp(itemStarts[i]), [0, 1], [-200, 0]);
            const checkScale = sp(itemStarts[i] + 15, { damping: 6, stiffness: 140 });
            return (
              <div
                key={i}
                style={{
                  display: 'flex',
                  alignItems: 'center',
                  gap: 30,
                  opacity: itemOp,
                  transform: `translateX(${itemX}px)`,
                }}
              >
                {/* Checkbox */}
                <div
                  style={{
                    width: 90,
                    height: 90,
                    border: '5px solid #e9d700',
                    borderRadius: 12,
                    background: '#1C1C1C',
                    display: 'flex',
                    alignItems: 'center',
                    justifyContent: 'center',
                    flexShrink: 0,
                  }}
                >
                  <svg width={60} height={60} viewBox="0 0 60 60" style={{ transform: `scale(${checkScale})` }}>
                    <path
                      d="M 10 30 L 25 45 L 50 15"
                      stroke="#FFC107"
                      strokeWidth={8}
                      fill="none"
                      strokeLinecap="round"
                      strokeLinejoin="round"
                    />
                  </svg>
                </div>
                {/* Label */}
                <div
                  style={{
                    fontFamily: 'Inter',
                    fontSize: 72,
                    fontWeight: 900,
                    color: '#F5F5F5',
                    letterSpacing: -1,
                  }}
                >
                  {label}
                </div>
              </div>
            );
          })}
        </div>

        {/* No fluff line */}
        <div
          style={{
            marginTop: 60,
            fontFamily: 'Inter',
            fontSize: 42,
            fontWeight: 700,
            color: '#FFC107',
            opacity: everyOp,
            transform: `scale(${everyScale})`,
            letterSpacing: 2,
            textTransform: 'uppercase',
          }}
        >
          ━━ No Fluff. Just Facts. ━━
        </div>
      </div>

      {/* Stick Figure */}
      <div
        style={{
          position: 'absolute',
          bottom: 50,
          right: 100,
          opacity: figOp,
          transform: `translateY(${figY}px)`,
        }}
      >
        <div
          style={{
            position: 'absolute',
            top: 20,
            left: -20,
            width: 140,
            height: 140,
            borderRadius: '50%',
            background: '#FFC107',
            filter: 'blur(40px)',
            opacity: 0.4,
          }}
        />
        <svg width={120} height={220} viewBox="0 0 120 220">
          <g transform={`rotate(${figHeadRot} 60 50)`}>
            <circle cx={60} cy={50} r={28} fill="none" stroke="#F5F5F5" strokeWidth={4} />
            <circle cx={50} cy={48} r={9} fill="none" stroke="#FFC107" strokeWidth={3} />
            <circle cx={70} cy={48} r={9} fill="none" stroke="#FFC107" strokeWidth={3} />
            <line x1={59} y1={48} x2={61} y2={48} stroke="#FFC107" strokeWidth={3} />
            <path d="M 48 60 Q 60 70 72 60" fill="none" stroke="#F5F5F5" strokeWidth={3} strokeLinecap="round" />
          </g>
          <line x1={60} y1={80} x2={60} y2={150} stroke="#F5F5F5" strokeWidth={4} />
          {/* Thumbs up arm */}
          <line x1={60} y1={100} x2={35} y2={75} stroke="#F5F5F5" strokeWidth={4} strokeLinecap="round" />
          <circle cx={35} cy={70} r={6} fill="#FFC107" />
          <line x1={60} y1={100} x2={90} y2={130} stroke="#F5F5F5" strokeWidth={4} strokeLinecap="round" />
          <line x1={60} y1={150} x2={40} y2={200} stroke="#F5F5F5" strokeWidth={4} strokeLinecap="round" />
          <line x1={60} y1={150} x2={80} y2={200} stroke="#F5F5F5" strokeWidth={4} strokeLinecap="round" />
        </svg>
      </div>

      {/* Vignette */}
      <AbsoluteFill
        style={{
          background: 'radial-gradient(circle at center, transparent 45%, rgba(0,0,0,0.55) 100%)',
          pointerEvents: 'none',
        }}
      />

      {/* Watermark */}
      <div
        style={{
          position: 'absolute',
          bottom: 30,
          right: 45,
          fontFamily: 'Inter, sans-serif',
          fontSize: 22,
          fontWeight: 600,
          color: '#9E9E9E',
          opacity: 0.3,
          letterSpacing: 1,
        }}
      >
        Senior Fund USA
      </div>
    </AbsoluteFill>
  );
};

export default Scene09_WalkThrough;
```
