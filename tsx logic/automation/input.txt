  return x - Math.floor(x);
};

const Scene10_Earthquake: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // Camera: tension zoom + earthquake shake on payoff
  const camScale = interpolate(frame, [0, durationInFrames], [1, 1.1], { easing: Easing.inOut(Easing.ease) });
  const quakeAt = 430;
  const quakeIntensity = frame >= quakeAt
    ? interpolate(frame, [quakeAt, quakeAt + 10, quakeAt + 55], [0, 14, 0], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' })
    : 0;
  const shakeX = Math.sin(frame * 4.1) * quakeIntensity;
  const shakeY = Math.cos(frame * 3.7) * quakeIntensity * 0.6;

  // 3% -> 30% morph
  const bigPct = Math.round(interpolate(frame, [180, 400], [3, 30], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp', easing: Easing.inOut(Easing.cubic) }));

  // Seismograph: amplitude grows through the scene
  const seisPts: string[] = [];
  for (let x = 0; x <= 1400; x += 8) {
    const t = x / 1400;
    const local = (frame * 10 - (1400 - x)) / 30;
    const amp = 6 + t * 110 * interpolate(frame, [60, 420], [0.15, 1], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' });
    const y = 90 + Math.sin(local * 2.2 + x * 0.05) * amp * (0.4 + seededRandom(Math.floor(x / 8)) * 0.6);
    seisPts.push(`${x},${y}`);
  }

  return (
    <AbsoluteFill style={{ backgroundColor: '#122A25', overflow: 'hidden', opacity: exitOp }}>
      {/* Gradient + Shape Layers + Texture + Particles + Content + Vignette + Watermark */}

      {/* Layer 1: Dynamic gradient background */}
      <AbsoluteFill style={{
        background: `radial-gradient(ellipse at 50% 65%, #24493F 0%, #122A25 52%, #091915 100%)`,
      }} />

      {/* Layer 2: Animated transparent shapes */}
      <div style={{
        position: 'absolute', width: 820, height: 820, borderRadius: '50%',
        border: '2px solid #EBEDE3', opacity: 0.04,
        left: -300 + Math.sin(frame / 105) * 25, top: 350,
        transform: `rotate(${frame * 0.05}deg)`,
      }} />
      <div style={{
        position: 'absolute', width: 540, height: 540,
        border: '3px solid #FFF449', opacity: 0.04,
        right: -160, top: -180, borderRadius: 60,
        transform: `rotate(${frame * -0.07}deg) scale(${1 + Math.sin(frame / 92) * 0.05})`,
      }} />

      {/* Layer 3: Motion texture — cracked fault lines appear after quake */}
      <svg width="1920" height="1080" style={{ position: 'absolute', opacity: frame > quakeAt ? 0.1 : 0.03 }}>
        <path d="M0 820 L340 790 L520 850 L880 800 L1210 870 L1560 810 L1920 860" fill="none" stroke="#EBEDE3" strokeWidth="2" />
        <path d="M200 1080 L360 900 L410 940 M1100 1080 L1240 890 L1310 930" fill="none" stroke="#EBEDE3" strokeWidth="2" />
      </svg>

      {/* Layer 4: Floating particles — debris after impact */}
      {Array.from({ length: 15 }).map((_, i) => (
        <div key={i} style={{
          position: 'absolute',
          left: seededRandom(i + 91) * 1920 + Math.sin(frame / 35 + i) * (quakeIntensity > 0 ? 30 : 14),
          top: (seededRandom(i + 121) * 1080 + frame * (0.2 + seededRandom(i + 2) * 0.4)) % 1080,
          width: 2 + seededRandom(i + 8) * 5, height: 2 + seededRandom(i + 8) * 5,
          backgroundColor: i % 4 === 0 ? '#FFF449' : '#EBEDE3',
          borderRadius: i % 2 === 0 ? '50%' : 2,
          opacity: 0.06 + seededRandom(i + 15) * 0.1,
          transform: `rotate(${frame * (1 + seededRandom(i) * 2)}deg)`,
        }} />
      ))}

      {/* Layer 5: Main content */}
      <AbsoluteFill style={{
        transform: `scale(${camScale}) translate(${shakeX}px, ${shakeY}px)`,
        justifyContent: 'center', alignItems: 'center',
      }}>
        <div style={{ textAlign: 'center', width: 1500 }}>

          <div style={{
            fontFamily: 'Inter, sans-serif', fontSize: 40, fontWeight: 600, color: '#C9D6D0', letterSpacing: 4,
            opacity: sp(15), transform: `translateY(${(1 - sp(15)) * 30}px)`,
          }}>
            FINANCIAL PESSIMISM — <span style={{ color: '#FFF449', fontWeight: 800 }}>HIGHEST SINCE 2018</span>
          </div>

          {/* 3% -> 30% headline morph */}
          <div style={{ display: 'flex', justifyContent: 'center', alignItems: 'baseline', gap: 50, marginTop: 40, opacity: sp(120) }}>
            <div>
              <div style={{ fontFamily: 'JetBrains Mono, monospace', fontSize: 30, color: '#9DB8B0', letterSpacing: 6 }}>2021</div>
              <div style={{ fontFamily: 'Mont Heavy, Inter, sans-serif', fontSize: 90, fontWeight: 900, color: '#9DB8B0' }}>3%</div>
            </div>
            <svg width="130" height="60" viewBox="0 0 130 60" style={{ alignSelf: 'center' }}>
              <path d="M8 30 H100 M100 30 L78 12 M100 30 L78 48" stroke="#FFF449" strokeWidth="6" strokeLinecap="round" fill="none"
                strokeDasharray="160" strokeDashoffset={160 * (1 - interpolate(frame, [160, 210], [0, 1], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' }))} />
            </svg>
            <div>
              <div style={{ fontFamily: 'JetBrains Mono, monospace', fontSize: 30, color: '#FFF449', letterSpacing: 6 }}>2025</div>
              <div style={{
                fontFamily: 'Mont Heavy, Inter, sans-serif', fontSize: 210, fontWeight: 900, color: '#FFF449', lineHeight: 0.95,
                textShadow: '0 0 100px rgba(255,244,73,0.3)',
              }}>
                {bigPct}%
              </div>
            </div>
          </div>

          <div style={{
            fontFamily: 'Inter, sans-serif', fontSize: 36, color: '#C9D6D0', marginTop: 10, opacity: sp(240),
          }}>
            of families now call inflation their #1 financial problem
          </div>

          {/* Seismograph */}
          <svg width="1400" height="180" viewBox="0 0 1400 180" style={{ marginTop: 30, opacity: sp(70) }}>
            <line x1="0" y1="90" x2="1400" y2="90" stroke="rgba(235,237,227,0.15)" strokeWidth="1" />
            <polyline points={seisPts.join(' ')} fill="none" stroke="#FFF449" strokeWidth="3.5" strokeLinejoin="round" />
          </svg>

          {/* Payoff stamp */}
          <div style={{
            marginTop: 24, opacity: sp(quakeAt), transform: `scale(${1.7 - sp(quakeAt) * 0.7}) rotate(${-2 + sp(quakeAt) * 2}deg)`,
          }}>
            <span style={{
              fontFamily: 'Mont Heavy, Inter, sans-serif', fontSize: 108, fontWeight: 900, color: '#EBEDE3', letterSpacing: 4,
            }}>
              NOT A TREND. <span style={{ color: '#FFF449' }}>AN EARTHQUAKE.</span>
            </span>
          </div>
        </div>
      </AbsoluteFill>

      {/* Quake flash */}
      {frame >= quakeAt && frame < quakeAt + 8 && (
        <AbsoluteFill style={{ backgroundColor: '#FFF449', opacity: (quakeAt + 8 - frame) * 0.02 }} />
      )}

      {/* Layer 6: Vignette overlay */}
      <AbsoluteFill style={{ background: 'radial-gradient(ellipse at center, transparent 52%, rgba(4,12,10,0.82) 100%)', pointerEvents: 'none' }} />

      {/* Layer 7: Watermark */}
      <div style={{
        position: 'absolute', bottom: 30, right: 45, fontFamily: 'Inter, sans-serif',
        fontSize: 22, fontWeight: 600, color: '#9E9E9E', opacity: 0.3,
      }}>
        KuxAI.7x
      </div>
    </AbsoluteFill>
  );
};

export default Scene10_Earthquake;
