```tsx
import React from 'react';
import { useCurrentFrame, useVideoConfig, interpolate, Easing, spring, AbsoluteFill } from 'remotion';

export const compositionConfig = {
  id: 'Scene01_HookOpening',
  durationInSeconds: 9,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene01_HookOpening: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // Camera slow zoom in
  const cameraZoom = interpolate(frame, [0, durationInFrames], [1, 1.08], { extrapolateRight: 'clamp' });

  // Gradient breathing
  const gradShift = Math.sin(frame * 0.02) * 8;

  // Hero "65" entrance
  const heroScale = sp(20, { damping: 10, stiffness: 90 });
  const heroOp = interpolate(frame, [20, 40], [0, 1], { extrapolateRight: 'clamp' });

  // Headline reveal
  const headlineX = interpolate(sp(55), [0, 1], [-150, 0]);
  const headlineOp = interpolate(frame, [55, 75], [0, 1], { extrapolateRight: 'clamp' });

  // Sub reveal
  const subOp = interpolate(frame, [90, 110], [0, 1], { extrapolateRight: 'clamp' });
  const subY = interpolate(sp(90), [0, 1], [30, 0]);

  // Pulsing ring around 65
  const ringScale = 1 + Math.sin(frame * 0.08) * 0.05;
  const ringOp = 0.3 + Math.sin(frame * 0.08) * 0.15;

  // Stick figure entrance
  const figOp = interpolate(frame, [60, 90], [0, 1], { extrapolateRight: 'clamp' });
  const figY = Math.sin(frame * 0.05) * 2;
  const figHeadRot = Math.sin(frame * 0.04) * 3;

  // Shake at frame 35 (impact)
  const shakeX = frame > 35 && frame < 50 ? Math.sin(frame * 2) * (50 - frame) * 0.4 : 0;

  return (
    <AbsoluteFill style={{ backgroundColor: '#1C1C1C', overflow: 'hidden', opacity: exitOp }}>
      {/* Gradient background */}
      <AbsoluteFill
        style={{
          background: `radial-gradient(circle at ${50 + gradShift}% ${40 + gradShift}%, #2D2D2D 0%, #1C1C1C 65%)`,
        }}
      />

      {/* Shape Layer - large drifting geometric forms */}
      <div
        style={{
          position: 'absolute',
          top: -200,
          left: -100,
          width: 700,
          height: 700,
          borderRadius: '50%',
          border: '2px solid #e9d700',
          opacity: 0.06,
          transform: `rotate(${frame * 0.1}deg) scale(${1 + Math.sin(frame * 0.02) * 0.05})`,
        }}
      />
      <div
        style={{
          position: 'absolute',
          bottom: -150,
          right: -100,
          width: 550,
          height: 550,
          background: '#FFC107',
          opacity: 0.04,
          transform: `rotate(${-frame * 0.08}deg)`,
        }}
      />
      <div
        style={{
          position: 'absolute',
          top: '20%',
          right: '15%',
          width: 0,
          height: 0,
          borderLeft: '120px solid transparent',
          borderRight: '120px solid transparent',
          borderBottom: '200px solid #e9d700',
          opacity: 0.05,
          transform: `rotate(${frame * 0.15}deg)`,
        }}
      />

      {/* Motion Texture - horizontal lines */}
      <svg style={{ position: 'absolute', inset: 0, width: '100%', height: '100%', opacity: 0.08 }}>
        {Array.from({ length: 18 }).map((_, i) => (
          <line
            key={i}
            x1={0}
            y1={i * 60 + (frame * 0.5) % 60}
            x2={1920}
            y2={i * 60 + (frame * 0.5) % 60}
            stroke="#e9d700"
            strokeWidth={1}
          />
        ))}
      </svg>

      {/* Floating Particles */}
      {Array.from({ length: 30 }).map((_, i) => {
        const seed = seededRandom(i + 1);
        const x = seed * 1920;
        const y = (seededRandom(i + 50) * 1080 + frame * 0.7) % 1080;
        const size = 2 + seededRandom(i + 100) * 4;
        return (
          <div
            key={i}
            style={{
              position: 'absolute',
              left: x,
              top: y,
              width: size,
              height: size,
              borderRadius: '50%',
              background: '#e9d700',
              opacity: 0.5,
            }}
          />
        );
      })}

      {/* Main Content */}
      <div
        style={{
          position: 'absolute',
          inset: 0,
          display: 'flex',
          flexDirection: 'column',
          alignItems: 'center',
          justifyContent: 'center',
          transform: `scale(${cameraZoom}) translateX(${shakeX}px)`,
        }}
      >
        {/* Pulsing ring behind 65 */}
        <div
          style={{
            position: 'absolute',
            top: '20%',
            width: 500,
            height: 500,
            borderRadius: '50%',
            border: '4px solid #e9d700',
            opacity: ringOp * heroOp,
            transform: `scale(${ringScale * heroScale})`,
          }}
        />

        {/* Hero "65" */}
        <div
          style={{
            fontFamily: 'Inter, Poppins, sans-serif',
            fontSize: 260,
            fontWeight: 900,
            color: '#e9d700',
            lineHeight: 1,
            opacity: heroOp,
            transform: `scale(${heroScale})`,
            textShadow: '0 0 40px rgba(233, 215, 0, 0.5)',
            marginBottom: 20,
          }}
        >
          65
        </div>

        {/* Headline */}
        <div
          style={{
            fontFamily: 'Inter, Poppins, sans-serif',
            fontSize: 110,
            fontWeight: 800,
            color: '#F5F5F5',
            opacity: headlineOp,
            transform: `translateX(${headlineX}px)`,
            letterSpacing: -2,
            marginBottom: 20,
          }}
        >
          TURNING <span style={{ color: '#ff0000' }}>65?</span>
        </div>

        {/* Subhead */}
        <div
          style={{
            fontFamily: 'Inter, Poppins, sans-serif',
            fontSize: 50,
            fontWeight: 600,
            color: '#9E9E9E',
            opacity: subOp,
            transform: `translateY(${subY}px)`,
            letterSpacing: 4,
            textTransform: 'uppercase',
          }}>
          Watch Until The End
        </div>
      </div>

      {/* Stick Figure with Buffett glasses */}
      <div
        style={{
          position: 'absolute',
          bottom: 80,
          left: 100,
          opacity: figOp,
          transform: `translateY(${figY}px)`,
        }}
      >
        {/* Golden glow */}
        <div
          style={{
            position: 'absolute',
            top: 20,
            left: -20,
            width: 140,
            height: 140,
            borderRadius: '50%',
            background: '#FFC107',
            filter: 'blur(40px)',
            opacity: 0.3,
          }}
        />
        <svg width={120} height={220} viewBox="0 0 120 220">
          {/* Head */}
          <g transform={`rotate(${figHeadRot} 60 50)`}>
            <circle cx={60} cy={50} r={28} fill="none" stroke="#F5F5F5" strokeWidth={4} />
            {/* Buffett glasses */}
            <circle cx={50} cy={48} r={9} fill="none" stroke="#FFC107" strokeWidth={3} />
            <circle cx={70} cy={48} r={9} fill="none" stroke="#FFC107" strokeWidth={3} />
            <line x1={59} y1={48} x2={61} y2={48} stroke="#FFC107" strokeWidth={3} />
            {/* Smile */}
            <path d="M 50 60 Q 60 67 70 60" fill="none" stroke="#F5F5F5" strokeWidth={3} strokeLinecap="round" />
          </g>
          {/* Body */}
          <line x1={60} y1={80} x2={60} y2={150} stroke="#F5F5F5" strokeWidth={4} />
          {/* Arms */}
          <line x1={60} y1={100} x2={30} y2={130} stroke="#F5F5F5" strokeWidth={4} strokeLinecap="round" />
          <line x1={60} y1={100} x2={90} y2={130} stroke="#F5F5F5" strokeWidth={4} strokeLinecap="round" />
          {/* Legs */}
          <line x1={60} y1={150} x2={40} y2={200} stroke="#F5F5F5" strokeWidth={4} strokeLinecap="round" />
          <line x1={60} y1={150} x2={80} y2={200} stroke="#F5F5F5" strokeWidth={4} strokeLinecap="round" />
        </svg>
      </div>

      {/* Vignette */}
      <AbsoluteFill
        style={{
          background: 'radial-gradient(circle at center, transparent 40%, rgba(0,0,0,0.6) 100%)',
          pointerEvents: 'none',
        }}
      />

      {/* Watermark */}
      <div
        style={{
          position: 'absolute',
          bottom: 30,
          right: 45,
          fontFamily: 'Inter, sans-serif',
          fontSize: 22,
          fontWeight: 600,
          color: '#9E9E9E',
          opacity: 0.3,
          letterSpacing: 1,
        }}
      >
        Senior Fund USA
      </div>
    </AbsoluteFill>
  );
};

export default Scene01_HookOpening;
```
