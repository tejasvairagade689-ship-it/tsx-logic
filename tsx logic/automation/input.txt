  id: 'Scene1_Imagine2019',
  durationInSeconds: 5,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene1_Imagine2019: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 14, mass: 0.5, stiffness: 110, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  const yearReveal = sp(20);
  const wordPop = sp(45);
  const linePull = interpolate(sp(60), [0, 1], [0, 1]);
  const camZoom = interpolate(frame, [0, durationInFrames], [1, 1.04], { extrapolateRight: 'clamp' });

  return (
    <AbsoluteFill style={{ backgroundColor: '#E8E8E6', overflow: 'hidden', opacity: exitOp }}>
      {/* Soft paper gradient */}
      <AbsoluteFill style={{
        background: 'radial-gradient(ellipse at 30% 20%, #F2F2F0 0%, #E8E8E6 45%, #DCDCD8 100%)',
      }} />

      {/* Floating geometric shapes */}
      {Array.from({ length: 6 }).map((_, i) => {
        const seed = seededRandom(i + 1);
        const size = 180 + seed * 220;
        const x = seededRandom(i * 3.1) * 1920;
        const y = seededRandom(i * 7.7) * 1080;

  fps: 30,
  width: 1920,
  height: 1080,
};
