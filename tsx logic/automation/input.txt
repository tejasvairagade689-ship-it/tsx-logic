```tsx
import React from "react";
import {
  AbsoluteFill,
  interpolate,
  spring,
  useCurrentFrame,
  useVideoConfig,
} from "remotion";

export const BlackHoleScene: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps } = useVideoConfig();

  const scale = spring({
    fps,
    frame,
    config: {
      damping: 12,
    },
  });

  const ringRotate1 = frame * 2;
  const ringRotate2 = frame * -1.5;

  const titleOpacity = interpolate(frame, [0, 30], [0, 1], {
    extrapolateRight: "clamp",
  });

  const text1Opacity = interpolate(frame, [60, 90], [0, 1], {
    extrapolateRight: "clamp",
  });

  const text2Opacity = interpolate(frame, [180, 210], [0, 1], {
    extrapolateRight: "clamp",
  });

  const text3Opacity = interpolate(frame, [300, 330], [0, 1], {
    extrapolateRight: "clamp",
  });

  return (
    <AbsoluteFill
      style={{
        background:
          "radial-gradient(circle at center, #1a1a2e 0%, #000000 70%)",
        justifyContent: "center",
        alignItems: "center",
        overflow: "hidden",
        fontFamily: "Arial",
      }}
    >
      {/* Stars */}
      {Array.from({ length: 120 }).map((_, i) => (
        <div
          key={i}
          style={{
            position: "absolute",
            width: 2,
            height: 2,
            borderRadius: "50%",
            background: "white",
            left: `${(i * 37) % 100}%`,
            top: `${(i * 53) % 100}%`,
            opacity: 0.8,
          }}
        />
      ))}

      {/* Title */}
      <div
        style={{
          position: "absolute",
          top: 120,
          fontSize: 80,
          fontWeight: "bold",
          color: "white",
          opacity: titleOpacity,
          textAlign: "center",
        }}
      >
        BLACK HOLES
      </div>

      {/* Black Hole */}
      <div
        style={{
          position: "relative",
          width: 500,
          height: 500,
          transform: `scale(${1 + scale * 0.2})`,
        }}
      >
        {/* Outer Ring */}
        <div
          style={{
            position: "absolute",
            inset: 0,
            borderRadius: "50%",
            border: "20px solid #7f5af0",
            transform: `rotate(${ringRotate1}deg)`,
            boxShadow: "0 0 50px #7f5af0",
          }}
        />

        {/* Inner Ring */}
        <div
          style={{
            position: "absolute",
            inset: 40,
            borderRadius: "50%",
            border: "15px solid #2cb67d",
            transform: `rotate(${ringRotate2}deg)`,
            boxShadow: "0 0 40px #2cb67d",
          }}
        />

        {/* Event Horizon */}
        <div
          style={{
            position: "absolute",
            inset: 100,
            borderRadius: "50%",
            background: "#000",
            boxShadow:
              "0 0 80px rgba(255,255,255,0.3), inset 0 0 30px rgba(255,255,255,0.2)",
          }}
        />
      </div>

      {/* Caption 1 */}
      <div
        style={{
          position: "absolute",
          bottom: 300,
          color: "white",
          fontSize: 55,
          fontWeight: "bold",
          textAlign: "center",
          opacity: text1Opacity,
          width: "85%",
        }}
      >
        The most mysterious objects in space.
      </div>

      {/* Caption 2 */}
      <div
        style={{
          position: "absolute",
          bottom: 300,
          color: "#FFD700",
          fontSize: 55,
          fontWeight: "bold",
          textAlign: "center",
          opacity: text2Opacity,
          width: "85%",
        }}
      >
        Their gravity is so strong...
      </div>

      {/* Caption 3 */}
      <div
        style={{
          position: "absolute",
          bottom: 300,
          color: "#00FFFF",
          fontSize: 55,
          fontWeight: "bold",
          textAlign: "center",
          opacity: text3Opacity,
          width: "85%",
        }}
      >
        Even light cannot escape!
      </div>

      {/* Follow Text */}
      <div
        style={{
          position: "absolute",
          bottom: 100,
          color: "white",
          fontSize: 40,
          opacity: interpolate(frame, [420, 450], [0, 1], {
            extrapolateRight: "clamp",
          }),
        }}
      >
        Follow for more space facts 🚀
      </div>
    </AbsoluteFill>
  );
};
```
