import React from 'react';

import { useCurrentFrame, useVideoConfig, interpolate, Easing, spring, AbsoluteFill } from 'remotion';

export const compositionConfig = {
  id: 'Scene1Opening',
  durationInSeconds: 4,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene1Opening: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // Camera zoom in slowly
  const camScale = interpolate(frame, [0, durationInFrames], [1, 1.05]);

  // Typewriter for first line
  const word1 = sp(10);
  const word2 = sp(30);
  const word3 = sp(55);
  const lineGrow = interpolate(frame, [40, 90], [0, 1], { extrapolateRight: 'clamp', easing: Easing.out(Easing.cubic) });

  // Whisper underline pulse
  const pulse = Math.sin(frame * 0.12) * 0.3 + 0.7;

  return (
    <AbsoluteFill style={{ backgroundColor: '#AEB784', overflow: 'hidden', opacity: exitOp }}>
      {/* Gradient background */}
      <AbsoluteFill style={{
        background: 'radial-gradient(circle at 30% 40%, #BFC79A 0%, #AEB784 50%, #8E9968 100%)',
      }} />

      {/* Shape layer - large drifting circles */}
      <div style={{
        position: 'absolute', width: 900, height: 900, borderRadius: '50%',
        border: '2px solid #262626', opacity: 0.06,
        top: -200, left: -200,
        transform: `rotate(${frame * 0.1}deg) scale(${1 + Math.sin(frame * 0.02) * 0.05})`,
      }} />
      <div style={{
        position: 'absolute', width: 600, height: 600, borderRadius: '50%',
        border: '2px solid #262626', opacity: 0.05,
        bottom: -150, right: -100,
        transform: `rotate(${-frame * 0.15}deg)`,
      }} />

      {/* Motion texture - subtle grid lines */}
      <svg style={{ position: 'absolute', inset: 0, opacity: 0.08 }} width="1920" height="1080">
        {Array.from({ length: 20 }).map((_, i) => (
          <line key={i} x1={i * 100} y1="0" x2={i * 100} y2="1080" stroke="#262626" strokeWidth="1" />
        ))}
      </svg>

      {/* Floating particles */}
      {Array.from({ length: 25 }).map((_, i) => {
        const r = seededRandom(i + 1);
        const x = r * 1920;
        const y = ((seededRandom(i + 50) * 1080) + frame * 0.4) % 1080;
        return (
          <div key={i} style={{
            position: 'absolute', left: x, top: y,
            width: 4, height: 4, borderRadius: '50%',
            backgroundColor: '#262626', opacity: 0.25,
          }} />
        );
      })}

      {/* Main content */}
      <AbsoluteFill style={{
        transform: `scale(${camScale})`,
        display: 'flex', flexDirection: 'column',
        justifyContent: 'center', alignItems: 'center',
      }}>
        <div style={{
          fontFamily: 'Yantra One', color: '#262626',
          fontSize: 119, fontWeight: 700, lineHeight: 1.1,
          textAlign: 'center', letterSpacing: '-2px',
        }}>
          <span style={{
            display: 'inline-block',
            opacity: word1, transform: `translateY(${(1 - word1) * 30}px)`,
            marginRight: 24,
          }}>एक</span>
          <span style={{
            display: 'inline-block',
            opacity: word2, transform: `translateY(${(1 - word2) * 30}px)`,
            marginRight: 24,
          }}>ऐसी</span>
          <span style={{
            display: 'inline-block',
            opacity: word3, transform: `translateY(${(1 - word3) * 30}px)`,
            color: '#5C2E2E',
          }}>सच्चाई</span>
        </div>

        <div style={{
          width: lineGrow * 400, height: 4,
          backgroundColor: '#262626', marginTop: 30,
          opacity: pulse,
        }} />

        <div style={{
          fontFamily: 'Noto Sans Devanagari', color: '#262626',
          fontSize: 51, fontWeight: 500, marginTop: 40,
          opacity: interpolate(frame, [70, 100], [0, 0.8], { extrapolateRight: 'clamp' }),
          textAlign: 'center',
        }}>
          जो कोई आपको सीधे नहीं बोलेगा
        </div>
      </AbsoluteFill>

      {/* Vignette */}
      <AbsoluteFill style={{
        background: 'radial-gradient(circle, transparent 50%, rgba(0,0,0,0.35) 100%)',
        pointerEvents: 'none',
      }} />

      {/* Watermark */}
      <div style={{
        position: 'absolute', bottom: 30, right: 45,
        fontFamily: 'Inter', fontSize: 22, fontWeight: 600,
        color: '#262626', opacity: 0.4,
      }}>ProBuffet HQ</div>
    </AbsoluteFill>
  );
};

export default Scene1Opening;

              const op = interpolate(dripFrame, [0, 20, 60, 80], [0, 1, 1, 0]);
              return (
                <circle key={i} cx="160" cy={380 + dy} r="6" fill="#e9d700" opacity={op} />
              );
            })}
          </svg>
          <div style={{
            fontFamily: 'Noto Sans Devanagari', color: '#5C2E2E',
            fontSize: 38, fontWeight: 700, marginTop: 10,
          }}>
            हर रोज़ रिसती जा रही
          </div>
        </div>

        {/* RIGHT - Thermometer / boiling frog */}
        <div style={{
          opacity: labelIn,
          width: 400, height: 600,
          display: 'flex', flexDirection: 'column', alignItems: 'center',
        }}>
          <div style={{
            fontFamily: 'Inter', color: '#262626',
            fontSize: 30, fontWeight: 700, letterSpacing: 2, marginBottom: 15,
          }}>
            INFLATION HEAT
          </div>
          <svg width="200" height="450" viewBox="0 0 200 450">
            {/* Thermometer outer */}
            <rect x="80" y="30" width="40" height="340" rx="20" fill="none" stroke="#262626" strokeWidth="6" />
            <circle cx="100" cy="400" r="40" fill="#5C2E2E" stroke="#262626" strokeWidth="6" />
            {/* Fill */}
            <rect x="85" y={370 - thermFill * 340} width="30" height={thermFill * 340} fill="#5C2E2E" />
            <circle cx="100" cy="400" r="35" fill="#5C2E2E" />
            {/* Markings */}
            {[0, 0.25, 0.5, 0.75, 1].map((m, i) => (
              <line key={i} x1="125" y1={370 - m * 340} x2="140" y2={370 - m * 340}
                stroke="#262626" strokeWidth="3" />
            ))}
            <text x="150" y="375" fontSize="20" fill="#262626" fontFamily="Inter" fontWeight="700">25°</text>
            <text x="150" y="205" fontSize="20" fill="#262626" fontFamily="Inter" fontWeight="700">60°</text>
            <text x="150" y="40" fontSize="20" fill="#262626" fontFamily="Inter" fontWeight="700">100°</text>
          </svg>
          <div style={{
            fontFamily: 'Yantra One', color: '#262626',
            fontSize: 64, fontWeight: 700, marginTop: 10,
          }}>
            {Math.round(tempVal)}°
          </div>
          <div style={{
            fontFamily: 'Noto Sans Devanagari', color: '#5C2E2E',
            fontSize: 30, fontWeight: 600, marginTop: 5,
          }}>
            पता ही नहीं चलता
          </div>
        </div>
      </AbsoluteFill>

      {/* Warning at bottom */}
      <div style={{
        position: 'absolute', bottom: 70, left: 0, right: 0,
        textAlign: 'center',
        opacity: warnIn, transform: `scale(${warnIn})`,
      }}>
        <div style={{
          display: 'inline-block',
          backgroundColor: '#262626', color: '#F5F5F5',
          padding: '16px 40px', borderRadius: 12,
          fontFamily: 'Yantra One', fontSize: 45, fontWeight: 700,
        }}>
          तब तक  <span style={{ color: '#ff0000' }}>बहुत देर</span>  हो चुकी होती है
        </div>
      </div>

      {/* Vignette */}
      <AbsoluteFill style={{
        background: 'radial-gradient(circle, transparent 50%, rgba(0,0,0,0.4) 100%)',
        pointerEvents: 'none',
      }} />

      {/* Watermark */}
      <div style={{
        position: 'absolute', bottom: 30, right: 45,
        fontFamily: 'Inter', fontSize: 22, fontWeight: 600,
        color: '#262626', opacity: 0.4,
      }}>ProBuffet HQ</div>
    </AbsoluteFill>
  );
};
