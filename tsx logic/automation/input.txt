import React from 'react';

import {
  AbsoluteFill,
  Easing,
  interpolate,
  spring,
  useCurrentFrame,
  useVideoConfig,
} from 'remotion';

export const compositionConfig = {
  id: 'Scene01VaultPause',
  durationInSeconds: 30,
  fps: 30,
  width: 1920,
  height: 1080,
};

const C = {
  bg: '#C3FF99',
  ink: '#000000',
  paper: '#F4F2F2',
  moss: '#8FD65A',
  deep: '#2C3B1D',
  pale: '#E4FFC7',
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Particles: React.FC<{frame: number}> = ({frame}) => (
  <AbsoluteFill style={{pointerEvents: 'none'}}>
    {new Array(26).fill(0).map((_, i) => {
      const x = seededRandom(i + 1) * 1920;
      const drift = ((frame * (0.4 + seededRandom(i + 9) * 0.9)) % 1200) - 100;
      const y = 1080 - drift;
      const s = 4 + seededRandom(i + 21) * 9;
      return (
        <div
          key={i}
          style={{
            position: 'absolute',
            left: x,
            top: y,
            width: s,
            height: s,
            borderRadius: i % 3 === 0 ? 0 : s,
            background: i % 4 === 0 ? C.ink : C.paper,
            opacity: 0.22,
            transform: `rotate(${frame * 0.6 + i * 20}deg)`,
          }}
        />
      );
    })}
  </AbsoluteFill>
);

const Vignette: React.FC = () => (
  <AbsoluteFill
    style={{
      background:
        'radial-gradient(circle at 50% 48%, rgba(0,0,0,0) 45%, rgba(0,0,0,0.38) 100%)',
      pointerEvents: 'none',
    }}
  />
);

const Watermark: React.FC<{opacity: number}> = ({opacity}) => (
  <div
    style={{
      position: 'absolute',
      bottom: 30,
      right: 45,
      fontFamily: 'Inter, Poppins, sans-serif',
      fontSize: 22,
      fontWeight: 600,
      color: C.deep,
      opacity,
      letterSpacing: 1,
    }}
  >
    ProBuffet HQ
  </div>
);

const Scene01VaultPause: React.FC = () => {
  const frame = useCurrentFrame();
  const {fps, durationInFrames} = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({
      frame: Math.max(0, frame - delay),
      fps,
      config: {damping: 12, mass: 0.4, stiffness: 120, ...config},
    });

  const exitOp = interpolate(
    frame,
    [durationInFrames - 15, durationInFrames],
    [1, 0],
    {extrapolateLeft: 'clamp', extrapolateRight: 'clamp'}
  );

  // Camera: slow push-in with a small tilt
  const camScale = interpolate(frame, [0, durationInFrames], [1.06, 1.14], {
    extrapolateRight: 'clamp',
    easing: Easing.inOut(Easing.ease),
  });
  const camX = interpolate(frame, [0, durationInFrames], [22, -22], {
    extrapolateRight: 'clamp',
  });

  const vault = sp(18, {damping: 14, stiffness: 90});
  const dial = interpolate(frame, [18, 150], [0, 320], {
    extrapolateRight: 'clamp',
    easing: Easing.out(Easing.cubic),
  });
  const line1 = sp(30);
  const line2 = sp(52);
  const line3 = sp(78);
  const barIn = sp(6, {damping: 16, stiffness: 70});

  const shake = frame > 100 && frame < 118 ? Math.sin(frame * 2.4) * 6 : 0;
  const sweep = ((frame * 7) % 2400) - 400;

            left: 1180,
            top: -180,

            borderRadius: 760,
            background: C.ink,
            opacity: 0.05,
            transform: `scale(${1 + Math.sin(frame * 0.02) * 0.05}) rotate(${frame * 0.06}deg)`,

        />
        <div

            position: 'absolute',

            <g transform={`rotate(${dial} 100 100)`}>
              <circle cx="100" cy="100" r="56" fill="none" stroke={C.ink} strokeWidth="8" />
              <line x1="100" y1="44" x2="100" y2="156" stroke={C.ink} strokeWidth="8" />
              <line x1="44" y1="100" x2="156" y2="100" stroke={C.ink} strokeWidth="8" />
              <circle cx="100" cy="100" r="14" fill={C.moss} stroke={C.ink} strokeWidth="6" />
