import React from 'react';

import { useCurrentFrame, useVideoConfig, interpolate, Easing, spring, AbsoluteFill } from 'remotion';

export const compositionConfig = {
  id: 'Scene01_KitchenBurden',
  durationInSeconds: 27,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene01_KitchenBurden: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  const gradShift = interpolate(frame, [0, durationInFrames], [0, 30]);
  const cameraZoom = interpolate(frame, [0, durationInFrames], [1, 1.08]);

  const title1Op = sp(20);
  const title2Op = sp(50);
  const title3Op = sp(85);
  const lineExpand = interpolate(sp(110), [0, 1], [0, 100]);
  const subOp = sp(140);

  const breath = Math.sin(frame * 0.04) * 4;

  return (
    <AbsoluteFill style={{ backgroundColor: '#635985', overflow: 'hidden', opacity: exitOp }}>
      {/* Dynamic Gradient Background */}
      <AbsoluteFill
        style={{
          background: `radial-gradient(circle at ${30 + gradShift}% ${40 + gradShift / 2}%, #7B6FA8 0%, #635985 45%, #443C5C 100%)`,
        }}
      />

      {/* Shape Layer — Large drifting circles */}
      {[0, 1, 2, 3].map((i) => {
        const seed = seededRandom(i + 1);
        const x = 200 + seed * 1500 + Math.sin(frame * 0.008 + i) * 60;
        const y = 150 + seededRandom(i + 7) * 700 + Math.cos(frame * 0.01 + i) * 40;
        const size = 320 + seed * 280;
        const rot = frame * 0.15 + i * 45;
        return (
          <div
            key={`shape-${i}`}
            style={{
              position: 'absolute',
              left: x,
              top: y,
              width: size,
              height: size,
              borderRadius: i % 2 === 0 ? '50%' : '20%',
              border: `2px solid #B9F3FC`,
              opacity: 0.06,
              transform: `rotate(${rot}deg) scale(${1 + Math.sin(frame * 0.02 + i) * 0.05})`,
            }}
          />
        );
      })}

      {/* Motion Texture — vertical grid lines */}
      <AbsoluteFill style={{ opacity: 0.05 }}>
        {Array.from({ length: 20 }).map((_, i) => (
          <div
            key={`grid-${i}`}
            style={{
              position: 'absolute',
              left: `${i * 5}%`,
              top: 0,
              width: 1,
              height: '100%',
              backgroundColor: '#B9F3FC',
            }}
          />
        ))}
      </AbsoluteFill>

      {/* Floating Particles */}
      {Array.from({ length: 35 }).map((_, i) => {
        const seed = seededRandom(i + 50);
        const x = seed * 1920;
        const y = (seededRandom(i + 200) * 1080 + frame * 0.5) % 1080;
        const size = 2 + seededRandom(i + 300) * 4;
        return (
          <div
            key={`p-${i}`}
            style={{
              position: 'absolute',
              left: x,
              top: y,
              width: size,
              height: size,
              borderRadius: '50%',
              backgroundColor: '#FE7F2D',
              opacity: 0.3 + seededRandom(i + 400) * 0.4,
            }}
          />
        );
      })}

      {/* Main Camera Container */}
      <AbsoluteFill
        style={{
          transform: `scale(${cameraZoom}) translateY(${breath}px)`,
          display: 'flex',
          flexDirection: 'column',
          alignItems: 'center',
          justifyContent: 'center',
        }}
      >
        {/* Top Tag */}
        <div
          style={{
            opacity: title1Op,
            transform: `translateY(${(1 - title1Op) * 30}px)`,
            backgroundColor: '#FE7F2D',
            color: '#1C1C1C',
            fontFamily: 'Inter, Poppins',
            fontSize: 36,
            fontWeight: 800,
            padding: '14px 36px',
            borderRadius: 4,
            letterSpacing: 4,
            marginBottom: 50,
          }}
        >
          EPISODE 01 — रसोई का सच
        </div>

        {/* SVG Kitchen Bowl Icon */}
        <div style={{ opacity: title2Op, transform: `scale(${title2Op})`, marginBottom: 40 }}>
          <svg width="180" height="180" viewBox="0 0 200 200">
            <defs>
              <linearGradient id="bowlGrad" x1="0%" y1="0%" x2="100%" y2="100%">
                <stop offset="0%" stopColor="#B9F3FC" />
                <stop offset="100%" stopColor="#FE7F2D" />
              </linearGradient>
            </defs>
            {/* Steam */}
            <path
              d={`M70 ${50 + Math.sin(frame * 0.1) * 5} Q80 ${30 + Math.sin(frame * 0.1) * 8} 75 10`}
              stroke="#B9F3FC"
              strokeWidth="3"
              fill="none"
              opacity="0.6"
            />
            <path

            fontFamily: 'Yantra One',
            fontSize: 187,

        >
          रसोई का बोझ

        {/* Expanding Line */}
        <div
          style={{
            width: `${lineExpand}%`,

            height: 6,
            backgroundColor: '#FE7F2D',
            marginTop: 30,
            marginBottom: 30,
            boxShadow: '0 0 20px #FE7F2D',
