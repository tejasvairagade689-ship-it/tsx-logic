```tsx
import React from 'react';
import { useCurrentFrame, useVideoConfig, interpolate, Easing, spring, AbsoluteFill } from 'remotion';

export const compositionConfig = {
  id: 'Scene03_NotCoupons',
  durationInSeconds: 8,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene03_NotCoupons: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // "REAL MONEY" entrance
  const realOp = interpolate(frame, [10, 30], [0, 1], { extrapolateRight: 'clamp' });
  const realScale = sp(10, { damping: 8, stiffness: 110 });

  // Coupon entrance
  const couponOp = interpolate(frame, [60, 80], [0, 1], { extrapolateRight: 'clamp' });
  const couponRot = interpolate(sp(60), [0, 1], [-30, -8]);

  // Soup can entrance
  const soupOp = interpolate(frame, [90, 110], [0, 1], { extrapolateRight: 'clamp' });
  const soupY = interpolate(sp(90), [0, 1], [200, 0]);

  // Red X over coupon area at frame 140
  const xScale = sp(140, { damping: 8, stiffness: 130 });
  const xOp = interpolate(frame, [140, 160], [0, 1], { extrapolateRight: 'clamp' });

  // Figure
  const figOp = interpolate(frame, [30, 60], [0, 1], { extrapolateRight: 'clamp' });
  const figY = Math.sin(frame * 0.05) * 2;
  const figHeadRot = Math.sin(frame * 0.04) * 3;

  // Camera tilt
  const cameraRot = interpolate(frame, [0, durationInFrames], [-0.5, 0.5], { extrapolateRight: 'clamp' });

  return (
    <AbsoluteFill style={{ backgroundColor: '#1C1C1C', overflow: 'hidden', opacity: exitOp }}>
      {/* Gradient background */}
      <AbsoluteFill
        style={{
          background: 'linear-gradient(180deg, #1C1C1C 0%, #2D2D2D 100%)',
        }}
      />

      {/* Shape Layer */}
      <div
        style={{
          position: 'absolute',
          top: '15%',
          left: '10%',
          width: 350,
          height: 350,
          background: '#ff0000',
          opacity: 0.04,
          borderRadius: 40,
          transform: `rotate(${frame * 0.08}deg)`,
        }}
      />
      <div
        style={{
          position: 'absolute',
          top: '50%',
          right: '8%',
          width: 420,
          height: 420,
          borderRadius: '50%',
          background: '#e9d700',
          opacity: 0.05,
          transform: `scale(${1 + Math.sin(frame * 0.03) * 0.08})`,
        }}
      />
      <div
        style={{
          position: 'absolute',
          bottom: '10%',
          left: '20%',
          width: 300,
          height: 300,
          border: '4px solid #F5F5F5',
          opacity: 0.04,
          transform: `rotate(${frame * 0.12}deg)`,
        }}
      />

      {/* Motion Texture - diagonal lines */}
      <svg style={{ position: 'absolute', inset: 0, width: '100%', height: '100%', opacity: 0.05 }}>
        {Array.from({ length: 25 }).map((_, i) => (
          <line
            key={i}
            x1={-200 + i * 100}
            y1={0}
            x2={i * 100}
            y2={1080}
            stroke="#e9d700"
            strokeWidth={1}
          />
        ))}
      </svg>

      {/* Particles */}
      {Array.from({ length: 22 }).map((_, i) => {
        const seed = seededRandom(i + 5);
        const x = (seed * 1920 + Math.sin(frame * 0.02 + i) * 30) % 1920;
        const y = (seededRandom(i + 70) * 1080 + frame * 0.6) % 1080;
        return (
          <div
            key={i}
            style={{
              position: 'absolute',
              left: x,
              top: y,
              width: 3,
              height: 3,
              borderRadius: '50%',
              background: '#FFC107',
              opacity: 0.5,
            }}
          />
        );
      })}

      {/* Main Content */}
      <div
        style={{
          position: 'absolute',
          inset: 0,
          display: 'flex',
          flexDirection: 'column',
          alignItems: 'center',
          paddingTop: 120,
          transform: `rotate(${cameraRot}deg)`,
        }}
      >
        {/* Hero text */}
        <div
          style={{
            fontFamily: 'Inter, Poppins, sans-serif',
            fontSize: 180,
            fontWeight: 900,
            color: '#F5F5F5',
            opacity: realOp,
            transform: `scale(${realScale})`,
            letterSpacing: -3,
            lineHeight: 1,
          }}
        >
          REAL <span style={{ color: '#e9d700' }}>MONEY.</span>
        </div>

        <div
          style={{
            fontFamily: 'Inter, sans-serif',
            fontSize: 44,
            fontWeight: 600,
            color: '#9E9E9E',
            opacity: interpolate(frame, [40, 60], [0, 1], { extrapolateRight: 'clamp' }),
            marginTop: 20,
            letterSpacing: 2,
            textTransform: 'uppercase',
          }}
        >
          Not Couponing Pennies
        </div>

        {/* Coupon + Soup Row */}
        <div
          style={{
            display: 'flex',
            gap: 100,
            marginTop: 80,
            alignItems: 'center',
            position: 'relative',
          }}
        >
          {/* Coupon */}
          <div
            style={{
              opacity: couponOp,
              transform: `rotate(${couponRot}deg)`,
              background: '#e9d700',
              padding: '30px 50px',
              border: '4px dashed #2D2D2D',
              borderRadius: 8,
              position: 'relative',
            }}
          >
            <div style={{ fontFamily: 'Inter', fontSize: 24, fontWeight: 700, color: '#2D2D2D', letterSpacing: 2 }}>
              COUPON
            </div>
            <div style={{ fontFamily: 'Inter', fontSize: 80, fontWeight: 900, color: '#1C1C1C', lineHeight: 1, marginTop: 8 }}>
              50¢
            </div>
            <div style={{ fontFamily: 'Inter', fontSize: 18, fontWeight: 600, color: '#2D2D2D', marginTop: 6 }}>
              OFF ANY ITEM
            </div>
          </div>

          {/* Soup Can */}
          <div
            style={{
              opacity: soupOp,
              transform: `translateY(${soupY}px)`,
            }}
          >
            <svg width={180} height={240} viewBox="0 0 180 240">
              <ellipse cx={90} cy={30} rx={70} ry={15} fill="#9E9E9E" />
              <rect x={20} y={30} width={140} height={180} fill="#ff0000" />
              <ellipse cx={90} cy={210} rx={70} ry={15} fill="#2D2D2D" />
              <rect x={20} y={90} width={140} height={60} fill="#F5F5F5" />
              <text x={90} y={125} fontFamily="Inter" fontSize={28} fontWeight={900} fill="#ff0000" textAnchor="middle">
                SOUP
              </text>
              <text x={90} y={175} fontFamily="Inter" fontSize={14} fontWeight={600} fill="#F5F5F5" textAnchor="middle">
                CONDENSED
              </text>
            </svg>
          </div>

          {/* Red X over both */}
          <div
            style={{
              position: 'absolute',
              top: '50%',
              left: '50%',
              transform: `translate(-50%, -50%) scale(${xScale})`,
              opacity: xOp,
            }}
          >
            <svg width={520} height={300} viewBox="0 0 520 300">
              <line x1={40} y1={40} x2={480} y2={260} stroke="#ff0000" strokeWidth={20} strokeLinecap="round" />
              <line x1={480} y1={40} x2={40} y2={260} stroke="#ff0000" strokeWidth={20} strokeLinecap="round" />
            </svg>
          </div>
        </div>
      </div>

      {/* Stick Figure */}
      <div
        style={{
          position: 'absolute',
          bottom: 70,
          left: 80,
          opacity: figOp,
          transform: `translateY(${figY}px)`,
        }}
      >
        <div
          style={{
            position: 'absolute',
            top: 20,
            left: -20,
            width: 140,
            height: 140,
            borderRadius: '50%',
            background: '#FFC107',
            filter: 'blur(40px)',
            opacity: 0.3,
          }}
        />
        <svg width={120} height={220} viewBox="0 0 120 220">
          <g transform={`rotate(${figHeadRot} 60 50)`}>
            <circle cx={60} cy={50} r={28} fill="none" stroke="#F5F5F5" strokeWidth={4} />
            <circle cx={50} cy={48} r={9} fill="none" stroke="#FFC107" strokeWidth={3} />
            <circle cx={70} cy={48} r={9} fill="none" stroke="#FFC107" strokeWidth={3} />
            <line x1={59} y1={48} x2={61} y2={48} stroke="#FFC107" strokeWidth={3} />
            {/* Serious mouth */}
            <line x1={50} y1={65} x2={70} y2={65} stroke="#F5F5F5" strokeWidth={3} strokeLinecap="round" />
          </g>
          <line x1={60} y1={80} x2={60} y2={150} stroke="#F5F5F5" strokeWidth={4} />
          <line x1={60} y1={100} x2={30} y2={130} stroke="#F5F5F5" strokeWidth={4} strokeLinecap="round" />
          <line x1={60} y1={100} x2={90} y2={130} stroke="#F5F5F5" strokeWidth={4} strokeLinecap="round" />
          <line x1={60} y1={150} x2={40} y2={200} stroke="#F5F5F5" strokeWidth={4} strokeLinecap="round" />
          <line x1={60} y1={150} x2={80} y2={200} stroke="#F5F5F5" strokeWidth={4} strokeLinecap="round" />
        </svg>
      </div>

      {/* Vignette */}
      <AbsoluteFill
        style={{
          background: 'radial-gradient(circle at center, transparent 40%, rgba(0,0,0,0.6) 100%)',
          pointerEvents: 'none',
        }}
      />

      {/* Watermark */}
      <div
        style={{
          position: 'absolute',
          bottom: 30,
          right: 45,
          fontFamily: 'Inter, sans-serif',
          fontSize: 22,
          fontWeight: 600,
          color: '#9E9E9E',
          opacity: 0.3,
          letterSpacing: 1,
        }}
      >
        Senior Fund USA
      </div>
    </AbsoluteFill>
  );
};

export default Scene03_NotCoupons;
```
