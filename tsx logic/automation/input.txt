import React from 'react';

import { useCurrentFrame, useVideoConfig, interpolate, Easing, spring, AbsoluteFill } from 'remotion';

export const compositionConfig = {
  id: 'Scene01',
  durationInSeconds: 14,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene01: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // slow camera zoom
  const camZoom = interpolate(frame, [0, durationInFrames], [1.05, 1.12], { extrapolateRight: 'clamp' });
  const camPan = interpolate(frame, [0, durationInFrames], [-20, 20], { extrapolateRight: 'clamp' });

  const titleY = interpolate(sp(15, { stiffness: 90 }), [0, 1], [80, 0]);
  const titleOp = sp(15);
  const subOp = sp(45);
  const subY = interpolate(sp(45), [0, 1], [40, 0]);

  // steam rising animation
  const steam = (i: number) => {
    const s = seededRandom(i + 3);
    const rise = (frame * (1.5 + s) + i * 60) % 360;
    const op = interpolate(rise, [0, 120, 360], [0, 0.35, 0], { extrapolateRight: 'clamp' });
    const sway = Math.sin(frame * 0.05 + i) * 18;
    return { rise, op, sway };
  };

  const lineGrow = interpolate(sp(30, { stiffness: 70 }), [0, 1], [0, 1]);

  return (
    <AbsoluteFill style={{ backgroundColor: '#CDE990', overflow: 'hidden', opacity: exitOp }}>
      {/* Gradient overlay */}
      <AbsoluteFill style={{ background: 'radial-gradient(circle at 30% 30%, rgba(255,255,255,0.45), transparent 60%)' }} />

      {/* Shape Layers — drifting transparent geometry */}
      <AbsoluteFill style={{ transform: `scale(${camZoom}) translateX(${camPan}px)` }}>
        <div style={{
          position: 'absolute', top: '12%', left: '8%', width: 520, height: 520, borderRadius: '50%',
          border: '3px solid #000000', opacity: 0.05,
          transform: `rotate(${frame * 0.15}deg) scale(${1 + Math.sin(frame * 0.02) * 0.05})`,
        }} />
        <div style={{
          position: 'absolute', bottom: '6%', right: '10%', width: 420, height: 420,
          background: '#F62440', opacity: 0.04,
          transform: `rotate(${-frame * 0.12}deg)`,
        }} />
        <div style={{
          position: 'absolute', top: '40%', right: '22%', width: 0, height: 0,
          borderLeft: '160px solid transparent', borderRight: '160px solid transparent',
          borderBottom: '280px solid #000000', opacity: 0.04,
          transform: `rotate(${frame * 0.1}deg)`,
        }} />
      </AbsoluteFill>

      {/* Motion texture — grid lines */}
      <AbsoluteFill style={{ opacity: 0.06 }}>
        {Array.from({ length: 14 }).map((_, i) => (
          <div key={i} style={{
            position: 'absolute', left: `${(i / 14) * 100}%`, top: 0, bottom: 0, width: 1,
            background: '#000000', transform: `translateY(${Math.sin(frame * 0.03 + i) * 10}px)`,
          }} />
        ))}
      </AbsoluteFill>

      {/* Floating particles */}
      {Array.from({ length: 18 }).map((_, i) => {
        const s = seededRandom(i + 1);
        const px = (s * 1920 + frame * (0.3 + s)) % 1920;
        const py = (seededRandom(i + 20) * 1080 + Math.sin(frame * 0.02 + i) * 30);
        return (
          <div key={i} style={{
            position: 'absolute', left: px, top: py, width: 6 + s * 8, height: 6 + s * 8,
            borderRadius: '50%', background: i % 3 === 0 ? '#F62440' : '#000000', opacity: 0.18,
          }} />
        );
      })}

      {/* Tea cup illustration with steam */}
      <div style={{ position: 'absolute', left: '50%', top: '30%', transform: `translateX(-50%) scale(${sp(60)})` }}>
        {[0, 1, 2, 3].map((i) => {
          const st = steam(i);
          return (
            <div key={i} style={{
              position: 'absolute', left: 30 + i * 22, top: -st.rise * 0.6,
              width: 8, height: 60, borderRadius: 8, background: '#000000',
              opacity: st.op, transform: `translateX(${st.sway}px)`, filter: 'blur(2px)',
            }} />
          );
        })}
        <svg width="140" height="120" viewBox="0 0 140 120">
          <path d="M20 50 Q20 100 70 100 Q120 100 120 50 Z" fill="#F62440" stroke="#000000" strokeWidth="5" />
          <ellipse cx="70" cy="50" rx="50" ry="12" fill="#CDE990" stroke="#000000" strokeWidth="5" />
          <path d="M120 60 Q150 60 150 80 Q150 95 125 90" fill="none" stroke="#000000" strokeWidth="5" />
        </svg>
      </div>

      {/* Main content */}
      <AbsoluteFill style={{ justifyContent: 'center', alignItems: 'center', flexDirection: 'column', paddingTop: 120 }}>
        <div style={{
          fontFamily: "'Yantra One', sans-serif", fontSize: 187, lineHeight: 1.0, color: '#000000',
          fontWeight: 700, textAlign: 'center', opacity: titleOp, transform: `translateY(${titleY}px)`,
          textShadow: '6px 6px 0 #F62440',
        }}>
          एक शाम
        </div>

        <div style={{ width: lineGrow * 600, height: 8, background: '#F62440', marginTop: 20, borderRadius: 4 }} />

        <div style={{
          fontFamily: "'Noto Sans Devanagari', sans-serif", fontSize: 51, color: '#000000',
          fontWeight: 600, textAlign: 'center', marginTop: 30, opacity: subOp,
          transform: `translateY(${subY}px)`, maxWidth: 1200,
        }}>
          पुराने मुहल्ले की चाय की टपरी पर
        </div>
      </AbsoluteFill>

      {/* Vignette */}
      <AbsoluteFill style={{ boxShadow: 'inset 0 0 350px rgba(0,0,0,0.4)', pointerEvents: 'none' }} />

      {/* Watermark */}
      <div style={{
        position: 'absolute', bottom: 30, right: 45, fontFamily: 'Inter, sans-serif',
        fontSize: 22, fontWeight: 600, color: '#000000', opacity: 0.4,
      }}>
        ProBuffet HQ
      </div>
    </AbsoluteFill>
  );
};

export default Scene01;

        {/* kitchen inflation — steep line */}
        <line x1="400" y1="540" x2={400 + lineProg * 1100} y2={kitchenEndY}
          stroke="#F62440" strokeWidth="14" strokeLinecap="round" opacity={rightOp} />
        {/* gap shading */}
        {lineProg > 0.9 && (
          <polygon
            points={`${400 + 1100} ${reportEndY}, ${400 + 1100} ${kitchenEndY}, ${400} 540`}
            fill="#F62440" opacity={0.12 * gapOp}
          />
        )}
      </svg>

      {/* Labels */}
      <div style={{
        position: 'absolute', top: reportEndY - 70, left: 1450,
        fontFamily: "'Noto Sans Devanagari', sans-serif", fontSize: 40, fontWeight: 700,
        color: '#000000', opacity: leftOp, background: '#CDE990', padding: '8px 18px', borderRadius: 10,
        border: '3px solid #000000',
      }}>
        रिपोर्ट की महंगाई
      </div>
      <div style={{
        position: 'absolute', top: kitchenEndY - 20, left: 1450,
        fontFamily: "'Noto Sans Devanagari', sans-serif", fontSize: 40, fontWeight: 700,
        color: '#CDE990', opacity: rightOp, background: '#F62440', padding: '8px 18px', borderRadius: 10,
        border: '3px solid #000000',
      }}>
        रसोई की महंगाई
      </div>

      {/* Central headline */}
      <div style={{
        position: 'absolute', top: 90, left: 0, right: 0, textAlign: 'center',
        fontFamily: "'Yantra One', sans-serif", fontSize: 119, color: '#000000', fontWeight: 700,
        opacity: gapOp, transform: `scale(${gapScale})`, textShadow: '6px 6px 0 #F62440',
      }}>
        एक चौड़ी खाई
      </div>

      {/* Particles */}
      {Array.from({ length: 14 }).map((_, i) => {
        const s = seededRandom(i + 21);
        const py = (s * 1080 + frame * (0.4 + s)) % 1080;
        return (
          <div key={i} style={{
            position: 'absolute', left: seededRandom(i + 100) * 1920, top: py,
            width: 6, height: 6, borderRadius: '50%', background: '#000000', opacity: 0.16,
          }} />
        );
      })}

      {/* Vignette */}
      <AbsoluteFill style={{ boxShadow: 'inset 0 0 350px rgba(0,0,0,0.4)', pointerEvents: 'none' }} />

      {/* Watermark */}
      <div style={{
        position: 'absolute', bottom: 30, right: 45, fontFamily: 'Inter, sans-serif',
        fontSize: 22, fontWeight: 600, color: '#000000', opacity: 0.4,
      }}>
        ProBuffet HQ
      </div>
    </AbsoluteFill>
  );
};

export default Scene10;
