  fps: 30,
  width: 1920,
  height: 1080,

  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);

  const heroOp = interpolate(frame, [10, 25], [0, 1], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' });

    extrapolateLeft: 'clamp', extrapolateRight: 'clamp', easing: Easing.out(Easing.cubic),
  });

  const slamOp = interpolate(frame, [150, 158], [0, 1], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' });

  const brandOp = interpolate(frame, [420, 440], [0, 1], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' });

  return (
    <AbsoluteFill style={{ backgroundColor: '#F4CE14', overflow: 'hidden', opacity: exitOp }}>
      {/* Gradient Background */}
      <AbsoluteFill style={{
        background: `radial-gradient(circle at ${50 + Math.sin(frame * 0.01) * 10}% 40%, #F4CE14 0%, #E8BE00 60%, #D9AE00 100%)`,
      }} />

      {/* Animated Shape Layers */}
      <div style={{
        position: 'absolute', width: 900, height: 900, borderRadius: '50%',
        border: '3px solid #000000', opacity: 0.05,
        left: -250 + Math.sin(frame * 0.008) * 30, top: -300,
        transform: `scale(${1 + Math.sin(frame * 0.01) * 0.04}) rotate(${frame * 0.05}deg)`,
      }} />
      <div style={{
        position: 'absolute', width: 700, height: 700,
        border: '3px solid #000000', opacity: 0.04,
        right: -200, bottom: -250,
        transform: `rotate(${45 + frame * 0.06}deg)`,
      }} />
