  id: 'Scene01_PumpShock',
  durationInSeconds: 14,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // Counter animation 0 -> 73.41
  const counterProgress = interpolate(frame, [25, 95], [0, 73.41], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp', easing: Easing.out(Easing.cubic),
  });
  const counterValue = counterProgress.toFixed(2);

  // Impact shake when counter locks
  const shakeFrame = frame - 95;
  const shakeX = shakeFrame > 0 && shakeFrame < 20 ? Math.sin(shakeFrame * 2) * (20 - shakeFrame) * 0.8 : 0;
  const shakeY = shakeFrame > 0 && shakeFrame < 20 ? Math.cos(shakeFrame * 2.3) * (20 - shakeFrame) * 0.6 : 0;

  // Camera zoom
  const cameraZoom = interpolate(frame, [0, durationInFrames], [1.0, 1.08], { extrapolateRight: 'clamp' });

  // Headline reveal
  const headlineOp = interpolate(frame, [110, 140], [0, 1], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' });
  const headlineY = interpolate(sp(110, { damping: 14 }), [0, 1], [40, 0]);

  const subOp = interpolate(frame, [150, 180], [0, 1], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' });

  const receiptRot = interpolate(sp(60, { damping: 16, stiffness: 90 }), [0, 1], [-12, -4]);

  const nozzleY = interpolate(sp(15, { damping: 14 }), [0, 1], [-300, 0]);

        top: -200 + Math.sin(frame * 0.015) * 30,
        left: -200 + Math.cos(frame * 0.012) * 40,

      }} />
      <div style={{
        position: 'absolute', width: 600, height: 600,
        border: '2px solid #EEEEEE', opacity: 0.05,
        bottom: -150 + Math.cos(frame * 0.018) * 25,
        right: -100 + Math.sin(frame * 0.014) * 35,
        transform: `rotate(${-frame * 0.15}deg)`,
      }} />
      <div style={{
        position: 'absolute', width: 0, height: 0,
        borderLeft: '300px solid transparent',
        borderRight: '300px solid transparent',
        borderBottom: '500px solid #FF5722',
        opacity: 0.04,
        top: 100, right: 200,
        transform: `rotate(${15 + Math.sin(frame * 0.02) * 5}deg)`,
      }} />

      {/* Motion texture - vertical grid lines */}
      <svg style={{ position: 'absolute', inset: 0, width: '100%', height: '100%', opacity: 0.08 }}>
        {Array.from({ length: 20 }).map((_, i) => (
          <line key={i}
            x1={i * 96 + (frame * 0.5) % 96} y1={0}
            x2={i * 96 + (frame * 0.5) % 96} y2={1080}
            stroke="#EEEEEE" strokeWidth={1}
          />

        const seed = seededRandom(i + 1);
        const seed2 = seededRandom(i + 50);
        const px = (seed * 1920 + frame * (0.3 + seed2 * 0.5)) % 1920;
        const py = (seed2 * 1080 + Math.sin(frame * 0.02 + i) * 20) % 1080;
        return (

            position: 'absolute', width: 4 + seed * 4, height: 4 + seed * 4,

  const camX = interpolate(fuseProgress, [0, 1], [120, -120], { extrapolateRight: 'clamp' });
  const camScale = interpolate(frame, [0, durationInFrames], [1.02, 1.06], { extrapolateRight: 'clamp' });

  // Headline
  const headlineOp = interpolate(frame, [200, 240], [0, 1], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' });

  return (
    <AbsoluteFill style={{ backgroundColor: '#A1BC98', overflow: 'hidden', opacity: exitOp }}>
      <AbsoluteFill style={{
        background: 'linear-gradient(180deg, #8FAE85 0%, #A1BC98 50%, #93B188 100%)',
      }} />

      {/* Drifting shapes */}
      <div style={{
        position: 'absolute', width: 900, height: 900, borderRadius: '50%',
        backgroundColor: '#FF5722', opacity: 0.05,
        top: -300, left: -200,
        transform: `scale(${1 + Math.sin(frame * 0.025) * 0.06})`,
      }} />
      <div style={{
        position: 'absolute', width: 600, height: 600,
        border: '4px solid #EEEEEE', opacity: 0.06,
        bottom: -200, right: -200, borderRadius: 60,
        transform: `rotate(${frame * 0.15}deg)`,
      }} />

      {/* Motion texture - radiating from flame */}
      <svg style={{ position: 'absolute', inset: 0, opacity: 0.08 }} width="1920" height="1080">
        {Array.from({ length: 18 }).map((_, i) => {
          const angle = (i * 20) + frame * 0.5;
          const rad = (angle * Math.PI) / 180;
          return (
            <line key={i}
              x1={flameX} y1={fuseY}
              x2={flameX + Math.cos(rad) * 400}
              y2={fuseY + Math.sin(rad) * 400}
              stroke="#FF5722" strokeWidth={1}
            />
          );
        })}
      </svg>

      {/* Spark particles around flame */}
      {Array.from({ length: 30 }).map((_, i) => {
        const seed = seededRandom(i + 90);
        const seed2 = seededRandom(i + 900);
        const angle = seed * Math.PI * 2;
        const dist = 30 + seed2 * 80;
        const phase = (frame * 0.5 + i * 8) % 60;
        const px = flameX + Math.cos(angle + frame * 0.05) * (dist + phase);
        const py = fuseY + Math.sin(angle) * (dist + phase) - phase * 1.5;
        return (
          <div key={i} style={{
            position: 'absolute', width: 3 + seed * 4, height: 3 + seed * 4,
            borderRadius: '50%', backgroundColor: phase < 30 ? '#FF5722' : '#EEEEEE',
            opacity: 1 - phase / 60,
            left: px, top: py,
            filter: 'blur(0.5px)',
          }} />
        );
      })}

      <AbsoluteFill style={{ transform: `translateX(${camX}px) scale(${camScale})` }}>

        {/* LEFT - Persian Gulf map area */}
        <div style={{
          position: 'absolute', left: 100, top: 320,
          opacity: mapOp, transform: `scale(${mapScale})`,
          transformOrigin: 'left center',
        }}>
          {/* Simplified map */}
          <svg width="320" height="320" viewBox="0 0 200 200">
            {/* Water */}
            <rect width="200" height="200" fill="#1C1C1C" rx="12" />
            <rect width="200" height="200" fill="#2D2D2D" opacity="0.5" rx="12" />
            {/* Land masses (simplified) */}
            <path d="M 0 0 L 80 0 L 90 40 L 70 70 L 30 90 L 0 100 Z" fill="#A1BC98" opacity="0.85" />
            <path d="M 120 130 L 200 110 L 200 200 L 130 200 L 100 170 Z" fill="#A1BC98" opacity="0.85" />
            {/* Strait of Hormuz marker */}
            <circle cx="105" cy="100" r="10" fill="none" stroke="#FF5722" strokeWidth="2"
              opacity={0.4 + Math.sin(frame * 0.2) * 0.4} />
            <circle cx="105" cy="100" r="4" fill="#FF5722" />
            {/* Crosshatch over contested zone */}
            <rect x="85" y="85" width="40" height="30" fill="none" stroke="#FF5722" strokeWidth="1" strokeDasharray="3" />
          </svg>
          <div style={{
            marginTop: 16,
            fontFamily: 'JetBrains Mono', fontSize: 22, color: '#1C1C1C',
            fontWeight: 700, letterSpacing: 3,
          }}>
            PERSIAN GULF
          </div>
          <div style={{
            fontFamily: 'JetBrains Mono', fontSize: 18, color: '#FF5722',
            fontWeight: 700, letterSpacing: 2, marginTop: 4,
          }}>
            ● CONTESTED WATERS
          </div>
        </div>

        {/* FUSE LINE - center horizontal */}
        <svg style={{ position: 'absolute', inset: 0 }} width="1920" height="1080">
          {/* Fuse rope background */}
          <line x1={fuseStartX} y1={fuseY} x2={fuseEndX} y2={fuseY}
            stroke="#2D2D2D" strokeWidth="14" strokeLinecap="round" />
          {/* Burnt section */}
          <line x1={fuseStartX} y1={fuseY} x2={flameX} y2={fuseY}
            stroke="#1C1C1C" strokeWidth="14" strokeLinecap="round" />
          {/* Unburnt highlight */}
          <line x1={flameX} y1={fuseY} x2={fuseEndX} y2={fuseY}
            stroke="#EEEEEE" strokeWidth="3" strokeLinecap="round" opacity="0.5" />
          {/* Fuse texture - small ticks */}
          {Array.from({ length: 40 }).map((_, i) => {
            const x = fuseStartX + (i / 40) * fuseLength;
            const burnt = x < flameX;
            return (
              <line key={i}
                x1={x} y1={fuseY - 6} x2={x} y2={fuseY + 6}
                stroke={burnt ? '#1C1C1C' : '#9E9E9E'} strokeWidth="2"
              />
            );
          })}
        </svg>

        {/* FLAME at the fuse tip */}
        <div style={{
          position: 'absolute',
          left: flameX, top: fuseY,
          transform: `translate(-50%, -50%) scale(${1 + Math.sin(frame * 0.4) * 0.15})`,
        }}>
          <svg width="120" height="160" viewBox="0 0 60 80">
            <path d="M 30 5 Q 15 25 18 45 Q 20 60 30 75 Q 40 60 42 45 Q 45 25 30 5 Z"
              fill="#FF5722" />
            <path d="M 30 20 Q 22 35 25 50 Q 28 60 30 70 Q 32 60 35 50 Q 38 35 30 20 Z"
              fill="#EEEEEE" opacity="0.7" />
            <circle cx="30" cy="55" r="5" fill="#EEEEEE" />
          </svg>
        </div>

        {/* RIGHT - WALLET getting hit */}
        <div style={{
          position: 'absolute', right: 100, top: 320,
          transform: `translateX(${walletShake}px) translateY(${walletShake * 0.5}px) scale(${1 + walletBurn * 0.05})`,
        }}>
          <svg width="380" height="280" viewBox="0 0 190 140">
            {/* Wallet body */}
            <rect x="10" y="20" width="170" height="110" rx="10" fill="#2D2D2D" stroke="#1C1C1C" strokeWidth="3" />
            <rect x="10" y="20" width="170" height="40" rx="10" fill="#1C1C1C" />
            {/* Card slot */}
            <rect x="20" y="68" width="100" height="18" rx="3" fill="#FF5722" opacity="0.9" />
            <rect x="20" y="90" width="80" height="18" rx="3" fill="#9E9E9E" />
            {/* Coin */}
            <circle cx="150" cy="95" r="20" fill="#FFC107" />
            <text x="150" y="102" textAnchor="middle" fill="#1C1C1C" fontFamily="Nexa Bold" fontSize="20" fontWeight="900">$</text>
            {/* Burn overlay */}
            {walletBurn > 0 && (
              <rect x="10" y="20" width="170" height="110" rx="10"
                fill="#FF5722" opacity={walletBurn * 0.6} />
            )}
          </svg>
          {/* Burn flames over wallet */}
          {walletBurn > 0 && Array.from({ length: 5 }).map((_, i) => (
            <div key={i} style={{
              position: 'absolute',
              left: 60 + i * 60,
              top: -40 - Math.sin(frame * 0.3 + i) * 10,
              opacity: walletBurn,
            }}>
              <svg width="40" height="60" viewBox="0 0 30 40">
                <path d="M 15 2 Q 8 12 10 22 Q 12 32 15 38 Q 18 32 20 22 Q 22 12 15 2 Z" fill="#FF5722" />
              </svg>
            </div>
          ))}
          <div style={{
            marginTop: 16, textAlign: 'center',
            fontFamily: 'JetBrains Mono', fontSize: 22, color: '#1C1C1C',
            fontWeight: 700, letterSpacing: 3,
          }}>
            YOUR WALLET
          </div>
          <div style={{
            fontFamily: 'JetBrains Mono', fontSize: 18, color: '#FF5722',
            fontWeight: 700, letterSpacing: 2, marginTop: 4, textAlign: 'center',
            opacity: walletBurn,
          }}>
            ● IMPACT ZONE
          </div>
        </div>

        {/* Top headline */}
        <div style={{
          position: 'absolute', top: 80, left: 0, right: 0, textAlign: 'center',
        }}>
          <div style={{
            fontFamily: 'JetBrains Mono', fontSize: 26, color: '#FF5722',
            letterSpacing: 6, fontWeight: 700, marginBottom: 8,
            opacity: interpolate(frame, [10, 40], [0, 1], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' }),
          }}>
            A FUSE · A SHIPPING LANE · YOUR WALLET
          </div>
          <div style={{
            fontFamily: 'Mont Heavy, sans-serif', fontSize: 76, fontWeight: 900,
            color: '#1C1C1C', lineHeight: 1, letterSpacing: -1,
            opacity: headlineOp,
          }}>
            The burn travels <span style={{ color: '#FF5722' }}>straight up.</span>
          </div>
        </div>

        {/* Bottom data strip */}
        <div style={{
          position: 'absolute', bottom: 80, left: 120, right: 120,
          borderTop: '2px solid #1C1C1C', paddingTop: 16,
          display: 'flex', justifyContent: 'space-between', alignItems: 'center',
          fontFamily: 'JetBrains Mono', fontSize: 22, color: '#1C1C1C',
          opacity: interpolate(frame, [220, 260], [0, 1], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' }),
        }}>
          <span style={{ letterSpacing: 3 }}>GEOPOLITICS → CRUDE → CPI → YOU</span>
          <span style={{ color: '#FF5722', fontWeight: 700, letterSpacing: 2 }}>● BURN RATE: ACCELERATING</span>
        </div>

      </AbsoluteFill>

      {/* Vignette */}
      <AbsoluteFill style={{
        background: 'radial-gradient(ellipse at center, transparent 40%, rgba(28,28,28,0.45) 100%)',
        pointerEvents: 'none',
      }} />

      {/* Watermark */}
      <div style={{
        position: 'absolute', bottom: 30, right: 45,
        fontFamily: 'Inter, sans-serif', fontSize: 22, fontWeight: 600,
        color: '#9E9E9E', opacity: 0.3,
      }}>
        Failed Finances
      </div>
    </AbsoluteFill>
  );
};

export default Scene10_FuseToWallet;
