import React from 'react';

import { useCurrentFrame, useVideoConfig, interpolate, Easing, spring, AbsoluteFill } from 'remotion';

export const compositionConfig = {
  id: 'Scene01_PictureThis',
  durationInSeconds: 16,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene01_PictureThis: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // slow camera drift
  const camX = Math.sin(frame * 0.012) * 14;
  const camY = Math.cos(frame * 0.01) * 10;
  const camZoom = interpolate(frame, [0, durationInFrames], [1.04, 1.1], { extrapolateRight: 'clamp' });

  // number ticking DOWN (statement dropping)
  const startVal = 184250;
  const endVal = 171930;
  const numProgress = interpolate(frame, [40, 130], [0, 1], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp', easing: Easing.out(Easing.cubic) });
  const liveVal = Math.round(startVal - (startVal - endVal) * numProgress);

  const headIn = sp(20);
  const cardIn = sp(40, { stiffness: 90 });
  const arrowIn = sp(95);
  const subIn = sp(115);

  // tiny shake when number drops near end
  const shake = frame > 120 && frame < 132 ? Math.sin(frame * 3) * (132 - frame) * 0.4 : 0;

  return (
    <AbsoluteFill style={{ backgroundColor: '#F9F6F0', overflow: 'hidden', opacity: exitOp }}>
      {/* Gradient + Shape Layers + Texture + Particles + Content + Character + Vignette + Watermark */}

      {/* Dynamic gradient background */}
      <AbsoluteFill style={{ background: 'radial-gradient(circle at 30% 25%, #FFFDF8 0%, #F9F6F0 45%, #F1ECE2 100%)' }} />

      <AbsoluteFill style={{ transform: `translate(${camX + shake}px, ${camY}px) scale(${camZoom})` }}>
        {/* Animated shape layers (transparent geometric forms moving slowly) */}
        <div style={{ position: 'absolute', top: '12%', left: '8%', width: 560, height: 560, borderRadius: '50%', border: '2px solid #2C2523', opacity: 0.05, transform: `rotate(${frame * 0.2}deg) scale(${1 + Math.sin(frame * 0.03) * 0.05})` }} />
        <div style={{ position: 'absolute', bottom: '8%', right: '6%', width: 460, height: 460, background: '#5A524E', opacity: 0.05, transform: `rotate(${frame * -0.15}deg)`, borderRadius: 40 }} />
        <div style={{ position: 'absolute', top: '55%', left: '60%', width: 0, height: 0, borderLeft: '170px solid transparent', borderRight: '170px solid transparent', borderBottom: '300px solid #2C2523', opacity: 0.04, transform: `rotate(${Math.sin(frame * 0.02) * 12}deg)` }} />

        {/* Motion texture (thin grid lines) */}
        <svg width="1920" height="1080" style={{ position: 'absolute', top: 0, left: 0, opacity: 0.06 }}>
          {Array.from({ length: 16 }).map((_, i) => (
            <line key={`v${i}`} x1={i * 120 + (frame * 0.3) % 120} y1="0" x2={i * 120 + (frame * 0.3) % 120} y2="1080" stroke="#5A524E" strokeWidth="1" />
          ))}
        </svg>

        {/* Floating particles */}
        {Array.from({ length: 22 }).map((_, i) => {
          const r = seededRandom(i + 1);
          const r2 = seededRandom(i + 50);
          const px = r * 1920;
          const py = (r2 * 1080 + frame * (0.4 + r * 0.7)) % 1080;
          return (
            <div key={`p${i}`} style={{ position: 'absolute', left: px, top: py, width: 4 + r * 6, height: 4 + r * 6, borderRadius: '50%', backgroundColor: '#5A524E', opacity: 0.12 + r * 0.1 }} />
          );
        })}

        {/* Main content with spring animations */}
        {/* Headline */}
        <div style={{ position: 'absolute', top: 120, left: 130, opacity: headIn, transform: `translateY(${(1 - headIn) * 40}px)` }}>
          <div style={{ fontFamily: "'Mont Heavy', 'Inter', sans-serif", fontSize: 110, fontWeight: 900, color: '#2C2523', lineHeight: 1 }}>Picture this.</div>
          <div style={{ width: interpolate(headIn, [0, 1], [0, 360]), height: 8, backgroundColor: '#2C2523', marginTop: 18, borderRadius: 4 }} />
        </div>

        {/* Statement card */}
        <div style={{
          position: 'absolute', top: 360, left: '50%',
          transform: `translateX(-50%) translateY(${(1 - cardIn) * 60}px) scale(${cardIn})`,
          width: 1000, padding: '50px 60px', borderRadius: 28,
          backgroundColor: '#FFFFFF', boxShadow: '0 30px 80px rgba(44,37,35,0.18)',
          border: '1px solid #ECE6DA', opacity: cardIn,
        }}>
          <div style={{ display: 'flex', alignItems: 'center', gap: 18, marginBottom: 26 }}>
            {/* SVG document icon */}
            <svg width="58" height="58" viewBox="0 0 24 24" fill="none">
              <rect x="4" y="2" width="16" height="20" rx="2" stroke="#2C2523" strokeWidth="1.6" />
              <line x1="8" y1="7" x2="16" y2="7" stroke="#5A524E" strokeWidth="1.6" />
              <line x1="8" y1="11" x2="16" y2="11" stroke="#5A524E" strokeWidth="1.6" />
              <line x1="8" y1="15" x2="13" y2="15" stroke="#5A524E" strokeWidth="1.6" />
            </svg>
            <div style={{ fontFamily: "'Inter', sans-serif", fontSize: 46, fontWeight: 700, color: '#5A524E', letterSpacing: 1 }}>RETIREMENT STATEMENT</div>
          </div>

          <div style={{ fontFamily: "'JetBrains Mono', monospace", fontSize: 170, fontWeight: 800, color: '#2C2523', lineHeight: 1 }}>
            ${liveVal.toLocaleString('en-US')}
          </div>

          <div style={{ display: 'flex', alignItems: 'center', gap: 14, marginTop: 22, opacity: arrowIn, transform: `translateX(${(1 - arrowIn) * -30}px)` }}>
            {/* down arrow icon */}
            <svg width="42" height="42" viewBox="0 0 24 24" fill="none">
              <path d="M12 4 L12 18 M12 18 L6 12 M12 18 L18 12" stroke="#A33" strokeWidth="2.2" strokeLinecap="round" strokeLinejoin="round" />
            </svg>
            <span style={{ fontFamily: "'Inter', sans-serif", fontSize: 50, fontWeight: 800, color: '#A33' }}>
              −${(startVal - endVal).toLocaleString('en-US')} this week
            </span>
          </div>
        </div>

        {/* Subhead */}
        <div style={{ position: 'absolute', bottom: 140, left: '50%', transform: `translateX(-50%) translateY(${(1 - subIn) * 30}px)`, opacity: subIn, textAlign: 'center', width: 1300 }}>
          <div style={{ fontFamily: "'Inter', sans-serif", fontSize: 51, fontWeight: 600, color: '#5A524E' }}>
            A quiet Friday morning. Coffee in hand. And a number that makes your stomach drop.
          </div>
        </div>
      </AbsoluteFill>

      {/* Vignette overlay */}
      <AbsoluteFill style={{ boxShadow: 'inset 0 0 320px rgba(44,37,35,0.22)', pointerEvents: 'none' }} />

      {/* Watermark "Failed Finances" bottom-right */}
      <div style={{ position: 'absolute', bottom: 30, right: 45, fontFamily: "'Inter', sans-serif", fontSize: 22, fontWeight: 600, color: '#9E9E9E', opacity: 0.3 }}>Failed Finances</div>

export default Scene01_PictureThis;
