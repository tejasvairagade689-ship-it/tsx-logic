import React from 'react';
import { useCurrentFrame, useVideoConfig, interpolate, Easing, spring, AbsoluteFill } from 'remotion';

export const compositionConfig = {
  id: 'Scene01TheBigQuestion',
  durationInSeconds: 26,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene01TheBigQuestion: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // Camera slow zoom
  const camScale = interpolate(frame, [0, durationInFrames], [1, 1.06]);

  // Beat 1: Question mark + "एक सवाल"
  const q1 = sp(10);
  const q1Op = interpolate(frame, [10, 30, 180, 210], [0, 1, 1, 0], { extrapolateRight: 'clamp', extrapolateLeft: 'clamp' });

  // Beat 2: 10% vs 90% Bar chart reveal
  const barStart = 220;
  const barOp = interpolate(frame, [barStart, barStart + 20, 480, 510], [0, 1, 1, 0], { extrapolateRight: 'clamp', extrapolateLeft: 'clamp' });
  const bar10W = interpolate(frame, [barStart + 10, barStart + 60], [0, 180], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp', easing: Easing.out(Easing.cubic) });
  const bar90W = interpolate(frame, [barStart + 40, barStart + 110], [0, 1620], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp', easing: Easing.out(Easing.cubic) });
  const pct10 = Math.round(interpolate(frame, [barStart + 20, barStart + 80], [0, 10], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' }));
  const pct90 = Math.round(interpolate(frame, [barStart + 50, barStart + 120], [0, 90], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' }));

  // Beat 3: Final punch question "क्या किया?"
  const punchStart = 520;
  const punch = sp(punchStart);
  const punchOp = interpolate(frame, [punchStart, punchStart + 20, durationInFrames - 20], [0, 1, 1], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' });
  const shake = frame > punchStart && frame < punchStart + 25 ? Math.sin(frame * 2) * 4 : 0;

  // Floating particles
  const particles = Array.from({ length: 35 }, (_, i) => {
    const seed = i + 1;
    const x = seededRandom(seed) * 1920;
    const y = seededRandom(seed * 2) * 1080;
    const drift = Math.sin(frame * 0.02 + i) * 15;
    const op = 0.15 + seededRandom(seed * 3) * 0.25;
    const size = 3 + seededRandom(seed * 4) * 6;
    return { x, y: y + drift, op, size };
  });

  // Drifting shape layers
  const shapeRot = frame * 0.15;

  return (
    <AbsoluteFill style={{ backgroundColor: '#E2E8CE', overflow: 'hidden', opacity: exitOp }}>
      {/* Gradient background overlay */}
      <AbsoluteFill style={{
        background: 'radial-gradient(circle at 30% 40%, rgba(8,6,22,0.08) 0%, rgba(226,232,206,0) 60%)',
      }} />

      {/* Shape Layer 1: Large drifting circle */}
      <div style={{
        position: 'absolute',
        top: -200, left: -200,
        width: 700, height: 700,
        borderRadius: '50%',
        border: '2px solid #262626',
        opacity: 0.05,
        transform: `rotate(${shapeRot}deg) scale(${1 + Math.sin(frame * 0.02) * 0.05})`,
      }} />

      {/* Shape Layer 2: Triangle */}
      <div style={{
        position: 'absolute',
        bottom: -150, right: -100,
        width: 0, height: 0,
        borderLeft: '300px solid transparent',
        borderRight: '300px solid transparent',
        borderBottom: '500px solid #080616',
        opacity: 0.04,
        transform: `rotate(${-shapeRot * 0.5}deg)`,
      }} />

      {/* Shape Layer 3: Rotating square outline */}
      <div style={{
        position: 'absolute',
        top: '40%', left: '70%',
        width: 400, height: 400,
        border: '3px solid #262626',
        opacity: 0.06,
        transform: `rotate(${shapeRot * 0.7}deg)`,
      }} />

      {/* Motion texture: grid lines */}
      <svg width="1920" height="1080" style={{ position: 'absolute', opacity: 0.08 }}>
        {Array.from({ length: 20 }).map((_, i) => (
          <line key={`v${i}`} x1={i * 96} y1={0} x2={i * 96} y2={1080} stroke="#262626" strokeWidth="1" />
        ))}
        {Array.from({ length: 12 }).map((_, i) => (
          <line key={`h${i}`} x1={0} y1={i * 90} x2={1920} y2={i * 90} stroke="#262626" strokeWidth="1" />
        ))}
      </svg>

      {/* Floating particles */}
      {particles.map((p, i) => (
        <div key={i} style={{
          position: 'absolute',
          left: p.x, top: p.y,
          width: p.size, height: p.size,
          borderRadius: '50%',
          backgroundColor: '#080616',
          opacity: p.op,
        }} />
      ))}

      {/* Main camera container */}
      <div style={{
        position: 'absolute',
        width: '100%', height: '100%',
        transform: `scale(${camScale}) translateX(${shake}px)`,
        transformOrigin: 'center',
      }}>

        {/* BEAT 1: Big Question Mark + Hindi headline */}
        <div style={{
          position: 'absolute',

          textAlign: 'center',
          width: '100%',

            fontSize: 380,
