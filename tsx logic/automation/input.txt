export const compositionConfig = {
  id: 'Scene1_Intro',
  durationInSeconds: 12,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  const scanLineY = interpolate(frame, [0, durationInFrames], [-20, 1120], { extrapolateRight: 'clamp' });

  const zoomScale = interpolate(frame, [0, durationInFrames], [1.08, 1.0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp', easing: Easing.out(Easing.quad),
  });

  const nameReveal = interpolate(frame, [10, 40], [0, 1], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp', easing: Easing.out(Easing.cubic),
  });

  const subtitleSlide = sp(25, { damping: 14, stiffness: 100 });
  const lineWidth = interpolate(frame, [20, 55], [0, 340], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp', easing: Easing.out(Easing.expo),
  });

  const breathe = Math.sin(frame * 0.05) * 2;
  const headNod = Math.sin(frame * 0.04) * 3;
  const figureEntry = sp(40, { damping: 10, stiffness: 80 });
  const glowPulse = interpolate(Math.sin(frame * 0.06), [-1, 1], [0.4, 0.8]);

  const particles = Array.from({ length: 18 }, (_, i) => ({
    x: seededRandom(i * 3 + 1) * 1920,
    y: seededRandom(i * 3 + 2) * 1080,
    r: seededRandom(i * 3 + 3) * 3 + 1,
    speed: seededRandom(i * 7) * 0.4 + 0.1,
    op: seededRandom(i * 11) * 0.25 + 0.05,

      {/* Gradient Background */}
      <AbsoluteFill style={{
        background: 'radial-gradient(ellipse at 20% 50%, #2a2200 0%, #1C1C1C 55%, #0d0d0d 100%)',
        transform: `scale(${zoomScale})`,
      }} />

      {/* Shape Layer — large drifting circle */}
      <AbsoluteFill>
        <svg width="1920" height="1080" style={{ position: 'absolute' }}>
          <circle
            cx={960 + Math.sin(frame * 0.008) * 30}
            cy={540 + Math.cos(frame * 0.006) * 20}
            r={480}
            fill="none"
            stroke="#e9d700"
            strokeWidth="1"
            opacity={0.04}
          />
          <circle
            cx={200 + Math.sin(frame * 0.012) * 15}
            cy={200}
            r={220}
            fill="#e9d700"
            opacity={0.03}
          />
          <rect
            x={1500 + Math.sin(frame * 0.01) * 10}
            y={700}
            width={300}
            height={300}
            fill="none"
            stroke="#F5F5F5"
            strokeWidth="1"
            opacity={0.03}
            transform={`rotate(${frame * 0.05} 1650 850)`}
          />
        </svg>
      </AbsoluteFill>

      {/* Motion Texture — diagonal grid lines */}
      <AbsoluteFill>
        <svg width="1920" height="1080" style={{ position: 'absolute', opacity: 0.035 }}>
          {Array.from({ length: 14 }, (_, i) => (

              x1={i * 140 - 100}
              y1={0}
              x2={i * 140 + 200}

              strokeWidth="1"
