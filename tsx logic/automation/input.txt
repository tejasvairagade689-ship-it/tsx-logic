  const rays = interpolate(sp(120), [0, 1], [0, 1]);

  const particles = Array.from({ length: 60 }, (_, i) => ({
    x: seededRandom(i + 1) * 1920,
    y: seededRandom(i + 100) * 1080,
    s: seededRandom(i + 200) * 5 + 1,
    sp: seededRandom(i + 300) * 0.6 + 0.1,
  }));

  return (
    <AbsoluteFill style={{ backgroundColor: '#FBF5DD', overflow: 'hidden', opacity: exitOp }}>
      {/* Gradient */}
      <AbsoluteFill style={{
        background: `radial-gradient(circle at 50% 50%, rgba(255,193,7,${0.1 + glow * 0.15}) 0%, rgba(251,245,221,0) 70%)`,
      }} />

      {/* Shapes */}
      <div style={{
        position: 'absolute', left: -200, top: -200,
        width: 600, height: 600, borderRadius: '50%',
        border: '3px solid rgba(31,21,12,0.05)',
      }} />
      <div style={{
        position: 'absolute', right: -200, bottom: -200,
        width: 700, height: 700,
        backgroundColor: 'rgba(255,193,7,0.05)',
        transform: `rotate(${frame * 0.1}deg)`,
      }} />

      {/* Radial rays */}
      <svg style={{ position: 'absolute', inset: 0, opacity: rays * 0.15 }} width="1920" height="1080">
        {Array.from({ length: 12 }).map((_, i) => {
          const angle = (i / 12) * Math.PI * 2 + frame * 0.005;
          return (
            <line key={i}
              x1={960 + Math.cos(angle) * 200}
              y1={540 + Math.sin(angle) * 200}
              x2={960 + Math.cos(angle) * 900}
              y2={540 + Math.sin(angle) * 900}
              stroke="#FFC107"
              strokeWidth="4"
            />
          );
        })}
      </svg>

      {/* Grid */}
      <svg style={{ position: 'absolute', inset: 0, opacity: 0.06 }} width="1920" height="1080">
        <defs>
          <pattern id="grid10" width="80" height="80" patternUnits="userSpaceOnUse">
            <path d="M 80 0 L 0 0 0 80" fill="none" stroke="#1F150C" strokeWidth="1" />
          </pattern>
        </defs>
        <rect width="1920" height="1080" fill="url(#grid10)" />
      </svg>

      {/* Particles */}
      {particles.map((p, i) => (
        <div key={i} style={{
          position: 'absolute',
          left: p.x,
          top: (p.y - frame * p.sp + 1080) % 1080,
          width: p.s, height: p.s,
          borderRadius: '50%',
          backgroundColor: i % 3 === 0 ? '#FFC107' : '#1F150C',
          opacity: 0.3,
        }} />
      ))}

      <AbsoluteFill style={{
        transform: `scale(${camScale})`,
      }}>
        {/* LEFT: Vault opening */}
        <div style={{
          position: 'absolute',
          left: 140, top: 280,
          opacity: vaultIn,
          transform: `scale(${vaultIn})`,
        }}>
          <div style={{
            position: 'relative',
            width: 460, height: 460,
            backgroundColor: '#1F150C',
            border: '10px solid #1F150C',
            borderRadius: 30,
          }}>
            {/* Inside light */}
            <div style={{
              position: 'absolute', inset: 20,
              backgroundColor: '#e9d700',
              boxShadow: `inset 0 0 80px rgba(255,193,7,${glow})`,
              display: 'flex', alignItems: 'center', justifyContent: 'center',
              fontSize: 153,
              fontFamily: 'Yantra One, Inter',
              fontWeight: 900,
              color: '#1F150C',
            }}>
              ₹
            </div>
            {/* Vault door (rotating away) */}
            <div style={{
              position: 'absolute', inset: 20,
              backgroundColor: '#2D2D2D',
              borderRadius: 8,
              transformOrigin: 'left center',
              transform: `perspective(800px) rotateY(${vaultRot}deg)`,
              display: 'flex', alignItems: 'center', justifyContent: 'center',
              flexDirection: 'column',
              gap: 20,
            }}>
              {/* Handle */}
              <div style={{
                width: 80, height: 80, borderRadius: '50%',
                border: '8px solid #FFC107',
                position: 'relative',
              }}>
                <div style={{
                  position: 'absolute', top: '50%', left: '50%',
                  width: 60, height: 8,
                  backgroundColor: '#FFC107',
                  transform: `translate(-50%, -50%) rotate(${frame * 4}deg)`,
                }} />
                <div style={{
                  position: 'absolute', top: '50%', left: '50%',
                  width: 8, height: 60,
                  backgroundColor: '#FFC107',
                  transform: `translate(-50%, -50%) rotate(${frame * 4}deg)`,
                }} />
              </div>
              <div style={{
                fontSize: 32, fontFamily: 'Inter', fontWeight: 800,
                color: '#FFC107', letterSpacing: 4,
              }}>VAULT</div>
            </div>
          </div>
        </div>

        {/* RIGHT: Title block */}
        <div style={{
          position: 'absolute',
          right: 100, top: 200,
          width: 1000,
          textAlign: 'right',
        }}>
          <div style={{
            opacity: t1,
            transform: `translateX(${(1 - t1) * 60}px)`,
            fontSize: 38,
            fontFamily: 'Inter',
            fontWeight: 700,
            color: '#ff0000',
            letterSpacing: 5,
            marginBottom: 25,
          }}>
            जब पैसा काम करना बंद कर दे
          </div>

          <div style={{
            opacity: t2,
            transform: `translateX(${(1 - t2) * 60}px)`,
            fontSize: 136,
            fontFamily: 'Yantra One, Inter',
            fontWeight: 900,
            color: '#1F150C',
            lineHeight: 0.95,
            letterSpacing: -2,
          }}>
            तब क्या <span style={{ color: '#ff0000' }}>काम</span>
            <br />आएगा?
          </div>

          <div style={{
            opacity: t3,
            transform: `translateX(${(1 - t3) * 40}px)`,
            marginTop: 50,
            display: 'inline-block',
            backgroundColor: '#1F150C',
            color: '#FBF5DD',
            padding: '20px 40px',
            fontSize: 51,

            fontWeight: 900,
            border: '4px solid #FFC107',
          }}>
            आज खोलेंगे यही सच
          </div>
        </div>
      </AbsoluteFill>

      {/* Channel badge bottom-left */}
      <div style={{
        position: 'absolute', bottom: 70, left: 120,
        opacity: sp(240),
        display: 'flex', alignItems: 'center', gap: 16,
      }}>
        <div style={{
          width: 60, height: 60,
          backgroundColor: '#FFC107',
          border: '4px solid #1F150C',
          display: 'flex', alignItems: 'center', justifyContent: 'center',
          fontSize: 38,
          fontFamily: 'Yantra One, Inter', fontWeight: 900,
          color: '#1F150C',
        }}>P</div>
        <div style={{
          fontSize: 38,
          fontFamily: 'Inter',
          fontWeight: 800,
          color: '#1F150C',
          letterSpacing: 2,
        }}>ProBuffet HQ</div>
      </div>

      {/* Vignette */}
      <AbsoluteFill style={{
        background: 'radial-gradient(circle, transparent 50%, rgba(31,21,12,0.3) 100%)',
        pointerEvents: 'none',
      }} />

      {/* Watermark */}
      <div style={{
        position: 'absolute', bottom: 30, right: 45,
        fontFamily: 'Inter', fontSize: 22, fontWeight: 600,
        color: '#1F150C', opacity: 0.3,
      }}>
        ProBuffet HQ
      </div>
    </AbsoluteFill>
  );
};
