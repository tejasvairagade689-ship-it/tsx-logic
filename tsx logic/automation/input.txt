import React from 'react';
import { useCurrentFrame, useVideoConfig, interpolate, Easing, spring, AbsoluteFill } from 'remotion';

export const compositionConfig = {
  id: 'Scene1_TheDrawer',
  durationInSeconds: 14,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene1_TheDrawer: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();
  
  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  const bgZoom = interpolate(frame, [0, durationInFrames], [1, 1.1]);
  const textTitleY = interpolate(sp(10), [0, 1], [100, 0]);
  const textSubY = interpolate(sp(25), [0, 1], [100, 0]);

  return (
    <AbsoluteFill style={{ backgroundColor: '#1C1C1C', overflow: 'hidden', opacity: exitOp }}>
      {/* Gradient + Shape Layers + Texture + Particles + Content + Character + Vignette + Watermark */}
      
      {/* Dynamic Gradient */}
      <AbsoluteFill style={{
        background: `radial-gradient(circle at 50% ${50 + Math.sin(frame*0.02)*10}%, #2D2D2D 0%, #1C1C1C 80%)`,
        transform: `scale(${bgZoom})`
      }} />

      {/* Shape Layers */}
      <AbsoluteFill>
        {[...Array(5)].map((_, i) => (
          <div key={i} style={{
            position: 'absolute',
            left: `${seededRandom(i)*100}%`,
            top: `${seededRandom(i+5)*100}%`,
            width: 400, height: 400,
            border: '20px solid #F5F5F5',
            opacity: 0.04,
            transform: `rotate(${frame * 0.1 * (i%2==0?1:-1)}deg) scale(${1 + Math.sin(frame*0.05 + i)*0.2})`
          }} />
        ))}
      </AbsoluteFill>

      {/* Motion Texture: Diagonal Lines */}
      <AbsoluteFill style={{
        backgroundImage: 'repeating-linear-gradient(45deg, transparent, transparent 10px, rgba(255,255,255,0.02) 10px, rgba(255,255,255,0.02) 20px)',

          fontFamily: 'Yantra One',
          fontSize: '220px',
          color: '#ff0000',
          textTransform: 'uppercase',
          margin: 0,
          lineHeight: 1.1,
          textAlign: 'center',
          transform: `translateY(${textTitleY}px) scale(${sp(10)})`,
