import React from 'react';
import { useCurrentFrame, useVideoConfig, interpolate, Easing, spring, AbsoluteFill } from 'remotion';

export const compositionConfig = {
  id: 'Scene1_Intro',
  durationInSeconds: 12,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene1_Intro: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // Cinematic curtain reveal
  const curtainLeft = interpolate(frame, [0, 25], [0, -960], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
    easing: Easing.out(Easing.cubic),
  });
  const curtainRight = interpolate(frame, [0, 25], [0, 960], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
    easing: Easing.out(Easing.cubic),
  });

  // Floating particles
  const particles = Array.from({ length: 18 }, (_, i) => ({
    x: seededRandom(i * 31) * 1920,
    y: seededRandom(i * 17) * 1080,
    size: seededRandom(i * 7) * 4 + 2,
    speed: seededRandom(i * 13) * 0.4 + 0.1,
    opacity: seededRandom(i * 23) * 0.35 + 0.05,
  }));

  // Text reveal
  const line1Op = interpolate(frame, [28, 42], [0, 1], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' });
  const line1Y = sp(28) * 0;
  const line2Op = interpolate(frame, [40, 55], [0, 1], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' });

  // Ambient pulse
  const pulse = Math.sin(frame * 0.06) * 0.04 + 1;

  // Character entrance
  const charX = sp(45, { damping: 14, stiffness: 100 });
  const charBreathe = Math.sin(frame * 0.05) * 2;
  const headNod = Math.sin(frame * 0.04) * 3;

  return (
    <AbsoluteFill style={{ backgroundColor: '#1C1C1C', overflow: 'hidden', opacity: exitOp }}>

      {/* Deep gradient background */}
      <AbsoluteFill style={{
        background: 'radial-gradient(ellipse at 20% 50%, #2a1f00 0%, #1C1C1C 55%, #0a0a0a 100%)',
      }} />

      {/* Animated shape layer 1 - large drifting circle */}
      <AbsoluteFill style={{ position: 'absolute' }}>
        <div style={{
          position: 'absolute',
          width: 700,
          height: 700,
          borderRadius: '50%',
          border: '1px solid rgba(233,215,0,0.06)',
          left: -100,
          top: 180,
          transform: `scale(${pulse}) rotate(${frame * 0.08}deg)`,
        }} />
        <div style={{
          position: 'absolute',
          width: 400,
          height: 400,
          borderRadius: '50%',
          border: '1px solid rgba(233,215,0,0.04)',
          left: 50,
          top: 330,
          transform: `scale(${1 / pulse}) rotate(${-frame * 0.05}deg)`,
        }} />
      </AbsoluteFill>

      {/* Shape layer 2 - geometric rectangle drift */}
      <AbsoluteFill style={{ position: 'absolute' }}>
        <div style={{
          position: 'absolute',
          width: 320,
          height: 180,
          border: '1px solid rgba(255,255,255,0.03)',
          right: 120,
          top: 80,
          transform: `rotate(${12 + Math.sin(frame * 0.03) * 3}deg)`,
          opacity: 0.6,
        }} />
        <div style={{
          position: 'absolute',
          width: 180,
          height: 320,
          border: '1px solid rgba(255,193,7,0.04)',
          right: 200,
          bottom: 120,
          transform: `rotate(${-8 + Math.sin(frame * 0.04) * 2}deg)`,
        }} />
      </AbsoluteFill>

      {/* Motion texture - diagonal lines grid */}
      <AbsoluteFill style={{ position: 'absolute', opacity: 0.03 }}>
        <svg width="1920" height="1080">
          {Array.from({ length: 30 }, (_, i) => (
            <line
              key={i}
              x1={i * 64 - (frame * 0.3 % 64)}
              y1="0"
              x2={i * 64 - (frame * 0.3 % 64) - 200}
              y2="1080"
              stroke="#e9d700"
              strokeWidth="0.8"
            />
          ))}
        </svg>
      </AbsoluteFill>

      {/* Floating particles */}
      <AbsoluteFill style={{ position: 'absolute' }}>
        {particles.map((p, i) => (
          <div key={i} style={{
            position: 'absolute',
            left: (p.x + frame * p.speed * 15) % 1920,
            top: p.y - frame * p.speed * 8,
            width: p.size,
            height: p.size,
            borderRadius: '50%',
            backgroundColor: i % 3 === 0 ? '#e9d700' : '#F5F5F5',
            opacity: p.opacity,
          }} />
        ))}
      </AbsoluteFill>

      {/* Cinematic curtain */}
      <div style={{
        position: 'absolute',
        left: curtainLeft,
        top: 0,
        width: 960,
        height: 1080,

        zIndex: 50,

        position: 'absolute',

        zIndex: 50,

        <div style={{
          opacity: line1Op,
          fontFamily: 'Inter, sans-serif',
          fontSize: 28,
          fontWeight: 700,
          color: '#e9d700',
          letterSpacing: 8,
          textTransform: 'uppercase',
          marginBottom: 20,
        }}>
          एक कहानी
        </div>

        {/* Hero text */}
        <div style={{
          opacity: line1Op,
          transform: `translateY(${(1 - sp(28)) * 40}px)`,
          fontFamily: 'Noto Sans Devanagari, sans-serif',
          fontSize: 96,
          fontWeight: 900,
          color: '#F5F5F5',
          lineHeight: 1.1,

          मैं आपको एक<br />
