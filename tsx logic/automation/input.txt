  fps: 30,
  width: 1920,
  height: 1080,
};

  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // Camera slow zoom-in
  const camScale = interpolate(frame, [0, durationInFrames], [1, 1.08], { extrapolateRight: 'clamp' });

  // Word-by-word reveal of HERO
  const heroWord1 = sp(15);
  const heroWord2 = sp(35);
  const heroWord3 = sp(55);

  // Subline reveal
  const subOp = interpolate(sp(90), [0, 1], [0, 1]);
  const subY = interpolate(sp(90), [0, 1], [40, 0]);

  // Underline bar
  const barW = interpolate(sp(110), [0, 1], [0, 720]);

  // Caption bottom
  const capOp = interpolate(sp(140), [0, 1], [0, 1]);

  // Shape breathing
  const breath = Math.sin(frame * 0.03) * 8;
  const rot1 = frame * 0.15;
  const rot2 = -frame * 0.1;

  // Running figures motion (no stick character — abstract silhouettes as concept)
  const runOffset = (frame * 6) % 1920;

  // Quote marks scale pop
  const quotePop = sp(5);

  // Screen flash on hero land
  const flash = interpolate(frame, [15, 25], [0.25, 0], { extrapolateRight: 'clamp' });

  // Particles
  const particles = Array.from({ length: 28 }).map((_, i) => {
    const sx = seededRandom(i + 1) * 1920;
    const sy = seededRandom(i + 2) * 1080;
    const speed = 0.3 + seededRandom(i + 3) * 0.8;
    const size = 2 + seededRandom(i + 4) * 4;
    const y = (sy - frame * speed * 3 + 1080) % 1080;
    return { x: sx, y, size, op: 0.15 + seededRandom(i + 5) * 0.3 };
  });

  return (
    <AbsoluteFill style={{ backgroundColor: '#FF5722', overflow: 'hidden', opacity: exitOp, transform: `scale(${camScale})` }}>
      {/* Dynamic gradient background */}
      <AbsoluteFill
        style={{

        }}
      />

      {/* Shape Layer 1 — big drifting circle */}
      <div
        style={{
          position: 'absolute',
          top: -200 + breath,
          right: -150,
          width: 700,
          height: 700,
          borderRadius: '50%',
          border: '3px solid #FAFAFA',
          opacity: 0.07,
          transform: `rotate(${rot1}deg)`,
        }}
      />

      {/* Shape Layer 2 — rectangle */}
      <div
        style={{
          position: 'absolute',
          bottom: -100,
          left: -120,
          width: 520,
          height: 520,
          backgroundColor: '#FFF9D2',
          opacity: 0.05,
          transform: `rotate(${rot2}deg)`,
        }}
      />

      {/* Shape Layer 3 — triangle */}
      <svg
        style={{ position: 'absolute', top: 200, left: 100, opacity: 0.06, transform: `rotate(${frame * 0.2}deg)` }}
        width="300"
        height="300"
        viewBox="0 0 100 100"
      >
        <polygon points="50,5 95,90 5,90" fill="none" stroke="#FAFAFA" strokeWidth="2" />
      </svg>

      {/* Motion texture — diagonal lines grid */}
      <svg style={{ position: 'absolute', inset: 0, opacity: 0.08 }} width="1920" height="1080">
        <defs>
          <pattern id="diag" width="60" height="60" patternUnits="userSpaceOnUse" patternTransform={`translate(${-frame * 0.5} 0)`}>
            <path d="M0 60 L60 0" stroke="#FFF9D2" strokeWidth="1" />
          </pattern>
