import React from 'react';

import { useCurrentFrame, useVideoConfig, interpolate, Easing, spring, AbsoluteFill } from 'remotion';

export const compositionConfig = {
  id: 'Scene01_Stop',
  durationInSeconds: 18,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene01_Stop: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // Camera slow zoom
  const camZoom = interpolate(frame, [0, durationInFrames], [1.05, 1.0], { extrapolateRight: 'clamp' });

  // STOP word impact
  const stopScale = sp(8, { stiffness: 200, damping: 9 });
  const shake = frame < 30 ? Math.sin(frame * 2.5) * (1 - frame / 30) * 10 : 0;

  // Red bar wipe
  const barW = interpolate(sp(20), [0, 1], [0, 1]);

  // Subline reveal
  const subOp = interpolate(frame, [40, 60], [0, 1], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' });
  const subY = interpolate(sp(40), [0, 1], [40, 0]);

  // "2026" stamp
  const stampScale = sp(70, { stiffness: 180, damping: 8 });

  return (
    <AbsoluteFill style={{ backgroundColor: '#DDDDDD', overflow: 'hidden', opacity: exitOp }}>
      {/* Gradient background */}
      <AbsoluteFill style={{ background: 'radial-gradient(circle at 50% 40%, #E8E8E8 0%, #DDDDDD 55%, #C9C9C9 100%)' }} />

      {/* Animated shape layers */}
      <AbsoluteFill style={{ transform: `scale(${camZoom})` }}>
        {[0, 1, 2, 3].map((i) => {
          const s = seededRandom(i + 1);
          const rot = frame * 0.15 * (i % 2 ? 1 : -1);
          const breathe = 1 + Math.sin(frame * 0.03 + i) * 0.05;
          return (
            <div
              key={i}
              style={{
                position: 'absolute',
                left: `${s * 80}%`,
                top: `${seededRandom(i + 9) * 80}%`,
                width: 300 + s * 250,
                height: 300 + s * 250,
                border: `2px solid #000000`,
                borderRadius: i % 2 ? '50%' : '0%',
                opacity: 0.05,
                transform: `rotate(${rot}deg) scale(${breathe})`,
              }}
            />
          );
        })}
      </AbsoluteFill>

      {/* Motion texture — vertical grid lines */}
      <AbsoluteFill>
        {Array.from({ length: 14 }).map((_, i) => (
          <div
            key={i}
            style={{
              position: 'absolute',
              left: `${(i / 14) * 100}%`,
              top: 0,
              width: 1,
              height: '100%',
              backgroundColor: '#000000',
              opacity: 0.04,
              transform: `translateY(${Math.sin(frame * 0.04 + i) * 12}px)`,
            }}
          />
        ))}
      </AbsoluteFill>

      {/* Floating particles */}
      <AbsoluteFill>
        {Array.from({ length: 22 }).map((_, i) => {
          const s = seededRandom(i + 30);
          const y = (seededRandom(i + 60) * 100 + frame * 0.25 * (0.5 + s)) % 100;
          return (
            <div
              key={i}
              style={{
                position: 'absolute',
                left: `${s * 100}%`,
                top: `${y}%`,
                width: 4 + s * 6,
                height: 4 + s * 6,
                borderRadius: '50%',
                backgroundColor: i % 3 === 0 ? '#F62440' : '#000000',
                opacity: 0.18,
              }}
            />
          );
        })}
      </AbsoluteFill>

      {/* Main content */}
      <AbsoluteFill style={{ alignItems: 'center', justifyContent: 'center', transform: `translateX(${shake}px) scale(${camZoom})` }}>
        <div style={{ textAlign: 'center' }}>
          {/* Red bar behind STOP */}
          <div
            style={{
              position: 'absolute',
              left: '50%',
              top: '50%',
              width: `${barW * 760}px`,
              height: 250,
              backgroundColor: '#F62440',
              transform: 'translate(-50%, -50%) skewX(-6deg)',
              opacity: 0.95,
            }}
          />
          <div
            style={{
              position: 'relative',
              fontFamily: 'Anton, sans-serif',
              fontSize: 240,
              lineHeight: 1,
              color: '#000000',
              transform: `scale(${stopScale})`,
              letterSpacing: 4,
            }}
          >
            STOP
          </div>

          <div
            style={{
              marginTop: 30,
              opacity: subOp,
              transform: `translateY(${subY}px)`,
              fontFamily: 'Montserrat, sans-serif',
              fontWeight: 700,
              fontSize: 60,
              color: '#000000',
              maxWidth: 1300,
            }}
          >
            Ce que je vais dire change votre{' '}
            <span style={{ color: '#F62440' }}>fin de mois</span>
          </div>

          {/* 2026 stamp */}
