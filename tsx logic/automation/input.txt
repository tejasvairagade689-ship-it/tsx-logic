import React from 'react';
import { useCurrentFrame, useVideoConfig, interpolate, Easing, spring, AbsoluteFill } from 'remotion';

// Exporting standard Remotion composition configuration
export const compositionConfig = {
  id: 'Scene1_FreeWork',
  durationInSeconds: 8,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene1_FreeWork: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();
  
  // Spring animation for bouncy entrances
  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  // Interpolate opacity for exit animation during last 15 frames
  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  const bgGradient = `linear-gradient(${frame * 0.5}deg, #1C1C1C 0%, #2D2D2D 100%)`;
  
  const textScale = sp(15);
  const textY = interpolate(sp(15), [0, 1], [100, 0]);
  
  const highlightScale = sp(30);

  return (
    <AbsoluteFill style={{ background: bgGradient, overflow: 'hidden', opacity: exitOp }}>
      
      {/* Animated Shape Layers */}
      <div style={{
        position: 'absolute', width: 800, height: 800, borderRadius: '50%',
        backgroundColor: '#ff0000', opacity: 0.05,
        transform: `translate(${Math.sin(frame * 0.02) * 100 - 200}px, ${Math.cos(frame * 0.02) * 100 - 200}px) scale(${1 + Math.sin(frame * 0.05) * 0.1})`
      }} />
      <div style={{
        position: 'absolute', width: 600, height: 600, 
        backgroundColor: '#FFC107', opacity: 0.03, right: -100, bottom: -100,
        transform: `rotate(${frame * 0.5}deg)`
      }} />

      {/* Motion Texture: Grid */}
      <div style={{
        position: 'absolute', top: 0, left: 0, right: 0, bottom: 0,
        backgroundImage: `radial-gradient(#9E9E9E 1px, transparent 1px)`,
        backgroundSize: '40px 40px', opacity: 0.1,
        transform: `translateY(${frame * 0.5}px)`
      }} />

      {/* Floating Particles */}
      {[...Array(15)].map((_, i) => (
        <div key={i} style={{
          position: 'absolute', width: 4, height: 4, backgroundColor: '#F5F5F5', borderRadius: '50%',
          left: `${seededRandom(i) * 100}%`,
          top: `${(seededRandom(i + 10) * 100) - (frame * (seededRandom(i) * 2 + 1))}%`,
          opacity: 0.4
        }} />
      ))}

      {/* Main Content */}
      <div style={{
        flex: 1, display: 'flex', flexDirection: 'column', justifyContent: 'center', alignItems: 'center',
        transform: `scale(${textScale}) translateY(${textY}px)`
      }}>
        <h1 style={{
          fontFamily: 'Yantra One', fontSize: '140px', color: '#F5F5F5', margin: 0,
          textShadow: '0 10px 30px rgba(0,0,0,0.5)', textAlign: 'center', lineHeight: 1.1
        }}>
          साल के <span style={{ color: '#FFC107' }}>12 महीनों</span> में से
        </h1>
        <div style={{
          transform: `scale(${highlightScale})`, backgroundColor: '#ff0000',
          padding: '20px 60px', marginTop: '40px', borderRadius: '20px',
          boxShadow: '0 20px 50px rgba(255,0,0,0.3)'
        }}>
          <h2 style={{
            fontFamily: 'Yantra One', fontSize: '160px', color: '#F5F5F5', margin: 0,
            textShadow: '0 5px 15px rgba(0,0,0,0.5)'
          }}>
            4 महीने मुफ़्त काम!
          </h2>
        </div>
      </div>

      {/* Character Stick Figure */}
      <div style={{
        position: 'absolute', bottom: -20, left: 80,
        transform: `translateY(${Math.sin(frame * 0.05) * 2}px) rotate(${Math.sin(frame * 0.04) * 3}deg) translateY(${interpolate(sp(40), [0, 1], [400, 0])}px)`,
        filter: 'drop-shadow(0 0 30px rgba(255, 193, 7, 0.4))'
      }}>
        <svg width="250" height="400" viewBox="0 0 200 400">
          <circle cx="100" cy="80" r="50" fill="none" stroke="#F5F5F5" strokeWidth="8" />
          {/* Golden Glasses */}
          <rect x="65" y="65" width="30" height="20" rx="5" fill="none" stroke="#FFC107" strokeWidth="6" />
          <rect x="105" y="65" width="30" height="20" rx="5" fill="none" stroke="#FFC107" strokeWidth="6" />
          <line x1="95" y1="75" x2="105" y2="75" stroke="#FFC107" strokeWidth="4" />
